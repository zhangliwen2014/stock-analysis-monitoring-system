## 要防的错（写端点的人先读这一节）

1. **D113 线程不变量**：同步端点经 `run_in_threadpool` 执行，FastAPI 的**依赖与端点可能落在不同线程**。所以：依赖（`csrf_protect`、`form_data`）只碰 `request`；`require_user` / `require_admin` 是普通函数，在**端点函数体内**调用；一切仓库/服务调用都在端点体内经 `request.app.state.contexts.current()` 拿本线程上下文。**写端点一律 `def`，不许 `async def`**——async 会把并发请求挤到同一条线程的同一个上下文上。
2. CSRF：所有 POST 都要 `Depends(csrf_protect)`，唯一例外是 `/login`（那时还没有 cookie 可比对）。
3. 归属闸：读别人的资源显示 **404**；`alerts_store.set_status` 的 `user_id` 必传（repo 会挡）。
4. 改密码要吊销**其它**会话：`sessions.revoke_others(user_id, keep=request.state.sams_token)`；管理员重置才用 `revoke_all`。
5. 驳回告警必须写原因：repo 会挡，但页面要**先**检查、把话说到人能懂。
6. `/admin/sync` 的 `market` 必须在 `settings.markets` 里；不在就回错误，不许把非法市场塞进队列。
7. 立即清理必须**二次确认**：表单里 `confirm` 字段等于字面 `yes` 才调 `request_prune`。
8. 测试告警每用户 5 分钟一次：`WebState.test_alerts`（`RateCard(300)`）按 user_id 限。
9. K 线 JSON 与页面显示**不许**引用外部 URL；模板里不写 emoji。
10. 500 处理器的测试要用 `TestClient(app, raise_server_exceptions=False)` 并在测试里自加一个会炸的路由。
11. `tests/web/` 下**不建** `support.py`（`pythonpath=["tests"]` 下会遮住根目录的 `support`）；一切替身住 `tests/web/conftest.py`。
12. 简单表单用标量 `Form()`；有重复键（如 `channel_ids`）或动态参数的表单用 `form_data` 依赖拿 `dict[str, list[str]]`。两者可以共存：Starlette 缓存 `request.form()`，解析只发生一次。

```python
"""线程级上下文的登记处。

`build_context` 每造一台就开一条 DuckDB 连接（D113），而 HTTP 线程池的线程不固定，
不能预造几条塞给「某条线程」。所以按「第一次在这条线程上用到」装配；threading.local
的值无法从别的线程枚举，登记只能靠一把锁加一份列表，thread-local 只是快路径。
"""

from __future__ import annotations

import threading

from sams.bootstrap import AppCore, ServiceContext, build_context


class ThreadContexts:
    def __init__(self, core: AppCore, *, providers: dict | None = None) -> None:
        self._core = core
        self._providers = providers
        self._local = threading.local()
        self._lock = threading.Lock()
        self._all: list[ServiceContext] = []

    def current(self) -> ServiceContext:
        ctx = getattr(self._local, "ctx", None)
        if ctx is None:
            ctx = build_context(self._core, providers=self._providers)
            self._local.ctx = ctx
            with self._lock:
                self._all.append(ctx)
        return ctx

    @property
    def count(self) -> int:
        with self._lock:
            return len(self._all)

    def close_all(self) -> None:
        with self._lock:
            contexts, self._all = self._all, []
        for ctx in contexts:
            ctx.close()
```

```python
"""M4-A 的 HTTP 入口。

`create_app(runtime)`（不传 core）保留计划 1 的形状：只有 /healthz，应用不认识
依赖装配——看门狗与最简部署用。传 core 与 contexts 才有页面。

线程不变量（D113）：同步端点经 run_in_threadpool 执行，依赖与端点可能落在不同线程。
所以依赖只许碰 request；一切仓库与服务调用都写在端点函数体内，经
`request.app.state.contexts.current()` 拿本线程上下文。
"""

from __future__ import annotations

from contextlib import asynccontextmanager
from dataclasses import dataclass
from pathlib import Path

from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
from fastapi.staticfiles import StaticFiles
from starlette.templating import Jinja2Templates

from sams.bootstrap import AppCore
from sams.services.runtime import Runtime
from sams.web.context import ThreadContexts

_STATIC = Path(__file__).parent / "static"
_TEMPLATES = Path(__file__).parent / "templates"


@dataclass
class WebState:
    """一张页面要用的进程级东西。仓库不在这里：它们属于线程，走 `contexts.current()`。"""

    runtime: Runtime
    core: AppCore
    contexts: ThreadContexts
    templates: Jinja2Templates


def create_app(
    runtime: Runtime,
    core: AppCore | None = None,
    *,
    contexts: ThreadContexts | None = None,
) -> FastAPI:
    @asynccontextmanager
    async def lifespan(_app: FastAPI):
        yield
        if contexts is not None:
            contexts.close_all()

    app = FastAPI(
        title="SAMS",
        docs_url=None,
        redoc_url=None,
        openapi_url=None,
        lifespan=lifespan,
    )

    @app.get("/healthz")
    def healthz() -> JSONResponse:
        return JSONResponse(runtime.healthz(), status_code=200)

    if core is None:
        return app
    if contexts is None:
        raise ValueError("给了 core 就必须给 contexts：页面要按线程装配上下文")

    app.state.web = WebState(
        runtime=runtime,
        core=core,
        contexts=contexts,
        templates=Jinja2Templates(directory=str(_TEMPLATES)),
    )
    app.mount("/static", StaticFiles(directory=str(_STATIC)), name="static")
    return app
```

```python
"""登录、会话与 CSRF。

三枚 cookie（spec §6.2）：`sams_session` 是会话令牌（库里存哈希）；`sams_csrf`
是签过名的随机串，表单里回填一份，POST 时比对；`sams_active` 是活跃心跳——
require_user 每次盖章，中间件在响应上写回，过期就会话被吊销。
签名用 HMAC 只防伪造不防偷看：cookie 里没有值得读的东西。
"""

from __future__ import annotations

import hmac
import logging
import secrets
import time
from datetime import UTC, datetime, timedelta

from fastapi import APIRouter, Depends, Form, HTTPException, Request
from fastapi.responses import RedirectResponse

from sams.storage.repos.users import ACTIVE

logger = logging.getLogger("sams.web")

SESSION_COOKIE = "sams_session"
CSRF_COOKIE = "sams_csrf"
ACTIVE_COOKIE = "sams_active"


def sign(secret: str, value: str) -> str:
    digest = hmac.new(secret.encode(), value.encode(), "sha256").hexdigest()
    return f"{value}.{digest}"


def unsigned_ok(secret: str, signed: str) -> str | None:
    value, _, digest = signed.rpartition(".")
    if not value or not digest:
        return None
    expected = hmac.new(secret.encode(), value.encode(), "sha256").hexdigest()
    return value if hmac.compare_digest(digest, expected) else None


class LoginRequired(Exception):
    """require_user 没认出人：交给 app.py 的 handler 送去登录页。"""


class LoginLimiter:
    """按 (来源 IP, 用户名) 数失败；达到上限锁一阵子，把「慢慢试」的通道关掉。"""

    def __init__(self, *, max_failures: int, window_s: int, now=time.monotonic) -> None:
        self.max_failures = max_failures
        self.window_s = window_s
        self._now = now
        self._fails: dict[tuple[str, str], list[float]] = {}

    def blocked_for(self, key: tuple[str, str]) -> int:
        cutoff = self._now() - self.window_s
        tries = [t for t in self._fails.get(key, []) if t > cutoff]
        if len(tries) < self.max_failures:
            return 0
        return max(1, int(min(tries) + self.window_s - self._now()))

    def record_failure(self, key: tuple[str, str]) -> None:
        cutoff = self._now() - self.window_s
        tries = [t for t in self._fails.get(key, []) if t > cutoff]
        tries.append(self._now())
        self._fails[key] = tries

    def reset(self, key: tuple[str, str]) -> None:
        self._fails.pop(key, None)


class RateCard:
    """固定间隔的令牌：try_take 返回 0 表示放行，否则返回要等的秒数。"""

    def __init__(self, interval_s: int, *, now=time.monotonic) -> None:
        self.interval_s = interval_s
        self._now = now
        self._last: dict[str, float] = {}

    def try_take(self, key: str) -> int:
        now = self._now()
        last = self._last.get(key)
        if last is not None and now - last < self.interval_s:
            return int(self.interval_s - (now - last)) + 1
        self._last[key] = now
        return 0


def client_ip(request: Request) -> str:
    return request.client.host if request.client else "?"


def csrf_value(request: Request) -> str:
    raw = request.cookies.get(CSRF_COOKIE)
    if not raw:
        return ""
    return unsigned_ok(request.app.state.web.core.settings.secret_key, raw) or ""


async def csrf_protect(request: Request, csrf: str = Form("")) -> None:
    """CSRF 依赖。登录 POST 豁免（那时还没有 cookie 可比对）。"""
    expected = csrf_value(request)
    if not expected or not hmac.compare_digest(csrf, expected):
        raise HTTPException(status_code=403, detail="CSRF 校验失败：请刷新页面重试")


def audit(request: Request, *, action: str, target: str, ok: bool, detail: str = "") -> None:
    user = getattr(request.state, "sams_user", None)
    who = user.username if user is not None else "-"
    logger.info(
        "%s %s user=%s target=%s %s%s",
        "ok" if ok else "deny",
        action,
        who,
        target,
        client_ip(request),
        f" {detail}" if detail else "",
    )


def require_user(request: Request):
    """端点体内调用（不是依赖）：依赖与端点可能落在不同线程（D113）。"""
    web = request.app.state.web
    token = request.cookies.get(SESSION_COOKIE)
    ctx = web.contexts.current()
    user_id = ctx.sessions.verify(token) if token else None
    if user_id is None:
        raise LoginRequired()
    user = ctx.users.get(user_id)
    if user is None or user.status != ACTIVE:
        raise LoginRequired()
    if not _active_ok(request):
        ctx.sessions.revoke(token)
        raise LoginRequired()
    request.state.sams_user = user
    request.state.sams_token = token
    return user


def _active_ok(request: Request) -> bool:
    raw = request.cookies.get(ACTIVE_COOKIE)
    if not raw:
        return False
    value = unsigned_ok(request.app.state.web.core.settings.secret_key, raw)
    if value is None:
        return False
    stamped, _, _nonce = value.rpartition(":")
    try:
        moment = datetime.fromisoformat(stamped)
    except ValueError:
        return False
    idle = timedelta(minutes=request.app.state.web.core.settings.server.idle_timeout_min)
    return datetime.now(UTC) - moment <= idle


def require_admin(request: Request):
    user = require_user(request)
    if not user.is_admin:
        raise HTTPException(status_code=403, detail="这一页只有管理员能开")
    return user

def login(
    request: Request,
    username: str = Form(""),
    password: str = Form(""),
):
    web = request.app.state.web
    settings = web.core.settings.server
    key = (client_ip(request), username.strip().lower())
    wait = web.limiter.blocked_for(key)
    if wait:
        response = web.render(
            request, "login.html", status_code=429, title="登录",
            error=f"失败次数太多，{wait} 秒后再试",
        )
        response.headers["Retry-After"] = str(wait)
        return response
    user = web.contexts.current().users.authenticate(username, password)
    if user is None:
        web.limiter.record_failure(key)
        audit(request, action="login", target=username, ok=False)
        return web.render(
            request, "login.html", status_code=200, title="登录", error="用户名或密码不对"
        )
    web.limiter.reset(key)
    token = web.contexts.current().sessions.issue(
        user.id, ttl_days=settings.session_ttl_days
    )
    audit(request, action="login", target=user.username, ok=True)
    response = RedirectResponse("/", status_code=303)
    _set_session_cookies(response, request, token)
    return response
```

```python
"""个股页：一只票的全部现场——行情、K 线、挂在它身上的规则与论、最近的告警。

规则筛选认 `rule.target_type == "symbol"` 且 `target_ref == 市场/代码`：
这是 `repos/rules.py` 的 `_require_target_shape` 定死的形状。
"""

from __future__ import annotations

from datetime import timedelta

from fastapi import APIRouter, HTTPException, Request
from fastapi.responses import JSONResponse

from sams.core.errors import NotAdjustable, SamsError
from sams.core.quotes import NO_DATA, quote_state
from sams.web.auth import require_user
from sams.web.display import hms, pct, pct_class, state_label
from sams.web.pages.dashboard import _market_open

router = APIRouter()


def _require_symbol(request: Request, market: str, code: str):
    """市场没配或库里没有这只票都是 404：不给探测者区分「没有」与「不是你的」。"""
    web = request.app.state.web
    ctx = web.contexts.current()
    if market not in ctx.settings.markets:
        raise HTTPException(status_code=404)
    row = ctx.symbols.get(market, code)
    if row is None:
        raise HTTPException(status_code=404)
    return ctx, row


@router.get("/symbol/{market}/{code}")
def symbol_page(request: Request, market: str, code: str):
    user = require_user(request)
    web = request.app.state.web
    market = market.upper()
    ctx, symbol = _require_symbol(request, market, code)
    core = web.core
    now = core.clock.now_utc()

    quote = core.board.get(market, code)
    state = NO_DATA
    if quote is not None:
        state = quote_state(
            quote,
            now=now,
            market_open=_market_open(ctx, market, now),
            stale_s=ctx.settings.quote_stale_s,
        )

    ref = f"{market}/{code}"
    rules = [
        view
        for view in ctx.rule_service.list(user.id)
        if view.rule.target_type == "symbol" and view.rule.target_ref == ref
    ]
    theses = [
        thesis
        for thesis in ctx.theses.list(user.username)
        if thesis.market == market and thesis.code == code
    ]
    alerts = ctx.alerts_store.list_for(user.id, market=market, code=code, limit=10)

    return web.render(
        request,
        "symbol.html",
        title=f"{market}/{code}",
        market=market,
        code=code,
        symbol=symbol,
        price=quote.price if quote else None,
        quote_pct=quote.pct_chg if quote else None,
        pct_class=pct_class(quote.pct_chg if quote else None),
        state=state_label(state),
        quote_at=hms(quote.ts if quote else None),
        rules=rules,
        theses=theses,
        alerts=alerts,
    )


@router.get("/api/symbol/{market}/{code}/bars")
def bars_api(request: Request, market: str, code: str, days: int = 120):
    require_user(request)
    web = request.app.state.web
    market = market.upper()
    ctx, _symbol = _require_symbol(request, market, code)
    end = web.core.clock.now_utc().date()
    start = end - timedelta(days=max(1, min(days, 2000)))
    try:
        read = ctx.analysis.series(market, code, start=start, end=end, indicators=("ma",))
    except SamsError as exc:
        return JSONResponse(
            {"dates": [], "close": [], "ma20": [], "ma60": [], "note": str(exc)}
        )
    table = read.table
    return JSONResponse(
        {
            "dates": [str(value) for value in table["date"].to_list()],
            "close": table["close"].to_list(),
            "ma20": table["ma20"].to_list() if "ma20" in table.columns else [],
            "ma60": table["ma60"].to_list() if "ma60" in table.columns else [],
            "note": None,
        }
    )
```

```python
"""持仓与成交录入。录入走的是 `LedgerService` 的同一扇门，CLI 与页面没有第二套账。"""

from __future__ import annotations

from fastapi import APIRouter, Depends, Form, Request
from fastapi.responses import RedirectResponse

from sams.core.errors import SamsError, ValidationError
from sams.web.auth import csrf_protect, require_user
from sams.web.display import money

router = APIRouter()


def _number(raw: str, label: str) -> float:
    try:
        return float(raw)
    except ValueError as exc:
        raise ValidationError(f"{label} 要是数字，收到 {raw!r}") from exc


def _account_view(ctx, username: str, account) -> dict:
    summary = ctx.portfolio.summary(username, account.id)
    positions = [
        {
            "market": row.market,
            "code": row.code,
            "qty": row.qty,
            "avg_cost": row.avg_cost,
            "last_price": row.last_price,
            "price_pnl": row.price_pnl,
            "fx_pnl": row.fx_pnl,
            "total": row.total_report,
            "convertible": row.convertible,
            "note": row.unavailable_note,
        }
        for row in ctx.portfolio.positions(username, account.id)
    ]
    return {
        "id": account.id,
        "name": summary.name,
        "currency": summary.report_currency,
        "cash": summary.cash_by_currency,
        "total": summary.total_report,
        "price_pnl": summary.price_pnl_report,
        "fx_pnl": summary.fx_pnl,
        "realized": summary.realized_report,
        "as_of": summary.as_of,
        "notes": summary.notes,
        "positions": positions,
    }


def _portfolio_page(request: Request, *, error: str | None = None, status: int = 200):
    user = request.state.sams_user
    web = request.app.state.web
    ctx = web.contexts.current()
    accounts = [
        _account_view(ctx, user.username, account)
        for account in ctx.sim_accounts.list_for(user.id)
    ]
    theses = ctx.theses.list(user.username)
    return web.render(
        request, "portfolio.html", status_code=status, title="持仓",
        accounts=accounts, theses=theses, error=error,
        form={"account_id": "", "market": "A", "code": "", "side": "buy",
              "qty": "", "price": "", "fee": "", "thesis_id": ""},
    )


@router.get("/portfolio")
def portfolio_page(request: Request):
    require_user(request)
    return _portfolio_page(request)


@router.post("/ledger/fill")
def ledger_fill(
    request: Request,
    _csrf: None = Depends(csrf_protect),
    account_id: str = Form(""),
    market: str = Form(""),
    code: str = Form(""),
    side: str = Form("buy"),
    qty: str = Form(""),
    price: str = Form(""),
    fee: str = Form("0"),
    thesis_id: str = Form(""),
):
    user = require_user(request)
    web = request.app.state.web
    ctx = web.contexts.current()
    try:
        quantity = _number(qty, "数量")
        trade_price = _number(price, "价格")
        trade_fee = _number(fee, "费用") if fee.strip() else 0.0
    except ValidationError as exc:
        return _portfolio_page(request, status=400, error=str(exc))
    try:
        kwargs = dict(
            qty=quantity, price=trade_price, fee=trade_fee,
            thesis_id=thesis_id.strip() or None,
        )
        if side == "buy":
            ctx.ledger.buy(
                account_id, user.username, market.strip().upper(), code.strip(), **kwargs
            )
        elif side == "sell":
            ctx.ledger.sell(
                account_id, user.username, market.strip().upper(), code.strip(), **kwargs
            )
        else:
            raise ValidationError("方向只能是 buy 或 sell")
    except SamsError as exc:
        return _portfolio_page(request, status=400, error=str(exc))
    return RedirectResponse("/portfolio", status_code=303)
```

```python
"""论点：列表、新建、详情与推进。推进的合法档位只有 `next_steps` 一处知道。"""

from __future__ import annotations

from fastapi import APIRouter, Depends, Form, HTTPException, Request
from fastapi.responses import RedirectResponse

from sams.core.errors import SamsError, ValidationError
from sams.storage.errors import StorageError
from sams.web.auth import csrf_protect, require_user

router = APIRouter()


def _require_own(ctx, user, thesis_id: str):
    """不属于/不存在都 404：`require` 的 StorageError 不往上冒给浏览器。"""
    try:
        return ctx.theses_store.require(user.id, thesis_id)
    except StorageError:
        raise HTTPException(status_code=404) from None


def _list_page(request: Request, *, error: str | None = None, status: int = 200):
    user = request.state.sams_user
    web = request.app.state.web
    ctx = web.contexts.current()
    return web.render(
        request, "theses.html", status_code=status, title="论点",
        rows=ctx.theses.list(user.username), error=error,
    )


@router.get("/theses")
def theses_page(request: Request):
    require_user(request)
    return _list_page(request)


@router.post("/theses/create")
def theses_create(
    request: Request,
    _csrf: None = Depends(csrf_protect),
    market: str = Form(""),
    code: str = Form(""),
    style: str = Form("value"),
    note: str = Form(""),
):
    user = require_user(request)
    ctx = request.app.state.web.contexts.current()
    try:
        report = ctx.theses.add(
            user.username, market.strip().upper(), code.strip(),
            style=style.strip(), note=note.strip() or None,
        )
    except SamsError as exc:
        return _list_page(request, status=400, error=str(exc))
    return RedirectResponse(f"/theses/{report.thesis.id}", status_code=303)


@router.get("/theses/{thesis_id}")
def thesis_page(request: Request, thesis_id: str):
    user = require_user(request)
    web = request.app.state.web
    ctx = web.contexts.current()
    thesis = _require_own(ctx, user, thesis_id)
    events = ctx.theses_store.events(user.id, thesis_id)
    return web.render(
        request, "thesis.html", title=f"论点 {thesis_id}",
        thesis=thesis, events=events, steps=ctx.theses.next_steps(thesis.status),
        error=None,
    )


@router.post("/theses/{thesis_id}/advance")
def thesis_advance(
    request: Request,
    thesis_id: str,
    _csrf: None = Depends(csrf_protect),
    to: str = Form(""),
    note: str = Form(""),
):
    user = require_user(request)
    web = request.app.state.web
    ctx = web.contexts.current()
    thesis = _require_own(ctx, user, thesis_id)
    try:
        if to not in ctx.theses.next_steps(thesis.status):
            raise ValidationError(
                f"不能从 {thesis.status} 推进到 {to or '（空）'}："
                f"合法的下一步是 {'/'.join(ctx.theses.next_steps(thesis.status)) or '（没有）'}"
            )
        ctx.theses.advance(user.username, thesis_id, to, note=note.strip() or None)
    except SamsError as exc:
        events = ctx.theses_store.events(user.id, thesis_id)
        return web.render(
            request, "thesis.html", status_code=400, title=f"论点 {thesis_id}",
            thesis=thesis, events=events, steps=ctx.theses.next_steps(thesis.status),
            error=str(exc),
        )
    return RedirectResponse(f"/theses/{thesis_id}", status_code=303)
```

```python
"""表单读取与模板参数解析。

Starlette 缓存 `request.form()`：csrf 依赖先解析一遍，端点里的 `form_data`
拿的是同一份，不会解析两次。表单字段名与 `Template.params` 对齐：
`param_<name>`、`filter_<key>`、`channel_ids`（多选）。
"""

from __future__ import annotations

from fastapi import Request

from sams.core.errors import ValidationError
from sams.core.rules import Template


async def form_data(request: Request) -> dict[str, list[str]]:
    form = await request.form()
    data: dict[str, list[str]] = {}
    for key, value in form.multi_items():
        data.setdefault(key, []).append(value if isinstance(value, str) else "")
    return data


def one(data: dict[str, list[str]], key: str, default: str = "") -> str:
    values = data.get(key)
    return values[0] if values else default


def many(data: dict[str, list[str]], key: str) -> list[str]:
    return [value for value in data.get(key, []) if value]


def checked(data: dict[str, list[str]], key: str) -> bool:
    return key in data and data[key][0] not in ("", "0", "false")


def param_values(
    template: Template, data: dict[str, list[str]], *, include_defaults: bool
) -> dict[str, object]:
    """把 `param_*` 字段按模板的声明读成参数。

    `include_defaults=True`（新建）：没填的非必填参数落默认值；
    `False`（编辑）：没填就是「这一项不动」，交给 service 保留旧值。
    """
    params: dict[str, object] = {}
    for spec in template.params:
        raw = one(data, f"param_{spec.name}").strip()
        if not raw:
            if include_defaults and spec.default is not None:
                params[spec.name] = spec.default
            elif spec.required:
                raise ValidationError(f"缺少必填参数：{spec.label}（{spec.name}）")
            continue
        if spec.shape == "number":
            try:
                number = float(raw)
            except ValueError as exc:
                raise ValidationError(f"{spec.label} 要是数字，收到 {raw!r}") from exc
            if spec.minimum is not None and number < spec.minimum:
                raise ValidationError(f"{spec.label} 不能小于 {spec.minimum:g}")
            if spec.maximum is not None and number > spec.maximum:
                raise ValidationError(f"{spec.label} 不能大于 {spec.maximum:g}")
            params[spec.name] = number
        else:
            if spec.choices and raw not in spec.choices:
                raise ValidationError(
                    f"{spec.label} 只能是 {'/'.join(spec.choices)} 之一，收到 {raw!r}"
                )
            params[spec.name] = raw
    return params


def filter_values(data: dict[str, list[str]]) -> dict[str, bool]:
    """四个开关：勾了是 True。没勾的键不进字典，与「未设置」同义。"""
    from sams.core.rules import FILTER_KEYS

    return {key: True for key in FILTER_KEYS if checked(data, f"filter_{key}")}
```

```python
PAGE_SIZE = 50


def _day_bound(value: str, tz_name: str, *, end: bool = False) -> datetime | None:
    """「某一天」按 display_tz 切：since 取当天零点；until 是排他上界，取**次日**零点。"""
    if not value:
        return None
    try:
        day = datetime.strptime(value, "%Y-%m-%d").date()
    except ValueError as exc:
        raise HTTPException(
            status_code=400, detail=f"日期格式应为 YYYY-MM-DD：{value!r}"
        ) from exc
    if end:
        day += timedelta(days=1)
    return datetime(day.year, day.month, day.day, tzinfo=ZoneInfo(tz_name))


def _require_own(ctx, alert_id: str, user_id: str) -> Alert:
    alert = ctx.alerts_store.get(alert_id, user_id=user_id)
    if alert is None:
        raise HTTPException(status_code=404, detail="没有这条告警")
    return alert


def _detail_response(
    request: Request, *, status_code: int = 200, error: str = "", dispatch=None
):
    user = request.state.sams_user
    ctx = request.app.state.contexts.current()
    alert_id = request.path_params["alert_id"]
    alert = ctx.alerts_store.get(alert_id, user_id=user.id)
    if alert is None:
        raise HTTPException(status_code=404, detail="没有这条告警")
    return request.app.state.web.render(
        request,
        "alert.html",
        status_code=status_code,
        alert=alert,
        deliveries=ctx.alerts_store.deliveries(alert_id),
        dispatch=dispatch,
        error=error,
    )


@router.get("/alerts")
def inbox(request: Request):
    user = require_user(request)
    ctx = request.app.state.contexts.current()
    q = request.query_params
    status = q.get("status") or "new"
    if status not in ("new", "handled", "dismissed", "all"):
        raise HTTPException(status_code=400, detail=f"未知状态 {status!r}")
    try:
        page = max(1, int(q.get("page") or 1))
    except ValueError:
        page = 1
    tz = ctx.settings.server.display_tz
    rows = ctx.alerts_store.list_for(
        user.id,
        status=None if status == "all" else status,
        severity=q.get("severity") or None,
        market=q.get("market") or None,
        since=_day_bound(q.get("since") or "", tz),
        until=_day_bound(q.get("until") or "", tz, end=True),
        limit=PAGE_SIZE + 1,
        offset=(page - 1) * PAGE_SIZE,
    )
    has_more = len(rows) > PAGE_SIZE
    rows = rows[:PAGE_SIZE]
    groups: dict[tuple[str, str], list[Alert]] = {}
    for alert in rows:
        groups.setdefault((alert.market, alert.code), []).append(alert)

    def page_url(target: int) -> str:
        pairs = [f"page={target}"]
        for key in ("status", "severity", "market", "since", "until"):
            value = q.get(key)
            if value:
                pairs.append(f"{key}={value}")
        return "/alerts?" + "&".join(pairs)

    return request.app.state.web.render(
        request,
        "alerts.html",
        status=status,
        groups=groups,
        page=page,
        has_more=has_more,
        prev_url=page_url(page - 1) if page > 1 else "",
        next_url=page_url(page + 1) if has_more else "",
        market=q.get("market") or "",
        severity=q.get("severity") or "",
        since=q.get("since") or "",
        until=q.get("until") or "",
    )


@router.get("/alerts/{alert_id}")
def detail(request: Request):
    require_user(request)
    return _detail_response(request)


@router.post("/alerts/{alert_id}/acted")
def acted(
    request: Request,
    _csrf: None = Depends(csrf_protect),
    data: dict[str, list[str]] = Depends(form_data),
):
    user = require_user(request)
    ctx = request.app.state.contexts.current()
    alert = _require_own(ctx, request.path_params["alert_id"], user.id)
    ctx.alerts_store.set_status(
        alert.id, HANDLED, user_id=user.id, note=(one(data, "note") or "").strip()
    )
    audit(request, action="alert.acted", target=alert.id, ok=True, detail="handled")
    return RedirectResponse(f"/alerts/{alert.id}", status_code=303)


@router.post("/alerts/{alert_id}/dismiss")
def dismiss(
    request: Request,
    _csrf: None = Depends(csrf_protect),
    data: dict[str, list[str]] = Depends(form_data),
):
    user = require_user(request)
    ctx = request.app.state.contexts.current()
    alert = _require_own(ctx, request.path_params["alert_id"], user.id)
    reason = (one(data, "reason") or "").strip()
    if not reason:
        return _detail_response(
            request, status_code=400, error="驳回必须写原因：那一列是「忽略率」唯一的输入"
        )
    ctx.alerts_store.set_status(alert.id, DISMISSED, user_id=user.id, note=reason)
    audit(request, action="alert.dismiss", target=alert.id, ok=True)
    return RedirectResponse(f"/alerts/{alert.id}", status_code=303)


@router.post("/alerts/{alert_id}/resend")
def resend(
    request: Request,
    _csrf: None = Depends(csrf_protect),
    data: dict[str, list[str]] = Depends(form_data),
):
    user = require_user(request)
    ctx = request.app.state.contexts.current()
    alert = _require_own(ctx, request.path_params["alert_id"], user.id)
    dispatch = ctx.notify.deliver_existing(alert.id)
    audit(
        request,
        action="alert.resend",
        target=alert.id,
        ok=True,
        detail=f"any_sent={dispatch.any_sent} deliveries={len(dispatch.deliveries)}",
    )
    return _detail_response(request, dispatch=dispatch)
```

```python
    def update_config(self, channel_id: str, updates: Mapping[str, str]) -> Channel:
        """合并写配置：给出的非空值覆盖旧值，空串=保持不变。

        编辑表单不回显明文（掩码视图才上屏幕，§4.2 的唯一明文出口仍是
        `credentials`），所以「没填的键」只能意味着保持不变。要删掉一个键，
        删掉通道重建——配置里没有哪个键值得为它设计一个删除协议。
        """
        node = self._require(channel_id)
        cleaned = {
            key.strip(): value
            for key, value in updates.items()
            if key.strip() and value != ""
        }
        merged = self.credentials(channel_id)
        merged.update(cleaned)
        self.db.execute(
            "UPDATE channels SET config_encrypted = ? WHERE id = ?",
            [self.cipher.encrypt(merged), channel_id],
        )
        moved = self.show(channel_id)
        assert moved is not None  # 同一条连接刚写完
        return moved

    def list_all(self) -> list[Channel]:
        """全部用户的通道（admin 只读页）。掩码视图照常，admin 不看明文。"""
        rows = self.db.query(f"SELECT {_COLUMNS} FROM channels ORDER BY user_id, name")
        return [self._to_channel(row) for row in rows.to_dicts()]
```

```python
def _own_channel(ctx, channel_id: str, user_id: str):

def _parse_config(text: str) -> dict[str, str]:
    """多行 `key=value` → dict。空行跳过；缺等号或缺键名整行拒。"""
    config: dict[str, str] = {}
    for line in (text or "").splitlines():
        line = line.strip()
        if not line:
            continue
        key, sep, value = line.partition("=")
        if not sep or not key.strip():
            raise ValidationError("配置一行一个 key=value：这一行缺了等号或键名")
        config[key.strip()] = value.strip()
    return config
```

`run_maintenance` 之后加两个方法：

```python
    def request_prune(self) -> bool:
        """管理员手动触发的每日清理；已经在跑时返回 False：不排队、不开第二把枪。

        DuckDB 单写：清理线程与调度线程抢库锁会互相等，把「正在跑」挡在入口
        比让两条线程在锁上排队好。
        """
        with self._prune_lock:
            if self._prune_running:
                return False
            self._prune_running = True
        threading.Thread(target=self._prune_once, name="sams-prune-manual", daemon=True).start()
        return True

    def _prune_once(self) -> None:
        try:
            report = self.run_maintenance()
            self.last_prune = f"{len(report.steps)} 步、合计 {report.total} 行"
            self.last_prune_error = None
        except Exception as exc:  # 失败要留在页面上，不是炸掉后台线程
            self.last_prune_error = str(exc)
        finally:
            self.last_prune_at = self.clock.now_utc()
            with self._prune_lock:
                self._prune_running = False
```
