```python
"""`Database.transaction()`：仓库层的多表写必须有一个原子出口。"""

from __future__ import annotations

import pytest

from sams.storage.db import Database


@pytest.fixture
def widget(db: Database) -> Database:
    db.execute("CREATE TABLE widget (id INTEGER PRIMARY KEY)")
    return db


def test_a_committed_block_leaves_every_row(widget: Database) -> None:
    with widget.transaction():
        widget.execute("INSERT INTO widget VALUES (1)")
        widget.execute("INSERT INTO widget VALUES (2)")
    assert widget.row_count("widget") == 2


def test_an_exception_rolls_back_the_whole_block(widget: Database) -> None:
    """回滚必须是整段：留下第 1 行就等于留下一笔已经扣了现金却没建的仓。"""
    with pytest.raises(ValueError), widget.transaction():
        widget.execute("INSERT INTO widget VALUES (1)")
        raise ValueError("后半段炸了")
    assert widget.row_count("widget") == 0


def test_nested_blocks_join_the_outer_one(widget: Database) -> None:
    """内层不另开事务，也不自己提交。

    `record_fill` 包住一整笔成交，顺手动现金的 `adjust_cash` 自己也包一层。如果嵌套
    报错，这两个方法就只能有一个真的开事务，而「谁负责原子性」从代码掉进口头约定；
    DuckDB 对嵌套 BEGIN 的原始报错读起来又像驱动 bug。所以这里明确支持重入。
    """
    with widget.transaction():
        with widget.transaction():
            widget.execute("INSERT INTO widget VALUES (1)")
        assert widget.row_count("widget") == 1  # 外层还没提交，但同连接可见
    assert widget.row_count("widget") == 1


def test_an_inner_exception_rolls_back_the_outer_block(widget: Database) -> None:
    with pytest.raises(ValueError), widget.transaction():
        widget.execute("INSERT INTO widget VALUES (1)")
        with widget.transaction():
            raise ValueError("内层炸了")
    assert widget.row_count("widget") == 0


def test_the_transaction_flag_reflects_the_outermost_block(widget: Database) -> None:
    assert widget.in_transaction is False
    with widget.transaction():
        assert widget.in_transaction is True
        with widget.transaction():
            assert widget.in_transaction is True
        assert widget.in_transaction is True
    assert widget.in_transaction is False


def test_a_storage_error_inside_the_block_rolls_back_too(widget: Database) -> None:
    """主键冲突是仓库层最常见的失败，它必须和异常一样整段撤销。

    这里点名 `StorageError` 而不是 `Exception`：后者会把「根本没有 transaction()」
    也算成通过，一条本该红的用例就从此不说任何话了。
    """
    with pytest.raises(StorageError), widget.transaction():
        widget.execute("INSERT INTO widget VALUES (1)")
        widget.execute("INSERT INTO widget VALUES (1)")
    assert widget.row_count("widget") == 0
```

```python
from __future__ import annotations

from contextlib import contextmanager
from pathlib import Path
```

在 `execute` 之前插入：

```python
    @property
    def in_transaction(self) -> bool:
        """是否已经在一个事务块里（重入计数大于 0）。仓库层用它决定要不要自己开。"""
        return self._tx_depth > 0

    @contextmanager
    def transaction(self) -> Iterator[None]:
        """一个原子写单元：块内任何异常都整段回滚，包括调用方没接住的。

        账本的一笔卖出同时写 `sim_orders`、`sim_positions`、`sim_cash_balances`、
        `sim_cash_moves`：中途失败留下的「现金已扣、持仓未减」不是待修的脏数据，
        是整本账失去意义。重入时内层只是加入外层那个事务（DuckDB 不支持嵌套 BEGIN），
        所以任何一层抛异常，最外层统一回滚——「谁负责原子性」由第一个开块的人定，
        不由谁碰巧写了 BEGIN 定。迁移有自己的 BEGIN/COMMIT（`migrate.py`），
        那是 DDL 的引导路径，不需要也不该走这里。
        """
        if self._tx_depth == 0:
            self._run("BEGIN TRANSACTION", None)
        self._tx_depth += 1
        try:
            yield
        except Exception:
            if self._tx_depth == 1:
                self._run("ROLLBACK", None)
            raise
        else:
            if self._tx_depth == 1:
                self._run("COMMIT", None)
        finally:
            self._tx_depth -= 1
```

## Task T-2: `core/funds.py` —— 币种与两段拆分的唯一权威

**这是整个 M3-A 最重要的一块。** spec §5.4 的四条硬规则里，「录入与存储永远用标的币种」和「price_pnl + fx_pnl 两段拆分」都是数学，不是策略；数学放对地方，后面每个读用例（T-8/T-9）就只是调用，放错地方就会长出三份略微不一样的换算——而三份不一样的换算在账本里叫对不上账。

恒等式（已用代数验算，实现必须逐字对上）：

```
price_pnl = qty * (last_price − avg_cost)                       # 标的币种
fx_pnl    = qty * last_price * (rate_today − rate_base)          # 报告币种
price_pnl * rate_base + fx_pnl == qty * last_price * rate_today − qty * avg_cost * rate_base
```

右边正是「今日市值（报告币种）− 成本（报告币种）」。所以 `rate_base` 必须是**下单那天**的汇率（T-1 的 `fx_rate_ref`），`rate_today` 是今天；两段合起来不多不少等于总市值变动，既不会把换汇算成股价涨，也不会漏掉它。

```python
"""钱相关的纯数学（spec §5.4）。

放在 `core` 且不许 import 上层，是因为这段算术会被 T-8/T-9/T-12 和 M3-C 的规则引擎
反复用到：一旦有两份略微不同的换算，账本就从「可核对」变成「两份对不上的数」。
这里没有 IO：汇率、价格、均价都由调用方传进来，函数只负责算对。
"""

from __future__ import annotations

from dataclasses import dataclass

from sams.core.codes import canonical_market, canonical_code, currency_override
from sams.core.config import MarketCfg
from sams.core.errors import ValidationError

#: 超过这个天数的汇率算「陈旧」：4 = 周五汇率撑到周一不算陈旧，跨长假算。
FX_STALE_DAYS = 4

#: 报告币种里常见的三种符号；其余显示 `1234.50 XXX`，不猜符号。
_SYMBOLS = {"CNY": "¥", "USD": "$", "HKD": "HK$"}


def quote_currency(market: str, code: str, cfg: dict[str, MarketCfg]) -> str:
    """这个标的用什么币种记账（§5.4 硬规则 1：录入与存储永远用标的币种）。

    `currency_override` 先于市场默认值：B 股（`200*`/`900*`）以 USD/HKD 交易，
    HK 双柜台以 RMB 交易（D31）。反过来用市场币种会把同一只票的两个柜台
    记成两种成本。
    """
    key = canonical_market(market)
    try:
        currency = cfg[key].currency
    except KeyError:
        raise ValidationError(
            f"市场 {key} 未配置，给不出标的币种；请在配置的 [markets.{key}] 里补 currency"
        ) from None
    return currency_override(key, canonical_code(key, code)) or currency


def blended_cost(old_qty: float, old_avg: float, qty: float, price: float, fee: float) -> float:
    """加仓后的加权均价；手续费计入成本（§5.4「含费」口径）。"""
    if old_qty <= 0:
        return (qty * price + fee) / qty
    return (old_qty * old_avg + qty * price + fee) / (old_qty + qty)


def realized_pnl(qty: float, price: float, avg_cost: float, fee: float) -> float:
    """卖出实现的盈亏，标的币种，已扣手续费。"""
    return qty * (price - avg_cost) - fee


def weighted_rate(entries: list[tuple[float, float, bool]]) -> float | None:
    """按金额加权的汇率；`entries` 是 (该笔金额的标的币种数额, 其汇率, is_stale)。

    先剔陈旧：一个过期汇率混进多币种汇总，等于让周五的汇率决定周一的总市值。
    全空或全是陈旧返回 `None` 而不是 0——0 会把市值抹平成 0，而「算不出来」和
    「值是 0」是两件事。取数归调用方（`core` 不查库），这里只做算术。
    """
    usable = [(amount, rate) for amount, rate, stale in entries if not stale and amount > 0]
    if not usable:
        return None
    total_amount = sum(amount for amount, _ in usable)
    if total_amount <= 0:
        return None
    return sum(amount * rate for amount, rate in usable) / total_amount


@dataclass(frozen=True)
class PositionPnl:
    """一个持仓的两段盈亏。字段单位写在注释里，因为混读它们是本模块唯一的危险。"""

    qty: float
    avg_cost: float
    last_price: float
    price_pnl: float  # 标的币种
    fx_pnl: float  # 报告币种
    market_report: float  # 报告币种：qty*last_price*rate_today
    cost_report: float  # 报告币种：qty*avg_cost*rate_base
    rate_base: float | None
    rate_today: float | None
    rate_stale: bool

    @property
    def total_report(self) -> float:
        """报告币种里的「市值 − 成本」。恒等式：`price_pnl*rate_base + fx_pnl` 与它相等。"""
        return self.market_report - self.cost_report


def position_pnl(
    *,
    qty: float,
    avg_cost: float,
    last_price: float,
    rate_base: float | None,
    rate_today: float | None,
    rate_today_stale: bool = False,
) -> PositionPnl:
    """把持仓盈亏拆成价格段与换汇段（§5.4 line 434）。

    `rate_base` 是**下单那天**的汇率（`sim_orders.fx_rate_ref` 加权），`rate_today` 是今天。
    两者相同（同币种，或汇率没动）时 `fx_pnl` 为 0。传 `None` 表示同币种，
    此时不做换算，`market_report` 直接按 1:1 计入——这是报告币种=标的币种的正常路径，
    不是缺数降级。
    """
    price_pnl = qty * (last_price - avg_cost)
    if rate_base is None or rate_today is None:
        fx_pnl = 0.0
        market_report = qty * last_price
        cost_report = qty * avg_cost
        stale = False
    else:
        fx_pnl = qty * last_price * (rate_today - rate_base)
        market_report = qty * last_price * rate_today
        cost_report = qty * avg_cost * rate_base
        stale = rate_today_stale
    return PositionPnl(
        qty=qty,
        avg_cost=avg_cost,
        last_price=last_price,
        price_pnl=price_pnl,
        fx_pnl=fx_pnl,
        market_report=market_report,
        cost_report=cost_report,
        rate_base=rate_base,
        rate_today=rate_today,
        rate_stale=stale,
    )


def money_note(amount: float, currency: str) -> str:
    """带币种的金额文本（§5.4 硬规则 3：金额阈值必须带币种）。"""
    symbol = _SYMBOLS.get(currency)
    if symbol is None:
        return f"{amount:.2f} {currency}"
    return f"{symbol}{amount:.2f}"
```

```python
四张表的写入口收在这一个模块的两个类里：`SimAccountRepo`（账户/现金/流水）与

from __future__ import annotations

from datetime import UTC, datetime

import polars as pl

from sams.core.errors import StorageError, ValidationError
from sams.storage.db import Database

#: 成交/调现金的时刻：带时区才收，裸 datetime 一律拒。
#: TIMESTAMPTZ 列会按会话时区解释无时区的值，那等于让 A 股的一笔记成 UTC，差 8 小时。
def _moment(ts: datetime | None) -> datetime:
    if ts is None:
        return datetime.now(UTC)
    if ts.tzinfo is None:
        raise ValidationError("时间必须带时区（例如 2024-10-25T15:00+08:00）：裸时刻会被当成 UTC")
    return ts.astimezone(UTC)


@dataclass(frozen=True)
class Account:

    user_id: str
    name: str
    broker: str | None
    market_scope: tuple[str, ...]
    fee_model: str
    margin_enabled: bool
    report_currency: str


def _row_to_account(row: dict) -> Account:
    import json

    return Account(
        id=row["id"],
        user_id=row["user_id"],
        name=row["name"],
        broker=row["broker"],
        market_scope=tuple(json.loads(row["market_scope"] or "[]")),
        fee_model=row["fee_model"],
        margin_enabled=bool(row["margin_enabled"]),
        report_currency=row["report_currency"],

class SimAccountRepo:
    """账户、按币种的现金余额、以及每一分钱的来处。"""

    def __init__(self, db: Database) -> None:
        self.db = db

    def open(
        self,
        user_id: str,
        name: str,
        *,
        broker: str | None = None,
        market_scope: tuple[str, ...] = (),
        fee_model: str = "default",
        margin_enabled: bool = False,
        report_currency: str = "CNY",
    ) -> Account:
        import json

        if not name.strip():
            raise ValidationError("账户名不能为空")
        account_id = new_id()
        try:
            self.db.execute(
                "INSERT INTO sim_accounts (id, user_id, name, broker, market_scope, "
                "fee_model, margin_enabled, report_currency) VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
                [
                    account_id,
                    user_id,
                    name,
                    broker,
                    json.dumps(list(market_scope)),
                    fee_model,
                    margin_enabled,
                    report_currency,
                ],
            )
        except StorageError as exc:
            # 唯一键 (user_id, name) 冲突要读成「这个名字已经有了」，不是「SQL 失败」
            if "Constraint" in str(exc) or "unique" in str(exc).lower():
                raise StorageError(f"账户 {name!r} 已存在（每个用户的账户名唯一）") from exc
            raise
        account = self.get(user_id, account_id)
        assert account is not None  # 刚写完，读不到就是库出了问题
        return account

    def get(self, user_id: str, account_id: str) -> Account | None:
        rows = self.db.query(
            "SELECT * FROM sim_accounts WHERE id = ? AND user_id = ?", [account_id, user_id]
        )
        return None if rows.is_empty() else _row_to_account(rows.to_dicts()[0])

    def list_for(self, user_id: str) -> list[Account]:
        rows = self.db.query(
            "SELECT * FROM sim_accounts WHERE user_id = ? ORDER BY created_at", [user_id]
        )
        return [_row_to_account(r) for r in rows.to_dicts()]

    def cash_balances(self, account_id: str) -> dict[str, float]:
        rows = self.db.query(
            "SELECT currency, amount FROM sim_cash_balances WHERE account_id = ? "
            "ORDER BY currency",
            [account_id],
        )
        return {r["currency"]: float(r["amount"]) for r in rows.to_dicts()}

    def cash_history(self, account_id: str) -> pl.DataFrame:
        """审计行按时间正序：倒序的流水在核对「哪一步开始对不上」时更难读。"""
        return self.db.query(
            "SELECT currency, delta, reason, actor_id, ts FROM sim_cash_moves "
            "WHERE account_id = ? ORDER BY ts, rowid",
            [account_id],
        )

    def adjust_cash(
        self,
        account_id: str,
        currency: str,
        delta: float,
        *,
        reason: str,
        actor_id: str,
        ts: datetime | None = None,
    ) -> float:
        """动现金的唯一入口：余额与流水同事务，失败整段回滚。

        返回调整后的余额，因为调用方（成交记账）要拿它判断「这一步之后还有多少钱」。
        """
        if not reason or not reason.strip():
            raise ValidationError("调整现金必须给理由：审计行没有理由就不是审计")
        if delta == 0:
            raise ValidationError("delta 不能是 0：没动就不要留一条流水")
        if currency not in {"CNY", "USD", "HKD", "EUR", "JPY", "GBP", "SGD", "AUD", "CAD"}:
            raise ValidationError(f"不认识币种 {currency!r}：三字母代码，不做模糊匹配")
        account = self._require(account_id)
        moment = _moment(ts)
        current = float(
            self.db.scalar(
                "SELECT amount FROM sim_cash_balances WHERE account_id = ? AND currency = ?",
                [account_id, currency],
            )
            or 0.0
        )
        after = current + delta
        if after < 0 and not account.margin_enabled:
            raise StorageError(
                f"现金不足：{currency} 余额 {current:.2f}，本次要动 {delta:.2f}，"
                f"结果 {after:.2f}。无杠杆账户不透支（§5.4）；"
                f"要记融资请在开户时把 margin_enabled 打开，或换一笔够钱的金额"
            )
        with self.db.transaction():
            self.db.execute(
                "INSERT INTO sim_cash_moves (id, account_id, currency, delta, reason, actor_id, ts) "
                "VALUES (?, ?, ?, ?, ?, ?, ?)",
                [new_id(), account_id, currency, delta, reason, actor_id, moment],
            )
            self.db.execute(
                "INSERT OR REPLACE INTO sim_cash_balances (account_id, currency, amount, updated_at) "
                "VALUES (?, ?, ?, ?)",
                [account_id, currency, after, moment],
            )
        return after

    def _require(self, account_id: str) -> Account:
        """账户是否存在（不校验归属，那是读用例那一层带 user_id 的事）。"""
        rows = self.db.query(
            "SELECT * FROM sim_accounts WHERE id = ?", [account_id]
        )
        if rows.is_empty():
            raise StorageError(f"账户 {account_id} 不存在")
        return _row_to_account(rows.to_dicts()[0])
```

```python
BUY = "buy"
SELL = "sell"
SIDES = (BUY, SELL)

#: 一次成交动的四张表里，只有订单是「历史」；持仓和余额是「状态」。
#: 状态可以由历史重放出来，所以写状态必须在历史之后、同一个事务里。


@dataclass(frozen=True)
class Position:
    account_id: str
    market: str
    code: str
    currency: str
    qty: float
    avg_cost: float
    #: 第一次建仓那一刻（不是最后一次成交）：§9.3 的「拿了多久」全靠它
    open_ts: datetime | None
    last_price: float | None
    last_price_ts: datetime | None


@dataclass(frozen=True)
class Order:
    id: str
    account_id: str
    market: str
    code: str
    currency: str
    side: str
    qty: float
    price: float
    ts: datetime
    fee: float
    thesis_id: str | None
    fx_rate_ref: float | None


def _row_to_order(row: dict) -> Order:
    return Order(
        id=row["id"],
        account_id=row["account_id"],
        market=row["market"],
        code=row["code"],
        currency=row["currency"],
        side=row["side"],
        qty=float(row["qty"]),
        price=float(row["price"]),
        ts=row["ts"],
        fee=float(row["fee"] or 0.0),
        thesis_id=row["thesis_id"],
        fx_rate_ref=row.get("fx_rate_ref"),
    )


#: `isfinite` 从模块顶部 `from math import isfinite` 引进来，调用点直接
#: `if not isfinite(value): raise ...`——一个只回答「是/否」的助手会让三处调用
#: 各自拼一句报错文案，而那三句话最后会不一样。


class SimOrderRepo:
    """订单与持仓。`record_fill` 是唯一的成交入口。"""

    def __init__(self, db: Database, accounts: SimAccountRepo | None = None) -> None:
        self.db = db
        #: 现金与流水归 SimAccountRepo 管；成交要顺手动它，所以持有同一个 db 上的一个实例
        self.accounts = accounts or SimAccountRepo(db)

    def record_fill(
        self,
        *,
        account_id: str,
        user_id: str,
        market: str,
        code: str,
        side: str,
        qty: float,
        price: float,
        currency: str,
        fee: float = 0.0,
        thesis_id: str | None = None,
        fx_rate_ref: float | None = None,
        ts: datetime | None = None,
    ) -> Order:
        if side not in SIDES:
            raise ValidationError(f"side 只能是 {'/'.join(SIDES)}，收到 {side!r}")
        for label, value in (("qty", qty), ("price", price), ("fee", fee)):
            if not isfinite(value):
                raise ValidationError(f"{label} 必须是有限小数，收到 {value!r}")
        if qty <= 0:
            raise ValidationError("qty 必须大于 0：清仓请用 sell 而不是 buy 0")
        if fee < 0:
            raise ValidationError("手续费不能为负：负手续费等于把成本做低")
        market = canonical_market(market)
        code = canonical_code(market, code)
        moment = _moment(ts)
        existing = self.position(account_id, market, code, strict=False)
        held = existing.qty if existing else 0.0
        if side == SELL and qty > held:
            raise StorageError(f"超卖：{market} {code} 持有 {held:g}，本次要卖 {qty:g}")
        gross = qty * price
        cash_delta = -gross - fee if side == BUY else gross - fee
        order_id = new_id()
        with self.db.transaction():
            # `fx_rate_applied` 留 NULL：那一列按 spec §5.2 只在「这笔伴随换汇」时才有值，
            # 换汇本身要到 M5 的 fx_trades 才存在。把基准汇率抄进它，等于让它从此没有出处，
            # 而 0014 的注释刚说过这两列问的是两个问题。
            self.db.execute(
                "INSERT INTO sim_orders (id, account_id, market, code, currency, side, qty,"
                " price, ts, fee, fee_currency, thesis_id, fx_rate_ref)"
                " VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
                [
                    order_id, account_id, market, code, currency, side, qty, price,
                    moment, fee, currency, thesis_id, fx_rate_ref,
                ],
            )
            if side == BUY:
                new_qty = held + qty
                new_avg = funds.blended_cost(
                    held, existing.avg_cost if existing else 0.0, qty, price, fee
                )
            else:
                new_qty = held - qty
                # 清仓均价归 0：留着最后一笔成本会被读成「还持有一些」
                new_avg = existing.avg_cost if new_qty > 0 else 0.0
            self.db.execute(
                "INSERT OR REPLACE INTO sim_positions (account_id, market, code, currency, qty, "
                "avg_cost, open_ts, last_price, last_price_ts) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
                [
                    account_id, market, code, currency, new_qty, new_avg,
                    # 建仓日是第一次那一刻，不是最后一次：覆盖它等于抹掉「拿了多久」
                    existing.open_ts if existing else moment,
                    price,
                    moment,
                ],
            )
            self.accounts.adjust_cash(
                account_id,
                currency,
                cash_delta,
                reason=f"{'买入' if side == BUY else '卖出'} {code}",
                actor_id=user_id,
                ts=moment,
            )
        order = self.order(account_id, order_id)
        assert order is not None
        return order

    def order(self, account_id: str, order_id: str) -> Order | None:
        rows = self.db.query(
            "SELECT * FROM sim_orders WHERE id = ? AND account_id = ?", [order_id, account_id]
        )
        return None if rows.is_empty() else _row_to_order(rows.to_dicts()[0])

    def orders(self, account_id: str) -> list[Order]:
        rows = self.db.query(
            "SELECT * FROM sim_orders WHERE account_id = ? ORDER BY ts, id", [account_id]
        )
        return [_row_to_order(r) for r in rows.to_dicts()]

    def order_rows(self, account_id: str) -> int:
        return int(
            self.db.scalar("SELECT count(*) FROM sim_orders WHERE account_id = ?", [account_id])
            or 0
        )

    def orders_for_thesis(self, thesis_id: str) -> list[Order]:
        """U26：某个立论到底有没有真成交。跨账户查，因为 thesis 与账户都属于用户。"""
        rows = self.db.query(
            "SELECT * FROM sim_orders WHERE thesis_id = ? ORDER BY ts, id",
            [thesis_id],
        )
        return [_row_to_order(r) for r in rows.to_dicts()]

    def positions(self, account_id: str) -> list[Position]:
        rows = self.db.query(
            "SELECT * FROM sim_positions WHERE account_id = ? ORDER BY market, code", [account_id]
        )
        return [_row_to_position(r) for r in rows.to_dicts()]

    def position(
        self, account_id: str, market: str, code: str, *, strict: bool = True
    ) -> Position | None:
        rows = self.db.query(
            "SELECT * FROM sim_positions WHERE account_id = ? AND market = ? AND code = ?",
            [account_id, canonical_market(market), canonical_code(market, code)],
        )
        if rows.is_empty():
            if strict:
                raise StorageError(f"账户 {account_id} 没有 {market} {code} 的持仓")
            return None
        return _row_to_position(rows.to_dicts()[0])

    def realized_pnl(self, account_id: str, market: str, code: str) -> float:
        """已实现盈亏（标的币种，已扣费）：由订单历史算，不另存一列。

        存下来就得再维护「成本基础随加仓变化」这件事，而它已经在 `sim_orders` 里了。
        """
        rows = self.db.query(
            "SELECT side, qty, price, fee FROM sim_orders WHERE account_id = ? "
            "AND market = ? AND code = ? ORDER BY ts",
            [account_id, canonical_market(market), canonical_code(market, code)],
        )
        if rows.is_empty():
            return 0.0
        running = 0.0
        basis = 0.0
        total = 0.0
        for r in rows.to_dicts():
            qty, price, fee = float(r["qty"]), float(r["price"]), float(r["fee"] or 0.0)
            if r["side"] == BUY:
                basis += qty * price + fee
                running += qty
            else:
                avg = basis / running if running > 0 else 0.0
                total += funds.realized_pnl(qty=qty, price=price, avg_cost=avg, fee=fee)
                basis -= qty * avg
                running -= qty
        return total


def _row_to_position(row: dict) -> Position:
    return Position(
        account_id=row["account_id"],
        market=row["market"],
        code=row["code"],
        currency=row["currency"],
        qty=float(row["qty"]),
        avg_cost=float(row["avg_cost"]),
        open_ts=row["open_ts"],
        last_price=row["last_price"],
        last_price_ts=row["last_price_ts"],
    )
```

```python
"""`fx_rates` 的读写（spec §5.4 规则 2/3 的数据侧）。

存的时候只存一个方向：`(base, quote)` 表示 1 单位 base = rate 单位 quote。
读的时候**不取倒数**——方向写反的症状是「大约对」，那是最难查的错。
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import date
from math import isfinite

from sams.core.errors import ValidationError
from sams.core.funds import FX_STALE_DAYS
from sams.storage.db import Database


@dataclass(frozen=True)
class FxRate:
    day: date
    base: str
    quote: str
    rate: float
    source: str | None
    is_stale: bool
    #: `latest` 才填：距今几天。当天为 0。
    age_days: int = 0


class FxRepo:
    def __init__(self, db: Database) -> None:
        self.db = db

    def put(
        self,
        day: date,
        base: str,
        quote: str,
        rate: float,
        *,
        source: str | None = None,
        is_stale: bool = False,
    ) -> None:
        if not isfinite(rate) or rate <= 0:
            raise ValidationError(f"汇率必须是正有限小数，收到 {rate!r}")
        if base == quote:
            raise ValidationError(f"{base} 对 {quote} 的汇率不需要存：同币种恒为 1")
        self.db.execute(
            "INSERT OR REPLACE INTO fx_rates (date, base_ccy, quote_ccy, rate, source, is_stale) "
            "VALUES (?, ?, ?, ?, ?, ?)",
            [day, base, quote, rate, source, is_stale],
        )

    def get(self, day: date, base: str, quote: str) -> FxRate | None:
        if base == quote:
            return FxRate(day, base, quote, 1.0, "same", False, 0)
        rows = self.db.query(
            "SELECT * FROM fx_rates WHERE date = ? AND base_ccy = ? AND quote_ccy = ?",
            [day, base, quote],
        )
        return None if rows.is_empty() else _row(rows.to_dicts()[0], 0)

    def latest(
        self, day: date, base: str, quote: str, *, max_age_days: int | None = None
    ) -> FxRate | None:
        """`day` 或之前最近的一条；超过 `max_age_days` 就不给（宁可没有，不要过期）。

        默认 `FX_STALE_DAYS`：周末用周五的汇率是合法的（标的不涨也不跌），
        长假第七天还在用同一个数是假数。
        """
        if base == quote:
            return FxRate(day, base, quote, 1.0, "same", False, 0)
        limit = FX_STALE_DAYS if max_age_days is None else max_age_days
        rows = self.db.query(
            "SELECT * FROM fx_rates WHERE base_ccy = ? AND quote_ccy = ? AND date <= ? "
            "ORDER BY date DESC LIMIT 1",
            [base, quote, day],
        )
        if rows.is_empty():
            return None
        row = rows.to_dicts()[0]
        age = (day - row["date"]).days
        if age > limit:
            return None
        return _row(row, age)


def _row(row: dict, age_days: int) -> FxRate:
    return FxRate(
        day=row["date"],
        base=row["base_ccy"],
        quote=row["quote_ccy"],
        rate=float(row["rate"]),
        source=row["source"],
        #: 陈旧是算出来的，不是存进去的：存进去的那一刻它不陈旧，第二天就陈旧了
        is_stale=bool(row["is_stale"]) or age_days > 0,
        age_days=age_days,
    )
```

```python
"""账本的写用例（spec §5.4）。

币种由配置推定、由仓库校验；thesis 链接由服务层校验，因为「这个立论现在能不能记成交」
问的是状态机，而状态机已经在 `repos/theses.py` 的 `TRANSITIONS` 里（D81）。

不 publish 事件：M3-A 没有消费者（M3-C 的规则引擎才会订 `QuoteUpdated`），先挂一个
没人听的 `publish` 会让「这个事件有没有人吃」重新变成要靠读代码猜的事。
"""

from __future__ import annotations

from datetime import datetime

from sams.core import funds
from sams.core.codes import canonical_code, canonical_market
from sams.core.config import Settings
from sams.core.errors import ValidationError
from sams.storage.repos.ledger import Account, Order, Position, SimAccountRepo, SimOrderRepo
from sams.storage.repos.theses import OBSERVING, PROPOSED, ThesisRepo

#: 能挂成交的立论状态：还没建仓的不能挂——那会在时间线上留下一个从未发生的「我当时买了」。
#: 归档/失效的那两条能挂：「事后回头看，这笔是按当时的立论买的」是真实发生过的事。
_UNBORN = (OBSERVING, PROPOSED)


class LedgerService:
    def __init__(
        self,
        accounts: SimAccountRepo,
        orders: SimOrderRepo,
        theses: ThesisRepo,
        settings: Settings,
    ) -> None:
        self.accounts = accounts
        self.orders = orders
        self.theses = theses
        self.settings = settings

    def open_account(
        self,
        user_id: str,
        name: str,
        *,
        broker: str | None = None,
        market_scope: tuple[str, ...] = (),
        margin_enabled: bool = False,
    ) -> Account:
        return self.accounts.open(
            user_id,
            name,
            broker=broker,
            market_scope=market_scope,
            margin_enabled=margin_enabled,
            report_currency=self.settings.report_currency,
        )

    def deposit(
        self, account_id: str, currency: str, amount: float, *, reason: str, actor_id: str
    ) -> float:
        return self.accounts.adjust_cash(
            account_id, currency, abs(amount), reason=reason, actor_id=actor_id
        )

    def withdraw(
        self, account_id: str, currency: str, amount: float, *, reason: str, actor_id: str
    ) -> float:
        return self.accounts.adjust_cash(
            account_id, currency, -abs(amount), reason=reason, actor_id=actor_id
        )

    def buy(
        self,
        account_id: str,
        user_id: str,
        market: str,
        code: str,
        *,
        qty: float,
        price: float,
        fee: float = 0.0,
        thesis_id: str | None = None,
        fx_rate_ref: float | None = None,
        ts: datetime | None = None,
    ) -> Order:
        return self._fill(
            account_id, user_id, market, code, side="buy", qty=qty, price=price, fee=fee,
            thesis_id=thesis_id, fx_rate_ref=fx_rate_ref, ts=ts,
        )

    def sell(
        self,
        account_id: str,
        user_id: str,
        market: str,
        code: str,
        *,
        qty: float,
        price: float,
        fee: float = 0.0,
        thesis_id: str | None = None,
        fx_rate_ref: float | None = None,
        ts: datetime | None = None,
    ) -> Order:
        return self._fill(
            account_id, user_id, market, code, side="sell", qty=qty, price=price, fee=fee,
            thesis_id=thesis_id, fx_rate_ref=fx_rate_ref, ts=ts,
        )

    def _fill(
        self,
        account_id: str,
        user_id: str,
        market: str,
        code: str,
        *,
        side: str,
        qty: float,
        price: float,
        fee: float,
        thesis_id: str | None,
        fx_rate_ref: float | None,
        ts: datetime | None,
    ) -> Order:
        market = canonical_market(market)
        code = canonical_code(market, code)
        # 未配置的市场在这里就拒：它的币种推不出来（同 D84 的立论闸门），
        # 而且此时还没开事务，报错不会留下半笔账
        currency = funds.quote_currency(market, code, self.settings.markets)
        if thesis_id is not None:
            self._check_thesis(user_id, thesis_id)
        return self.orders.record_fill(
            account_id=account_id,
            user_id=user_id,
            market=market,
            code=code,
            side=side,
            qty=qty,
            price=price,
            currency=currency,
            fee=fee,
            thesis_id=thesis_id,
            fx_rate_ref=fx_rate_ref,
            ts=ts,
        )

    def _check_thesis(self, user_id: str, thesis_id: str) -> None:
        thesis = self.theses.get(user_id, thesis_id)
        if thesis is None:
            # 不存在与不属于同一个回答（`ThesisRepo.get` 按 user_id 查，拿不到内容）
            raise ValidationError(f"立论 {thesis_id} 不存在或不属于用户 {user_id}")
        if thesis.status in _UNBORN:
            raise ValidationError(
                f"立论 {thesis_id} 还在 {thesis.status} 阶段，不能记成交："
                f"先走到建仓再记，否则时间线上会出现一个从未发生的「我当时买了」"
            )

    def accounts_of(self, user_id: str) -> list[Account]:
        return self.accounts.list_for(user_id)

    def position(self, account_id: str, market: str, code: str) -> Position | None:
        """没有持仓就是没有，不是报错：屏上「空仓」与「这只票不存在」是两回事。"""
        return self.orders.position(account_id, market, code, strict=False)

    def cash(self, account_id: str) -> dict[str, float]:
        return self.accounts.cash_balances(account_id)
```

## Task T-8: `PortfolioService.positions` —— 现价取未复权收盘（D73）

`avg_cost` 是成交那天的真价格，所以现价必须取**未复权 close**；取 hfq 绝对值会把
「持仓盈亏」算成一个从来没存在过的价格差（D73 立这条正是因为这两条链在 M2 已经分开了）。
没有行情时现价是 `None`，不是「用成本价顶一下」——顶一下会把「今天没数」显示成「今天持平」。

汇率那一天取：**与现价同日或之前最近的一条**（`FxRepo.latest(price_day, ...)`）。
这样 `PortfolioService` 不需要墙钟，也顺手避开了 `core/clock.py` 开头那句警告——
用 UTC 今日去推「哪个交易日」在美股上恒错一天，而把汇率锚在**已经有行情的那一天**
根本不需要问今天。

**上线时定下的两处与草案不同：**

1. 构造函数不接 `Settings`：报告币种取的是 `sim_accounts.report_currency` 那一列（开户时
   由 `LedgerService` 从配置写入）。读账户自己的列而不是再问一次配置，将来允许一个用户
   开一个美元报告账户时这里不用再改，而两处各读一份迟早会分叉。
2. 换不出报告币种的那一行 `pnl=None`（草案是把 `position_pnl(rate_base=None, rate_today=None)`
   塞进去、靠 `convertible` 让调用方自己躲）。`position_pnl` 对 `None` 汇率走的是「同币种 1:1」
   那条正常路径，返回的 `market_report` 是**标的币种**的数额；把它放在一个叫 `total_report`
   的属性后面，等于留一个只有读源码才能发现的坑。现在的不变式是：**`pnl is None` ⇔ 这一行
   不贡献任何报告币种里的数**，`unavailable_note` 说原因。

```python
    last_price: float | None
    last_price_date: date | None
    #: None = 这一行给不出任何报告币种里的数（缺现价或缺汇率），原因在 `unavailable_note`
    pnl: funds.PositionPnl | None = None
    #: False = 不进汇总。它不是「数为 0」，而是「这一笔此刻无法换算」
    convertible: bool = False
    #: 给不出现价 / 换不出币时的一句话（显示与 `--json` 读同一句，别长两张嘴）
    unavailable_note: str | None = None

    @property
    def price_pnl(self) -> float | None:
        """标的币种里的价格盈亏；给不出可信拆分时是 None，不是 0。"""
        return None if self.pnl is None else self.pnl.price_pnl

    @property
    def fx_pnl(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.fx_pnl

    @property
    def total_report(self) -> float:
        """报告币种里的市值−成本；不可换算的那一行贡献 0，并在 `unavailable_note` 里点名。"""
        return 0.0 if self.pnl is None else self.pnl.total_report

    @property
    def market_value_report(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.market_report

    @property
    def cost_report(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.cost_report

    @property
    def rate_base(self) -> float | None:
        return None if self.pnl is None else self.pnl.rate_base

    @property
    def rate_today(self) -> float | None:
        return None if self.pnl is None else self.pnl.rate_today

    @property
    def rate_stale(self) -> bool:
        return False if self.pnl is None else self.pnl.rate_stale


class PortfolioService:
    def __init__(
        self,
        accounts: SimAccountRepo,
        orders: SimOrderRepo,
        bars: BarRepo,
        fx: FxRepo,
    ) -> None:
        self.accounts = accounts
        self.orders = orders
        self.bars = bars
        self.fx = fx

    def positions(self, user_id: str, account_id: str) -> list[PositionRow]:
        """当前持仓逐行给出（已清仓的不出现）；报告币种取的是**账户**那一列。

        读也带 user_id：与 `repos/theses.py` 同形，不提供「查全量再由调用方过滤」的接口
        ——后者把隔离交给每一个调用方，总有一个忘记。
        """
        account = self.accounts.get(user_id, account_id)
        if account is None:
            raise ValidationError(f"账户 {account_id} 不存在或不属于用户 {user_id}")
        rows: list[PositionRow] = []
        for pos in self.orders.positions(account_id):
            if pos.qty <= 0:
                continue
            rows.append(self._one(account_id, pos, account.report_currency))
        return rows

    def _one(self, account_id: str, pos: Position, report: str) -> PositionRow:
        base = PositionRow(
            market=pos.market,
            code=pos.code,
            currency=pos.currency,
            qty=pos.qty,
            avg_cost=pos.avg_cost,
            last_price=None,
            last_price_date=None,
        )
        price, price_day = self._last_price(pos.market, pos.code)
        if price is None:
            return replace(
                base,
                unavailable_note=(
                    f"{pos.market} {pos.code} 取不到现价（库里没有行情），这一行不计入盈亏"
                ),
            )
        rate_base, rate_today, stale, note = self._rates(account_id, pos, price_day, report)
        if note is not None:
            return replace(
                base,
                last_price=price,
                last_price_date=price_day,
                unavailable_note=note,
            )
        return replace(
            base,
            last_price=price,
            last_price_date=price_day,
            pnl=funds.position_pnl(
                qty=pos.qty,
                avg_cost=pos.avg_cost,
                last_price=price,
                rate_base=rate_base,
                rate_today=rate_today,
                rate_today_stale=stale,
            ),
            convertible=True,
        )

    def _last_price(self, market: str, code: str) -> tuple[float | None, date | None]:
        """未复权收盘（D73）：`BarRepo` 的读取一律带 `adjust = 'raw'`，后复权序列不参与。"""
        latest = self.bars.latest_date(market, code)
        if latest is None:
            return None, None
        frame = self.bars.load(market, [code], start=latest, end=latest)
        if frame.is_empty():
            return None, None
        value = frame["close"][-1]
        return (None if value is None else float(value)), latest

    def _rates(
        self, account_id: str, pos: Position, price_day: date, report: str
    ) -> tuple[float | None, float | None, bool, str | None]:
        """(买入基准汇率, 行情日汇率, 是否陈旧, 换算不出时的一句话)。

        汇率取**行情日或之前**最近的一条，不取「今天」：把换算锚在已经有行情的那一天，
        就不需要墙钟（`FxRepo.latest` 超过窗口直接返回 None，宁缺毋陈旧）。
        换不出时给 note 而不给数：把一笔 USD 的数额当成 CNY 加进总市值，是一个看起来
        完全正常的错数。
        """
        if pos.currency == report:
            return None, None, False, None
        base = self._average_base_rate(account_id, pos)
        found = self.fx.latest(price_day, pos.currency, report)
        if found is None:
            return base, None, True, (
                f"{pos.market} {pos.code} 以 {pos.currency} 计价，取不到 {price_day} 或之前"
                f"可用的 {pos.currency}→{report} 汇率：这一行不计入汇总（可运行 `sams sync fx`"
                f"或手工补一条）"
            )
        if base is None:
            return None, found.rate, found.is_stale, (
                f"{pos.market} {pos.code} 的买入记录里没有基准汇率，无法把盈亏拆成"
                f"价格段与换汇段：这一行不计入汇总"
            )
        return base, found.rate, found.is_stale, None

    def _average_base_rate(self, account_id: str, pos: Position) -> float | None:
        """这笔持仓的基准汇率 = 各笔买入的 `fx_rate_ref` 按**股数**加权（§5.4 line 434）。

        权重是股数不是金额：`avg_cost` 是每股成本，恒等式两边都乘以它才对得上。
        少一笔已知的基准就整段不拆——拆错的 `fx_pnl` 比不拆更难查。
        """
        entries: list[tuple[float, float, bool]] = []
        for order in self.orders.orders(account_id):
            if order.market != pos.market or order.code != pos.code or order.side != "buy":
                continue
            if order.fx_rate_ref is None:
                return None
            entries.append((order.qty, order.fx_rate_ref, False))
        return funds.weighted_rate(entries)
```

## Task T-9: `summary` —— 分币种小计、换汇单列、跨账户重复持仓

一个账户可能同时有 CNY/HKD/USD 三笔现金和两种币种的持仓。汇总有一种错法比「数字算错」
更隐蔽：把所有数额直接相加报成一个 CNY 数，看着完全正常。所以**先按币种分组给出本币小计，
再逐笔换算成报告币种**，`fx_pnl` 单独成行；换不出的那一笔不进合计但在 `notes` 里点名。

跨账户重复持仓（同一个用户在两个账户都买了 601088）单独提示：`sim_positions` 的主键是
`(account_id, market, code)`，库里不会冲突，但用户在两个地方看到同一只票两个成本时，
得有人告诉他那是两个账户而不是一只票坏了。

**上线时定下的三处与草案不同：**

1. 没有现价的持仓**不进** `position_value_by_currency`（草案在这里用 `qty * avg_cost` 顶一下）。
   顶一下等于把「今天没行情」显示成「今天市值等于成本」，而那一行本来已经在 `notes` 里
   说了取不到数——同一屏上两句互相矛盾的话。
2. `_as_of` 在「没有持仓」时问注入的 `Clock`，不写 `date.today()`：`core/clock.py` 开头那条
   规矩是墙钟只能有一处，重放要能钉住时间。所以构造函数多了 `clock: Clock = SYSTEM_CLOCK`。
3. `duplicate_holdings(user_id, account_id)` 不排除当前账户（草案排除它，然后要求
   `len(names) > 1`，于是那条测试永远拿不到命中）：列的是「本账户里那些在别的账户也有
   一份」的标的，把涉及的全部账户名都写出来。

```python
"""持仓与账户的读用例（spec §5.4）。

写不在这个类里（`LedgerService`）：读用例不该有机会改账。

现价取自 `BarRepo`（库里只有不复权那一套，`adjust = 'raw'` 写死在查询里），而不是走
M2 那条后复权链：`avg_cost` 是成交那天的真价格，拿 hfq 绝对值去减它，算出来的是一段
从来没存在过的价差（D73）。换算锚在**行情的那一天**而不是「今天」，这样这个类不需要
墙钟，也避开了用 UTC 日期推交易日在美股上恒错一天（`core/clock.py` 开头那句警告）。
"""

from __future__ import annotations

from dataclasses import dataclass, replace
from datetime import date

from sams.core import funds
from sams.core.clock import SYSTEM_CLOCK, Clock
from sams.core.errors import ValidationError
from sams.storage.repos.bars import BarRepo
from sams.storage.repos.fx import FxRepo
from sams.storage.repos.ledger import (
    Account,
    Position,
    SimAccountRepo,
    SimOrderRepo,
)


@dataclass(frozen=True)
class PositionRow:
    market: str
    code: str
    currency: str
    qty: float
    avg_cost: float
    last_price: float | None
    last_price_date: date | None
    #: None = 这一行给不出任何报告币种里的数（缺现价或缺汇率），原因在 `unavailable_note`
    pnl: funds.PositionPnl | None = None
    #: False = 不进汇总。它不是「数为 0」，而是「这一笔此刻无法换算」
    convertible: bool = False
    #: 给不出现价 / 换不出币时的一句话（显示与 `--json` 读同一句，别长两张嘴）
    unavailable_note: str | None = None

    @property
    def price_pnl(self) -> float | None:
        """标的币种里的价格盈亏；给不出可信拆分时是 None，不是 0。"""
        return None if self.pnl is None else self.pnl.price_pnl

    @property
    def fx_pnl(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.fx_pnl

    @property
    def total_report(self) -> float:
        """报告币种里的市值−成本；不可换算的那一行贡献 0，并在 `unavailable_note` 里点名。"""
        return 0.0 if self.pnl is None else self.pnl.total_report

    @property
    def market_value_report(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.market_report

    @property
    def cost_report(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.cost_report

    @property
    def rate_base(self) -> float | None:
        return None if self.pnl is None else self.pnl.rate_base

    @property
    def rate_today(self) -> float | None:
        return None if self.pnl is None else self.pnl.rate_today

    @property
    def rate_stale(self) -> bool:
        return False if self.pnl is None else self.pnl.rate_stale


@dataclass(frozen=True)
class AccountSummary:
    """一个账户的汇总：本币小计分开、逐笔换算、换汇单列（§5.4 规则 2/3）。"""

    account_id: str
    name: str
    report_currency: str
    #: 各币种的现金余额，原样给出（本币数额，不做换算）
    cash_by_currency: dict[str, float]
    #: 各币种持仓的市值，取各行情日的收盘价；没有行情的那一行不进这里
    position_value_by_currency: dict[str, float]
    #: 报告币种里的总市值（现金 + 持仓）。换不出的那一笔不在里面，在 `notes` 里
    total_report: float
    price_pnl_report: float
    fx_pnl: float
    realized_report: float
    #: 换算锚在哪一天：持仓里最新的行情日，没有持仓才用钟（见 `_as_of`）
    as_of: date
    notes: tuple[str, ...]
    duplicate_holdings: tuple[str, ...]

    @property
    def total_pnl_report(self) -> float:
        return self.price_pnl_report + self.fx_pnl


class PortfolioService:
    def __init__(
        self,
        accounts: SimAccountRepo,
        orders: SimOrderRepo,
        bars: BarRepo,
        fx: FxRepo,
        *,
        clock: Clock = SYSTEM_CLOCK,
    ) -> None:
        self.accounts = accounts
        self.orders = orders
        self.bars = bars
        self.fx = fx
        self.clock = clock

    def positions(self, user_id: str, account_id: str) -> list[PositionRow]:
        """当前持仓逐行给出（已清仓的不出现）；报告币种取的是**账户**那一列。

        读也带 user_id：与 `repos/theses.py` 同形，不提供「查全量再由调用方过滤」的接口
        ——后者把隔离交给每一个调用方，总有一个忘记。
        """
        account = self._account(user_id, account_id)
        rows: list[PositionRow] = []
        for pos in self.orders.positions(account_id):
            if pos.qty <= 0:
                continue
            rows.append(self._one(account_id, pos, account.report_currency))
        return rows

    def _account(self, user_id: str, account_id: str) -> Account:
        account = self.accounts.get(user_id, account_id)
        if account is None:
            raise ValidationError(f"账户 {account_id} 不存在或不属于用户 {user_id}")
        return account

    def _one(self, account_id: str, pos: Position, report: str) -> PositionRow:
        base = PositionRow(
            market=pos.market,
            code=pos.code,
            currency=pos.currency,
            qty=pos.qty,
            avg_cost=pos.avg_cost,
            last_price=None,
            last_price_date=None,
        )
        price, price_day = self._last_price(pos.market, pos.code)
        if price is None:
            return replace(
                base,
                unavailable_note=(
                    f"{pos.market} {pos.code} 取不到现价（库里没有行情），这一行不计入盈亏"
                ),
            )
        rate_base, rate_today, stale, note = self._rates(account_id, pos, price_day, report)
        if note is not None:
            return replace(
                base,
                last_price=price,
                last_price_date=price_day,
                unavailable_note=note,
            )
        return replace(
            base,
            last_price=price,
            last_price_date=price_day,
            pnl=funds.position_pnl(
                qty=pos.qty,
                avg_cost=pos.avg_cost,
                last_price=price,
                rate_base=rate_base,
                rate_today=rate_today,
                rate_today_stale=stale,
            ),
            convertible=True,
        )

    def _last_price(self, market: str, code: str) -> tuple[float | None, date | None]:
        """未复权收盘（D73）：`BarRepo` 的读取一律带 `adjust = 'raw'`，后复权序列不参与。"""
        latest = self.bars.latest_date(market, code)
        if latest is None:
            return None, None
        frame = self.bars.load(market, [code], start=latest, end=latest)
        if frame.is_empty():
            return None, None
        value = frame["close"][-1]
        return (None if value is None else float(value)), latest

    def _rates(
        self, account_id: str, pos: Position, price_day: date, report: str
    ) -> tuple[float | None, float | None, bool, str | None]:
        """(买入基准汇率, 行情日汇率, 是否陈旧, 换算不出时的一句话)。

        汇率取**行情日或之前**最近的一条，不取「今天」：把换算锚在已经有行情的那一天，
        就不需要墙钟（`FxRepo.latest` 超过窗口直接返回 None，宁缺毋陈旧）。
        换不出时给 note 而不给数：把一笔 USD 的数额当成 CNY 加进总市值，是一个看起来
        完全正常的错数。
        """
        if pos.currency == report:
            return None, None, False, None
        base = self._average_base_rate(account_id, pos)
        found = self.fx.latest(price_day, pos.currency, report)
        if found is None:
            return base, None, True, (
                f"{pos.market} {pos.code} 以 {pos.currency} 计价，取不到 {price_day} 或之前"
                f"可用的 {pos.currency}→{report} 汇率：这一行不计入汇总（可运行 `sams sync fx`"
                f"或手工补一条）"
            )
        if base is None:
            return None, found.rate, found.is_stale, (
                f"{pos.market} {pos.code} 的买入记录里没有基准汇率，无法把盈亏拆成"
                f"价格段与换汇段：这一行不计入汇总"
            )
        return base, found.rate, found.is_stale, None

    def _average_base_rate(self, account_id: str, pos: Position) -> float | None:
        """这笔持仓的基准汇率 = 各笔买入的 `fx_rate_ref` 按**股数**加权（§5.4 line 434）。

        权重是股数不是金额：`avg_cost` 是每股成本，恒等式两边都乘以它才对得上。
        少一笔已知的基准就整段不拆——拆错的 `fx_pnl` 比不拆更难查。
        """
        entries: list[tuple[float, float, bool]] = []
        for order in self.orders.orders(account_id):
            if order.market != pos.market or order.code != pos.code or order.side != "buy":
                continue
            if order.fx_rate_ref is None:
                return None
            entries.append((order.qty, order.fx_rate_ref, False))
        return funds.weighted_rate(entries)

    def summary(self, user_id: str, account_id: str) -> AccountSummary:
        """账户汇总：先按币种给出本币小计，再逐笔换算（§5.4 规则 2/3）。

        把所有币种的数额直接相加报成一个数，是比「算错」更隐蔽的错：它看着完全正常。
        所以换不出的那一笔不进合计，但一定在 `notes` 里点名。
        """
        account = self._account(user_id, account_id)
        rows = self.positions(user_id, account_id)
        as_of = self._as_of(rows)
        notes: list[str] = []
        value_by_ccy: dict[str, float] = {}
        price_report = 0.0
        fx_total = 0.0
        positions_report = 0.0
        for row in rows:
            # 本币小计是「有几股、什么价」的事实，与能不能换算成报告币种无关
            if row.last_price is not None:
                value_by_ccy[row.currency] = (
                    value_by_ccy.get(row.currency, 0.0) + row.qty * row.last_price
                )
            if row.pnl is None:
                if row.unavailable_note is not None:
                    notes.append(row.unavailable_note)
                continue
            price_report += row.pnl.price_pnl * (row.pnl.rate_base or 1.0)
            fx_total += row.pnl.fx_pnl
            positions_report += row.pnl.market_report
        cash = self.accounts.cash_balances(account_id)
        cash_report = 0.0
        for ccy, amount in sorted(cash.items()):
            if amount == 0:
                continue
            cash_report += self._convert(
                account, ccy, amount, as_of, notes, f"现金 {amount:.2f} {ccy}"
            )
        return AccountSummary(
            account_id=account_id,
            name=account.name,
            report_currency=account.report_currency,
            cash_by_currency=cash,
            position_value_by_currency=value_by_ccy,
            total_report=positions_report + cash_report,
            price_pnl_report=price_report,
            fx_pnl=fx_total,
            realized_report=self._realized(account, as_of, notes),
            as_of=as_of,
            notes=tuple(notes),
            duplicate_holdings=self.duplicate_holdings(user_id, account_id),
        )

    def _as_of(self, rows: list[PositionRow]) -> date:
        """换算锚在哪一天：持仓行情日里最新的那个，没有持仓才问钟。

        持仓的换算根本不用它（各自锚在自己的行情日），用它的只有现金与已实现盈亏——
        那些没有「行情日」可言。这里读钟是为了不写 `date.today()`：重放要能钉住时间。
        """
        days = [row.last_price_date for row in rows if row.last_price_date is not None]
        return max(days) if days else self.clock.today()

    def _convert(
        self, account: Account, ccy: str, amount: float, as_of: date, notes: list[str], label: str
    ) -> float:
        """一笔本币数额换成报告币种；换不出就点名并贡献 0，不猜一个汇率。"""
        if ccy == account.report_currency:
            return amount
        found = self.fx.latest(as_of, ccy, account.report_currency)
        if found is None:
            notes.append(
                f"{label} 换不出 {account.report_currency}"
                f"（{as_of} 或之前没有可用的 {ccy}→{account.report_currency} 汇率），未计入合计"
            )
            return 0.0
        return amount * found.rate

    def _realized(self, account: Account, as_of: date, notes: list[str]) -> float:
        """已实现盈亏按币种汇总再换算：它算的是历史（`sim_orders` 重放），与当前持仓无关，
        所以已清仓的那几只也要算进来。"""
        by_ccy: dict[str, float] = {}
        for pos in self.orders.positions(account.id):
            amount = self.orders.realized_pnl(account.id, pos.market, pos.code)
            if amount == 0.0:
                continue
            by_ccy[pos.currency] = by_ccy.get(pos.currency, 0.0) + amount
        out = 0.0
        for ccy, amount in sorted(by_ccy.items()):
            out += self._convert(
                account, ccy, amount, as_of, notes, f"已实现 {amount:.2f} {ccy}"
            )
        return out

    def duplicate_holdings(self, user_id: str, account_id: str) -> tuple[str, ...]:
        """本账户里那些「在别的账户也有一份」的标的。

        库里不冲突（`sim_positions` 主键带 account_id），但读的人会以为一只票坏了：
        两个账户各有一笔 601088，两个成本，屏幕上同一只票两个价。
        """
        holders: dict[tuple[str, str], set[str]] = {}
        for account in self.accounts.list_for(user_id):
            for pos in self.orders.positions(account.id):
                if pos.qty > 0:
                    holders.setdefault((pos.market, pos.code), set()).add(account.name)
        mine = {
            (pos.market, pos.code)
            for pos in self.orders.positions(account_id)
            if pos.qty > 0
        }
        return tuple(
            f"{market} {code} 同时在 {'、'.join(sorted(names))} 里持有"
            for (market, code), names in sorted(holders.items())
            if len(names) > 1 and (market, code) in mine
        )
```

### `src/sams/services/actors.py`

```python
"""「谁在调用」这一步只写一遍：屏幕上的用户名 → 库里的那个 id。

归属这条不变式现在有三个读者（thesis、账本的写用例、账本的读用例）。抄三份的话，
先失效的那一处一定是某个入口漏掉「禁用」那半——症状不是报错，而是一个被停用的账号
还在往库里写。解析放在服务层也不放在命令层：后者等于指望每个新命令都记得查。
"""

from __future__ import annotations

from sams.core.errors import ValidationError
from sams.storage.repos.users import ACTIVE, User, UserRepo


def resolve_actor(users: UserRepo, username: str) -> User:
    user = users.get_by_username((username or "").strip())
    if user is None:
        raise ValidationError(
            f"没有用户 {username!r}：先 `sams user add {username}`"
            "（归属只认用户名，屏幕上的 `--user` 也是它）"
        )
    if user.status != ACTIVE:
        raise ValidationError(
            f"用户 {username} 已被禁用：启用之后才能读写，"
            "否则「禁用即停止一切动作」这条就不成立"
        )
    return user
```

### `src/sams/services/ledger.py`

```python
"""账本的写用例（spec §5.4）。

三件事只在这一层做：用户名解析成库里的 id（`services/actors.py`，与 thesis 认的是同一个 id，
否则 `--thesis` 那条链两边各认各的）、这笔账是不是你的（仓库层的 `_require` 不查归属）、
以及成交能不能挂在一条还没建仓的立论上（问的是状态机，它在 `repos/theses.py` 的
`TRANSITIONS` 里，D81）。币种在这里按配置推定、落库前由仓库校验，调用方不参与。

不 publish 事件：M3-A 没有消费者（M3-C 的规则引擎才会订 `QuoteUpdated`），先挂一个
没人听的 `publish` 会让「这个事件有没有人吃」重新变成要靠读代码猜的事。
"""

from __future__ import annotations

from datetime import datetime

from sams.core import funds
from sams.core.codes import canonical_code, canonical_market
from sams.core.config import Settings
from sams.core.errors import StorageError, ValidationError
from sams.services.actors import resolve_actor
from sams.storage.repos.ledger import Account, Order, Position, SimAccountRepo, SimOrderRepo
from sams.storage.repos.theses import OBSERVING, PROPOSED, ThesisRepo
from sams.storage.repos.users import User, UserRepo

#: 能挂成交的立论状态：还没建仓的不能挂——那会在时间线上留下一个从未发生的「我当时买了」。
#: 归档/失效的那两条能挂：「事后回头看，这笔是按当时的立论买的」是真实发生过的事。
_UNBORN = (OBSERVING, PROPOSED)


class LedgerService:
    def __init__(
        self,
        accounts: SimAccountRepo,
        orders: SimOrderRepo,
        theses: ThesisRepo,
        users: UserRepo,
        settings: Settings,
    ) -> None:
        self.accounts = accounts
        self.orders = orders
        self.theses = theses
        self.users = users
        self.settings = settings

    def open_account(
        self,
        username: str,
        name: str,
        *,
        broker: str | None = None,
        market_scope: tuple[str, ...] = (),
        margin_enabled: bool = False,
    ) -> Account:
        user = resolve_actor(self.users, username)
        return self.accounts.open(
            user.id,
            name,
            broker=broker,
            market_scope=market_scope,
            margin_enabled=margin_enabled,
            report_currency=self.settings.report_currency,
        )

    def deposit(
        self,
        account_id: str,
        currency: str,
        amount: float,
        *,
        username: str,
        reason: str,
        actor_id: str,
    ) -> float:
        self._require_account(resolve_actor(self.users, username), account_id)
        return self.accounts.adjust_cash(
            account_id, currency, abs(amount), reason=reason, actor_id=actor_id
        )

    def withdraw(
        self,
        account_id: str,
        currency: str,
        amount: float,
        *,
        username: str,
        reason: str,
        actor_id: str,
    ) -> float:
        self._require_account(resolve_actor(self.users, username), account_id)
        return self.accounts.adjust_cash(
            account_id, currency, -abs(amount), reason=reason, actor_id=actor_id
        )

    def buy(
        self,
        account_id: str,
        username: str,
        market: str,
        code: str,
        *,
        qty: float,
        price: float,
        fee: float = 0.0,
        thesis_id: str | None = None,
        fx_rate_ref: float | None = None,
        ts: datetime | None = None,
    ) -> Order:
        return self._fill(
            account_id, username, market, code, side="buy", qty=qty, price=price, fee=fee,
            thesis_id=thesis_id, fx_rate_ref=fx_rate_ref, ts=ts,
        )

    def sell(
        self,
        account_id: str,
        username: str,
        market: str,
        code: str,
        *,
        qty: float,
        price: float,
        fee: float = 0.0,
        thesis_id: str | None = None,
        fx_rate_ref: float | None = None,
        ts: datetime | None = None,
    ) -> Order:
        return self._fill(
            account_id, username, market, code, side="sell", qty=qty, price=price, fee=fee,
            thesis_id=thesis_id, fx_rate_ref=fx_rate_ref, ts=ts,
        )

    def _fill(
        self,
        account_id: str,
        username: str,
        market: str,
        code: str,
        *,
        side: str,
        qty: float,
        price: float,
        fee: float,
        thesis_id: str | None,
        fx_rate_ref: float | None,
        ts: datetime | None,
    ) -> Order:
        user = resolve_actor(self.users, username)
        self._require_account(user, account_id)
        market = canonical_market(market)
        code = canonical_code(market, code)
        # 未配置的市场在这里就拒：它的币种推不出来（同 D84 的立论闸门），
        # 而且此时还没开事务，报错不会留下半笔账
        currency = funds.quote_currency(market, code, self.settings.markets)
        if thesis_id is not None:
            self._check_thesis(user, thesis_id)
        return self.orders.record_fill(
            account_id=account_id,
            user_id=user.id,
            market=market,
            code=code,
            side=side,
            qty=qty,
            price=price,
            currency=currency,
            fee=fee,
            thesis_id=thesis_id,
            fx_rate_ref=fx_rate_ref,
            ts=ts,
        )

    def _require_account(self, user: User, account_id: str) -> Account:
        """这笔账是不是你的：仓库层的 `_require` 只查存在、不查归属。

        不是 ValidationError：`--account` 这个参数本身没写错，错的是库里没有（或不是你的）
        这么一笔账——那是业务拒绝（退出码 1，D77）。
        """
        account = self.accounts.get(user.id, account_id)
        if account is None:
            raise StorageError(f"账户 {account_id} 不存在或不属于用户 {user.username}")
        return account

    def _check_thesis(self, user: User, thesis_id: str) -> None:
        thesis = self.theses.get(user.id, thesis_id)
        if thesis is None:
            # 不存在与不属于同一个回答（`ThesisRepo.get` 按 user_id 查，拿不到内容）
            raise ValidationError(f"立论 {thesis_id} 不存在或不属于用户 {user.username}")
        if thesis.status in _UNBORN:
            raise ValidationError(
                f"立论 {thesis_id} 还在 {thesis.status} 阶段，不能记成交："
                f"先走到建仓再记，否则时间线上会出现一个从未发生的「我当时买了」"
            )

    def accounts_of(self, username: str) -> list[Account]:
        """用户名解析成 id 再问仓库：库里那一列存的是 id，屏幕上打的是用户名。"""
        return self.accounts.list_for(resolve_actor(self.users, username).id)

    def position(self, account_id: str, market: str, code: str) -> Position | None:
        """没有持仓就是没有，不是报错：屏上「空仓」与「这只票不存在」是两回事。"""
        return self.orders.position(account_id, market, code, strict=False)

    def cash(self, account_id: str) -> dict[str, float]:
        return self.accounts.cash_balances(account_id)
```

### `src/sams/services/portfolio.py`

```python
"""持仓与账户的读用例（spec §5.4）。

写不在这个类里（`LedgerService`）：读用例不该有机会改账。

现价取自 `BarRepo`（库里只有不复权那一套，`adjust = 'raw'` 写死在查询里），而不是走
M2 那条后复权链：`avg_cost` 是成交那天的真价格，拿 hfq 绝对值去减它，算出来的是一段
从来没存在过的价差（D73）。换算锚在**行情的那一天**而不是「今天」，这样这个类不需要
墙钟，也避开了用 UTC 日期推交易日在美股上恒错一天（`core/clock.py` 开头那句警告）。
"""

from __future__ import annotations

from dataclasses import dataclass, replace
from datetime import date

from sams.core import funds
from sams.core.clock import SYSTEM_CLOCK, Clock
from sams.core.errors import StorageError
from sams.services.actors import resolve_actor
from sams.storage.repos.bars import BarRepo
from sams.storage.repos.fx import FxRepo
from sams.storage.repos.ledger import (
    Account,
    Position,
    SimAccountRepo,
    SimOrderRepo,
)
from sams.storage.repos.users import User, UserRepo


@dataclass(frozen=True)
class PositionRow:
    market: str
    code: str
    currency: str
    qty: float
    avg_cost: float
    last_price: float | None
    last_price_date: date | None
    #: None = 这一行给不出任何报告币种里的数（缺现价或缺汇率），原因在 `unavailable_note`
    pnl: funds.PositionPnl | None = None
    #: False = 不进汇总。它不是「数为 0」，而是「这一笔此刻无法换算」
    convertible: bool = False
    #: 给不出现价 / 换不出币时的一句话（显示与 `--json` 读同一句，别长两张嘴）
    unavailable_note: str | None = None

    @property
    def price_pnl(self) -> float | None:
        """标的币种里的价格盈亏；给不出可信拆分时是 None，不是 0。"""
        return None if self.pnl is None else self.pnl.price_pnl

    @property
    def fx_pnl(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.fx_pnl

    @property
    def total_report(self) -> float:
        """报告币种里的市值−成本；不可换算的那一行贡献 0，并在 `unavailable_note` 里点名。"""
        return 0.0 if self.pnl is None else self.pnl.total_report

    @property
    def market_value_report(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.market_report

    @property
    def cost_report(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.cost_report

    @property
    def rate_base(self) -> float | None:
        return None if self.pnl is None else self.pnl.rate_base

    @property
    def rate_today(self) -> float | None:
        return None if self.pnl is None else self.pnl.rate_today

    @property
    def rate_stale(self) -> bool:
        return False if self.pnl is None else self.pnl.rate_stale


@dataclass(frozen=True)
class AccountSummary:
    """一个账户的汇总：本币小计分开、逐笔换算、换汇单列（§5.4 规则 2/3）。"""

    account_id: str
    name: str
    report_currency: str
    #: 各币种的现金余额，原样给出（本币数额，不做换算）
    cash_by_currency: dict[str, float]
    #: 各币种持仓的市值，取各行情日的收盘价；没有行情的那一行不进这里
    position_value_by_currency: dict[str, float]
    #: 报告币种里的总市值（现金 + 持仓）。换不出的那一笔不在里面，在 `notes` 里
    total_report: float
    price_pnl_report: float
    fx_pnl: float
    realized_report: float
    #: 换算锚在哪一天：持仓里最新的行情日，没有持仓才用钟（见 `_as_of`）
    as_of: date
    notes: tuple[str, ...]
    duplicate_holdings: tuple[str, ...]

    @property
    def total_pnl_report(self) -> float:
        return self.price_pnl_report + self.fx_pnl


class PortfolioService:
    def __init__(
        self,
        accounts: SimAccountRepo,
        orders: SimOrderRepo,
        bars: BarRepo,
        fx: FxRepo,
        users: UserRepo,
        *,
        clock: Clock = SYSTEM_CLOCK,
    ) -> None:
        self.accounts = accounts
        self.orders = orders
        self.bars = bars
        self.fx = fx
        self.users = users
        self.clock = clock

    def positions(self, username: str, account_id: str) -> list[PositionRow]:
        return self._positions(resolve_actor(self.users, username), account_id)

    def _positions(self, user: User, account_id: str) -> list[PositionRow]:
        """当前持仓逐行给出（已清仓的不出现）；报告币种取的是**账户**那一列。

        读也带归属：与 `repos/theses.py` 同形，不提供「查全量再由调用方过滤」的接口
        ——后者把隔离交给每一个调用方，总有一个忘记。
        """
        account = self._account(user, account_id)
        rows: list[PositionRow] = []
        for pos in self.orders.positions(account_id):
            if pos.qty <= 0:
                continue
            rows.append(self._one(account_id, pos, account.report_currency))
        return rows

    def _account(self, user: User, account_id: str) -> Account:
        """1 不是 2（D77）：`--account` 这个参数没写错，错的是库里没有（或不是你的）这笔账。

        写用例那边给的也是 1：同一个条件给两个退出码，脚本按哪个分支都是赌。
        """
        account = self.accounts.get(user.id, account_id)
        if account is None:
            raise StorageError(f"账户 {account_id} 不存在或不属于用户 {user.username}")
        return account

    def _one(self, account_id: str, pos: Position, report: str) -> PositionRow:
        base = PositionRow(
            market=pos.market,
            code=pos.code,
            currency=pos.currency,
            qty=pos.qty,
            avg_cost=pos.avg_cost,
            last_price=None,
            last_price_date=None,
        )
        price, price_day = self._last_price(pos.market, pos.code)
        if price is None:
            return replace(
                base,
                unavailable_note=(
                    f"{pos.market} {pos.code} 取不到现价（库里没有行情），这一行不计入盈亏"
                ),
            )
        rate_base, rate_today, stale, note = self._rates(account_id, pos, price_day, report)
        if note is not None:
            return replace(
                base,
                last_price=price,
                last_price_date=price_day,
                unavailable_note=note,
            )
        return replace(
            base,
            last_price=price,
            last_price_date=price_day,
            pnl=funds.position_pnl(
                qty=pos.qty,
                avg_cost=pos.avg_cost,
                last_price=price,
                rate_base=rate_base,
                rate_today=rate_today,
                rate_today_stale=stale,
            ),
            convertible=True,
        )

    def _last_price(self, market: str, code: str) -> tuple[float | None, date | None]:
        """未复权收盘（D73）：`BarRepo` 的读取一律带 `adjust = 'raw'`，后复权序列不参与。"""
        latest = self.bars.latest_date(market, code)
        if latest is None:
            return None, None
        frame = self.bars.load(market, [code], start=latest, end=latest)
        if frame.is_empty():
            return None, None
        value = frame["close"][-1]
        return (None if value is None else float(value)), latest

    def _rates(
        self, account_id: str, pos: Position, price_day: date, report: str
    ) -> tuple[float | None, float | None, bool, str | None]:
        """(买入基准汇率, 行情日汇率, 是否陈旧, 换算不出时的一句话)。

        汇率取**行情日或之前**最近的一条，不取「今天」：把换算锚在已经有行情的那一天，
        就不需要墙钟（`FxRepo.latest` 超过窗口直接返回 None，宁缺毋陈旧）。
        换不出时给 note 而不给数：把一笔 USD 的数额当成 CNY 加进总市值，是一个看起来
        完全正常的错数。
        """
        if pos.currency == report:
            return None, None, False, None
        base = self._average_base_rate(account_id, pos)
        found = self.fx.latest(price_day, pos.currency, report)
        if found is None:
            return base, None, True, (
                f"{pos.market} {pos.code} 以 {pos.currency} 计价，取不到 {price_day} 或之前"
                f"可用的 {pos.currency}→{report} 汇率：这一行不计入汇总（可运行 `sams sync fx`"
                f"或手工补一条）"
            )
        if base is None:
            return None, found.rate, found.is_stale, (
                f"{pos.market} {pos.code} 的买入记录里没有基准汇率，无法把盈亏拆成"
                f"价格段与换汇段：这一行不计入汇总"
            )
        return base, found.rate, found.is_stale, None

    def _average_base_rate(self, account_id: str, pos: Position) -> float | None:
        """这笔持仓的基准汇率 = 各笔买入的 `fx_rate_ref` 按**股数**加权（§5.4 line 434）。

        权重是股数不是金额：`avg_cost` 是每股成本，恒等式两边都乘以它才对得上。
        少一笔已知的基准就整段不拆——拆错的 `fx_pnl` 比不拆更难查。
        """
        entries: list[tuple[float, float, bool]] = []
        for order in self.orders.orders(account_id):
            if order.market != pos.market or order.code != pos.code or order.side != "buy":
                continue
            if order.fx_rate_ref is None:
                return None
            entries.append((order.qty, order.fx_rate_ref, False))
        return funds.weighted_rate(entries)

    def summary(self, username: str, account_id: str) -> AccountSummary:
        """账户汇总：先按币种给出本币小计，再逐笔换算（§5.4 规则 2/3）。

        把所有币种的数额直接相加报成一个数，是比「算错」更隐蔽的错：它看着完全正常。
        所以换不出的那一笔不进合计，但一定在 `notes` 里点名。
        """
        user = resolve_actor(self.users, username)
        account = self._account(user, account_id)
        rows = self._positions(user, account_id)
        as_of = self._as_of(rows)
        notes: list[str] = []
        value_by_ccy: dict[str, float] = {}
        price_report = 0.0
        fx_total = 0.0
        positions_report = 0.0
        for row in rows:
            # 本币小计是「有几股、什么价」的事实，与能不能换算成报告币种无关
            if row.last_price is not None:
                value_by_ccy[row.currency] = (
                    value_by_ccy.get(row.currency, 0.0) + row.qty * row.last_price
                )
            if row.pnl is None:
                if row.unavailable_note is not None:
                    notes.append(row.unavailable_note)
                continue
            price_report += row.pnl.price_pnl * (row.pnl.rate_base or 1.0)
            fx_total += row.pnl.fx_pnl
            positions_report += row.pnl.market_report
        cash = self.accounts.cash_balances(account_id)
        cash_report = 0.0
        for ccy, amount in sorted(cash.items()):
            if amount == 0:
                continue
            cash_report += self._convert(
                account, ccy, amount, as_of, notes, f"现金 {amount:.2f} {ccy}"
            )
        return AccountSummary(
            account_id=account_id,
            name=account.name,
            report_currency=account.report_currency,
            cash_by_currency=cash,
            position_value_by_currency=value_by_ccy,
            total_report=positions_report + cash_report,
            price_pnl_report=price_report,
            fx_pnl=fx_total,
            realized_report=self._realized(account, as_of, notes),
            as_of=as_of,
            notes=tuple(notes),
            duplicate_holdings=self._duplicate_holdings(user.id, account_id),
        )

    def _as_of(self, rows: list[PositionRow]) -> date:
        """换算锚在哪一天：持仓行情日里最新的那个，没有持仓才问钟。

        持仓的换算根本不用它（各自锚在自己的行情日），用它的只有现金与已实现盈亏——
        那些没有「行情日」可言。这里读钟是为了不写 `date.today()`：重放要能钉住时间。
        """
        days = [row.last_price_date for row in rows if row.last_price_date is not None]
        return max(days) if days else self.clock.today()

    def _convert(
        self, account: Account, ccy: str, amount: float, as_of: date, notes: list[str], label: str
    ) -> float:
        """一笔本币数额换成报告币种；换不出就点名并贡献 0，不猜一个汇率。"""
        if ccy == account.report_currency:
            return amount
        found = self.fx.latest(as_of, ccy, account.report_currency)
        if found is None:
            notes.append(
                f"{label} 换不出 {account.report_currency}"
                f"（{as_of} 或之前没有可用的 {ccy}→{account.report_currency} 汇率），未计入合计"
            )
            return 0.0
        return amount * found.rate

    def _realized(self, account: Account, as_of: date, notes: list[str]) -> float:
        """已实现盈亏按币种汇总再换算：它算的是历史（`sim_orders` 重放），与当前持仓无关，
        所以已清仓的那几只也要算进来。"""
        by_ccy: dict[str, float] = {}
        for pos in self.orders.positions(account.id):
            amount = self.orders.realized_pnl(account.id, pos.market, pos.code)
            if amount == 0.0:
                continue
            by_ccy[pos.currency] = by_ccy.get(pos.currency, 0.0) + amount
        out = 0.0
        for ccy, amount in sorted(by_ccy.items()):
            out += self._convert(
                account, ccy, amount, as_of, notes, f"已实现 {amount:.2f} {ccy}"
            )
        return out

    def duplicate_holdings(self, username: str, account_id: str) -> tuple[str, ...]:
        return self._duplicate_holdings(resolve_actor(self.users, username).id, account_id)

    def _duplicate_holdings(self, user_id: str, account_id: str) -> tuple[str, ...]:
        """本账户里那些「在别的账户也有一份」的标的。

        库里不冲突（`sim_positions` 主键带 account_id），但读的人会以为一只票坏了：
        两个账户各有一笔 601088，两个成本，屏幕上同一只票两个价。
        """
        holders: dict[tuple[str, str], set[str]] = {}
        for account in self.accounts.list_for(user_id):
            for pos in self.orders.positions(account.id):
                if pos.qty > 0:
                    holders.setdefault((pos.market, pos.code), set()).add(account.name)
        mine = {
            (pos.market, pos.code)
            for pos in self.orders.positions(account_id)
            if pos.qty > 0
        }
        return tuple(
            f"{market} {code} 同时在 {'、'.join(sorted(names))} 里持有"
            for (market, code), names in sorted(holders.items())
            if len(names) > 1 and (market, code) in mine
        )
```

### `src/sams/cli/common.py`

```python
"""命令模块共用的小工具：子命令分组、参数解析辅助。

放这里的东西必须真的被两个以上命令用到，否则写回各自命令模块。
"""

from __future__ import annotations

import argparse
import getpass
import json
from argparse import _SubParsersAction
from datetime import UTC, date

from sams.core.errors import ValidationError
from sams.core.fundamentals import RATIO_BOUNDS, out_of_caliber


def exit_for(exc: Exception) -> int:
    """1 与 2 的分界只在这里写一遍（D77）：认得出的错在参数上就是 2，其余是业务拒绝。

    脚本对两者的处置不同：2 是敲错了，重来多少次都一样；1 是这条记录还没到那一步，
    补数据或走完上一步就能重跑。把这条线抄在两个命令模块里，总有一处会先失效。
    """
    return 2 if isinstance(exc, ValidationError) else 1


def emit(payload: dict) -> None:
    print(json.dumps(payload, ensure_ascii=False, separators=(",", ":")))


def refused(args: argparse.Namespace, exc: Exception, code: int) -> int:
    """拒绝也要有 JSON（D57）：管道里读的人拿到的是一句原因加一个码，不是一个空 stdin。"""
    if getattr(args, "json", False):
        emit({"status": "refused", "reason": str(exc), "exit": code})
    else:
        print(f"{exc}")
    return code


def stamp(value) -> str:
    """时间戳统一到 UTC 再落屏，并把时区说出来。

    DuckDB 对 `TIMESTAMPTZ` 的回读不稳定：`theses.created_at` 回来是 UTC，
    `thesis_events.ts` 与 `sim_orders.ts` 回来是会话本地时区（Asia/Shanghai）。两者是同一个
    瞬间，但按原样 `strftime` 会在同一屏上印出「立论 15:00 / 时间线 23:00」——看着像事件比
    记录晚八小时，而这一屏的用意正是回看「什么时候走了哪一步」（D80）。
    到秒：两步落在同一分钟里时（脚本、M3 的触发器），分是分辨不出先后的。
    """
    if value is None:
        return "–"
    if value.tzinfo is not None:
        value = value.astimezone(UTC)
    return f"{value:%Y-%m-%d %H:%M:%S} UTC"


def group(subparsers: _SubParsersAction, name: str, help_text: str) -> _SubParsersAction:
    """建一层 `sams <name> <action>` 分组，返回 action 用的 subparsers。"""
    parser = subparsers.add_parser(name, help=help_text)
    return parser.add_subparsers(dest="action", required=True)


def ask_password(prompt: str = "密码") -> str:
    """交互取密码：给了 --password 就用它，否则问两次（无人值守时用 --password）。"""
    first = getpass.getpass(f"{prompt}: ")
    second = getpass.getpass(f"再输一次{prompt}: ")
    if first != second:
        raise ValueError(f"两次{prompt}不一致")
    return first


def iso_date(value: str) -> date:
    """解析 `YYYY-MM-DD`。

    包一层是因为 `date.fromisoformat` 抛的是裸 `ValueError`，而 CLI 只兜 `SamsError`：
    手滑打成一个 2024-7-1 会让整条命令吐一段栈，运维读到的是 Python 的行号，
    不是「这个日期该怎么写」。
    """
    try:
        return date.fromisoformat(value)
    except ValueError:
        raise ValidationError(
            f"日期 {value!r} 看不懂：要的是 YYYY-MM-DD，例如 2024-07-01"
        ) from None


def ratio_arg(name: str, value: str) -> float:
    """比率参数按**小数口径**解析（D58 + D66）：`--roe-estimate 15` 与 `--roe-estimate 5` 都拒收。

    这不是洁癖。PR 的 PB 那一支把 ROE 乘两遍（`PB ÷ ROE ÷ ROE ÷ 100`），
    把 15% 写成 15 会让那个数小两百倍——屏幕上它长得像「极度便宜」，而这是
    一条会把票排进 `PR ≤ 0.4` 前列的选股规则。仓库层对库里的数有同一道闸，
    命令行是这条口径的另一个入口，不能只挡一边：判据取 core 那一个 `out_of_caliber`，
    所以「库里存不下 1.0 以上的 ROE」与「命令行收不下 1.0 以上的预估」不会是两句话。
    """
    try:
        number = float(value)
    except ValueError:
        raise ValidationError(f"{name} 要的是一个数，收到 {value!r}") from None
    ceiling = RATIO_BOUNDS["roe"]
    if number <= 0 or out_of_caliber("roe", number):
        raise ValidationError(
            f"{name}={value} 不像一个小数口径的比率：要 0 < x < {ceiling:g}"
            "（15% 写 0.15，不是 15；而 1 以上在这里只能读成 100% 开外，同样不收）"
        )
    return number
```

```python
"""`sams ledger`：开户、动现金、记成交、看持仓与汇总（spec §5.4）。

# --- 成交时刻 -------------------------------------------------------------------------------


def _moment(args: argparse.Namespace, container, market: str) -> datetime:
    """成交时刻：`--on` 那天市场本地 15:00（收盘附近）。

    时区的唯一出处是 `MarketCfg.zone()`：CLI 硬编码 UTC 会让 A 股的一笔差 8 小时。
    `--on` 的必填在这里查而不是交给 argparse：拒的那一句也要能走 `--json` 那个出口（D57），
    而 argparse 的报错只有 stderr 和一个 2。默认「今天」则是另一回事——补录时会静默记错一天。
    """
    if args.on is None:
        raise ValidationError("--on 是必填的：记账要一个成交日（YYYY-MM-DD）")
    day = iso_date(args.on)
    zone = container.settings.market_cfg(canonical_market(market)).zone()
    return datetime(day.year, day.month, day.day, 15, 0, tzinfo=zone)

def run_account(container, args) -> int:
    try:
        if args.action == "list":
            accounts = container.ledger.accounts_of(args.user)
            payload = {
                "status": "ok",
                "accounts": [_account_payload(row) for row in accounts],
            }
        else:
            if not args.name:
                raise ValidationError("`account add` 要 --name：账户名是名下唯一认得出的东西")
            account = container.ledger.open_account(
                args.user, args.name, broker=args.broker, margin_enabled=args.margin
            )
            payload = {"status": "ok", "account": _account_payload(account)}
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit(payload)
        return 0
    if args.action == "list":
        _print_account_list(payload["accounts"], args.user)
    else:
        _print_account(payload["account"], args.user)
    return 0


def run_cash(container, args) -> int:
    try:
        if args.amount <= 0:
            raise ValidationError(
                f"--amount 要的是一个正数（方向由 --withdraw 决定），收到 {args.amount}"
            )
        move = container.ledger.withdraw if args.withdraw else container.ledger.deposit
        balance = move(
            args.account,
            args.currency,
            args.amount,
            username=args.user,
            reason=args.reason,
            actor_id=args.user,
        )
        payload = {
            "status": "ok",
            "account": args.account,
            "currency": args.currency,
            "balance": balance,
        }
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit(payload)
        return 0
    verb = "取出" if args.withdraw else "注资"
    print(
        f"{verb} {args.amount:.2f} {args.currency} → 余额 {balance:.2f} {args.currency}"
        f"（{args.reason}）"
    )
    return 0


def run_buy(container, args) -> int:
    return _fill(container, args, "buy")


def run_sell(container, args) -> int:
    return _fill(container, args, "sell")


def _fill(container, args, side: str) -> int:
    book = container.ledger
    record = book.buy if side == "buy" else book.sell
    try:
        order = record(
            args.account,
            args.user,
            args.market,
            args.code,
            qty=args.qty,
            price=args.price,
            fee=args.fee,
            thesis_id=args.thesis,
            fx_rate_ref=args.fx_rate,
            ts=_moment(args, container, args.market),
        )
        position = book.position(order.account_id, order.market, order.code)
        payload = {
            "status": "ok",
            "order": _order_payload(order),
            "position": _held_payload(position),
            "cash": book.cash(order.account_id),
        }
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit(payload)
        return 0
    _print_fill(payload, args.user, stamp(order.ts))
    return 0


def run_positions(container, args) -> int:
    try:
        rows = container.portfolio.positions(args.user, args.account)
        payload = {"status": "ok", "positions": [_position_payload(row) for row in rows]}
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit(payload)
        return 0
    _print_positions(rows)
    return 0


def run_summary(container, args) -> int:
    try:
        summary = container.portfolio.summary(args.user, args.account)
        payload = {"status": "ok", **_summary_payload(summary)}
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit(payload)
        return 0
    _print_summary(summary)
    return 0

        f"{order['price']:g} {order['currency']}{fee}  →  {when}"
    )
    held = payload["position"]
    if held is None:
        print("  这笔之后清仓（`positions` 里不再出现，已实现盈亏在 `summary` 里）")
    else:
        print(f"  持有 {held['qty']:g} 股，均价 {held['avg_cost']:g} {held['currency']}")
    cash = "、".join(f"{ccy} {amount:.2f}" for ccy, amount in sorted(payload["cash"].items()))
    print(f"  现金：{cash or '（没有）'}")
    print(f"  下一步：sams ledger summary --user {username} --account {order['account']}")


def _print_positions(rows: list[PositionRow]) -> None:
    if not rows:
        print("空仓：一笔成交都还没有（`sams ledger buy --help` 看怎么记）")
        return
    print(f"{'市场/代码':<14}{'数量':>9}{'成本':>10}{'现价':>10}{'盈亏':>11}{'换汇':>9}")
    for row in rows:
        label = f"{row.market}/{row.code}"
        if row.price_pnl is None:
            print(f"{label:<14}{row.qty:>9g}{'—':>10}{'—':>10}{'—':>11}{'—':>9}")
            # 那一行为什么没有数，与它是 0，是两句不同的话
            print(f"    {row.unavailable_note}")
            continue
        day = f"（{row.last_price_date} 收盘）" if row.last_price_date else ""
        print(
            f"{label:<14}{row.qty:>9g}{row.avg_cost:>10.4g}{row.last_price:>10.4g}"
            f"{row.price_pnl:>11.2f}{row.fx_pnl:>9.2f}  {row.currency}{day}"
        )


def _print_summary(summary: AccountSummary) -> None:
    report = summary.report_currency
    print(f"{summary.name}  报告币种 {report}  锚在 {summary.as_of}")
    for ccy, amount in sorted(summary.cash_by_currency.items()):
        _line(f"现金 {ccy}", amount)
    for ccy, amount in sorted(summary.position_value_by_currency.items()):
        _line(f"持仓 {ccy}", amount)
    _line("合计", summary.total_report, unit=report, rule=True)
    _line("价格盈亏", summary.price_pnl_report, unit=report)
    _line("换汇影响", summary.fx_pnl, unit=report)
    _line("已实现", summary.realized_report, unit=report)
    # notes 与跨账户点名逐行打在数字之后：它们不是日志，是「这个合计为什么不完整」的出处
    tail = list(summary.notes) + list(summary.duplicate_holdings)
    if tail:
        print("注意：")
        for note in tail:
            print(f"  · {note}")


def _line(label: str, amount: float, *, unit: str = "", rule: bool = False) -> None:
    pad = "─" * 24 if rule else ""
    suffix = f" {unit}" if unit else ""
    print(f"  {label:<{_LABEL}}{pad} {amount:>16,.2f}{suffix}")
```

```python
"""CLI 层：解析参数、组装依赖、把结果打成文本。

边界规则：命令模块只通过 Container 拿服务，不 import storage/datasource 的实现，
也不在这里写业务判断（「该不该跳过周末」这类分支属于 services）。
加一个命令 = 加一个本包下的模块，不改其它模块。

from __future__ import annotations

import argparse
import sys

from sams.bootstrap import build_container
from sams.cli import analysis, calendar, db, health, ledger, screen, source, sync, thesis, user
from sams.core.config import Settings
from sams.core.errors import SamsError

COMMAND_MODULES = (db, user, sync, calendar, health, source, analysis, screen, thesis, ledger)


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="sams", description="SAMS 运维入口")
    parser.add_argument(
        "--config",
        help="配置文件路径，默认取 SAMS_CONFIG 或 config/settings.toml",
    )
    subparsers = parser.add_subparsers(dest="command", required=True)
    for module in COMMAND_MODULES:
        module.add_to(subparsers)
    return parser


def run(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    try:
        container = build_container(Settings.load(args.config))
    except SamsError as exc:
        print(f"配置错误: {exc}", file=sys.stderr)
        return 1
    try:
        if args.command != "db":
            # 迁移幂等，因此非 db 命令顺手补齐表结构：新装环境直接 `sams user add`
            # 不该撞 CatalogException；`db migrate/status` 保留手工检视迁移的语义。
            container.migrate()
        return args.handler(container, args)
    except SamsError as exc:
        print(f"错误: {exc}", file=sys.stderr)
        return 1
    finally:
        container.close()


def main() -> None:  # pragma: no cover - 控制台入口
    sys.exit(run())
```

## Task T-11: `sams sync fx` 与手工降级 ✅

汇率有数才能拆 `fx_pnl`。这一节按交付形态回填（步骤全走完、判据全绿）。
计划与实现不一致的地方有十三处，留在这里说明为什么改：

1. **capability 的键是 `fx_rate`，不是计划里写的 `fx`**：那张表有一条不变式「键 = 适配器
   方法名」（`test_capabilities.py` 守着）。登记成 `fx` 会让判据与目录各认一套名字，
   `sams source` 那一屏上「谁能给汇率」就查不到了。
2. **`fx_rate(base, quote, start, end)` 一次问整段**：计划里写的是单日 `fx_rate(day, base,
   quote)` 再加 ±4 天窗口。按日循环会把一次限流风险变成三十次，而汇价是连续序列，
   逐日问的答复不会比一次问的多；Yahoo 的历史接口本来就给一段。代价是源的窗口常比
   我们问的宽，所以 `_call_fx` 末尾按 `[start, end]` 裁剪——多出来的日子落进库会让人以为
   `--end` 之前真的都同步过。
3. **`EmptyResponse` 在这里算一句答复，异常不算**（D43 的日线侧 + D36）：接口正常回空就是
   「这段日子没有报价」，可以写进 `confirmed_empty_by`；一家说没有、另一家超时时，那句
   「没有」还不能写成确认，照样抛。基本面那条链是反的（0 行不算答复），这里刻意不同侧。
4. **链耗尽时按 `RETRYABLE` 分类，不是按 `TransientError`**：判据从「全链都是抖动」改成
   「全链都值得再试一次」。冒烟时真实链上跑出来的是 Yahoo 429，按计划那一条会报成
   `DatasourceError`（链坏了），而它其实是「下轮再试」——报成硬失败会让健康页一直红着。
   掺进一个 `FieldDrift` 才抛 `DatasourceError`，因为重跑不会让 Yahoo 把列名改回来。
5. **`FxResult` 与 `FxSyncSummary` 各自新建，不复用 `FrameResult`/`SyncSummary`**：后两者的
   必填字段是 `market`/`code`，而外汇没有市场维度——塞 `market=""` 会让每句拒绝文案与
   每行日志多出两个读不通的空格。summary 比计划多 `skipped`、`confirmed_empty_by`、
   `latest_date` 三栏，少 `error` 一栏（失败走异常，不走字段）。
6. **`_SourceFailed` 多带一个 `cause`**：上面第 4 条要按类型分类，而类型在字符串里活不下来。
   日线那条链只用它的 `error` 文字，行为不变。
7. **`normalize_fx` 三处比计划多写**：读不出数的行（NaN/None）丢掉并记警告（`0` 是一个数，
   `None` 才是「这一格没值」）；日期解析失败抛 `FieldDrift` 而不是当「这段没数据」（答成
   「没有」会让降级链收下这句确认，库里从此缺这一段）；返回前按 `date` 排序。
   `EmptyResponse(source, …)` 要两个参数——计划里的单参数写法当场就 TypeError。
8. **`set_manual` 放在 `FxSyncService` 上**：分层闸不许命令模块碰仓库层
   （`test_layering.py`），`container.fx_store` 只出现在测试里。落库后读回来再打印——
   命令行走的是**库里的样子**，不是调用方传进来的那串字符。
9. **新增 `cli/common.currency_arg`**：小写币种不会让任何一层报错，只会让那一行永远查不到
   （换算拿配置里的 `USD` 去问，库里躺着 `usd`，症状是「换汇影响永远是 0」）。它与仓库层
   「`RMB`/`rmb` 一律拒绝」不是同一件事：这里不改字母，只把大小写收到唯一那种写法上。
10. **`ledger fx set` 刻意不吃 `--user`**：`fx_rates` 那一行没有归属，硬加 `--user` 会读成
    「这行汇率只对该用户生效」，而换算时谁都不会被问到是谁。它是 `ledger` 下唯一一条。
11. **多一份 `tests/cli/test_sync_fx_cli.py`**（计划里没有）：`sams sync fx` 的对外行为就是
    那句打印加退出码，不测这一层就没有人守「0 条」与「链上没有能给汇率的源」必须分开。
12. **`sources.py` 的 `_NOT_ADAPTERS` 加 `"fx"`**：发现性用例会把 `datasource/` 下每个模块
    当成一个应当注册实现的适配器文件，`fx.py` 是契约与规整，和 `actions.py`、
    `fundamentals.py`、`listed_date.py` 同一类。
13. **真实网络用例的 skip 只兜 `RateLimited`/`TransientError`**：`EmptyResponse` 与 `FieldDrift`
    照旧红——那条用例唯一要验的就是「这段有报价、列名是那几列」。降级链的形状由桩测守着。

`fx_rate` 挂在 `YfinanceAdapter` 上而不是新增第五个 kind，理由与估值同侧（D38）：它是同一个
源的第四种能力，拆开会让「Yahoo 挂了」变成两条互不相干的告警。它复用注入的 `self._fetch`，
所以限速与错误归类走的是日线那一条。

### `src/sams/datasource/fx.py`（全文）

```python
"""汇率采集的契约与规整（spec §5.4 规则 2 的数据来源）。

这一层只把「外面怎么写」翻译成「库里怎么写」，翻译之外不决定任何事。两件事在这里定死：

* **方向**（§5.4 规则 2）。`(base, quote)` 读作「1 单位 base = rate 单位 quote」，与
  `fx_rates` 那一列同义。源给反方向时取倒数会得到一个「大约对」的数：它对到足以让市值、
  成本、合计全都看着正常，又错到足以让每一笔跨币种盈亏都差几个百分点，而且库里查不出
  来。所以方向不对就当没答上，绝不换一个数。
* **`EmptyResponse` 在这里算一句答复**（与日线同侧、与基本面反侧，D43）。汇率没有
  「这张表里没这只票」那种 0 行形态：接口正常回空就是「这段日子没有报价」。而**挂了不算**：
  一家说没有、另一家超时，那句「没有」还不能写成确认。
"""

from __future__ import annotations

import logging
from datetime import date
from math import isfinite
from typing import Any

import polars as pl

from sams.core.errors import EmptyResponse, FieldDrift

logger = logging.getLogger("sams.datasource.fx")

#: 标准列集：`source` 是「这一行的数来自哪一家」（D54），手工覆盖时它是唯一的线索
FX_COLUMNS = ("date", "base", "quote", "rate", "source")

_SCHEMA = {
    "date": pl.Date,
    "base": pl.Utf8,
    "quote": pl.Utf8,
    "rate": pl.Float64,
    "source": pl.Utf8,
}


def has_fx(obj: Any) -> bool:
    """能力判据只看方法在不在（D38），与 `actions.has_actions` 同一条写法。"""
    return callable(getattr(obj, "fx_rate", None))


def empty_fx() -> pl.DataFrame:
    """全链都说「这段没有」时返回的形状：列集与真帧一致，调用方无需分支。"""
    return pl.DataFrame(schema=_SCHEMA)


def normalize_fx(
    raw: pl.DataFrame, *, base: str, quote: str, source: str = "unknown"
) -> pl.DataFrame:
    """校验方向、解析日期、补来源列。缺列 / 空框 / 方向反了都抛，不返回空框。

    「非有限」的那几行（NaN、None）丢掉并记一行警告：那是 pandas 侧的「这一格没值」，
    与「这一天的汇率是 0」不是一件事——后者会被仓库层的闸门当场拒收，那是应当出声的错。
    """
    missing = [column for column in ("date", "rate") if column not in raw.columns]
    if missing or raw.is_empty():
        raise EmptyResponse(source, f"汇率响应缺列或为空：{missing or '空框'}")
    rows: list[dict[str, Any]] = []
    dropped = 0
    for row in raw.to_dicts():
        value = _finite(row.get("rate"))
        if value is None:
            dropped += 1
            continue
        got_base = str(row.get("base") or base)
        got_quote = str(row.get("quote") or quote)
        if (got_base, got_quote) != (base, quote):
            raise EmptyResponse(
                source,
                f"汇率方向不对：要 {base}→{quote}，源给的是 {got_base}→{got_quote}"
                "（取倒数是「大约对」，方向写反查不出来）",
            )
        try:
            day = date.fromisoformat(str(row["date"])[:10])
        except ValueError:
            # 日期列换了写法是改版，不是「这段没数据」：把它答成「没有」会让降级链
            # 收下这句确认，然后库里从此缺这一段
            raise FieldDrift(
                source, f"汇率日期 {row['date']!r} 读不出是哪一天（要 YYYY-MM-DD 打头）"
            ) from None
        rows.append({"date": day, "base": base, "quote": quote, "rate": value, "source": source})
    if not rows:
        logger.warning("%s→%s 有 %d 行汇率读不出数，已丢弃", base, quote, dropped)
        raise EmptyResponse(source, f"{base}→{quote} 这一段没有可读出的汇率")
    return pl.DataFrame(rows, schema=_SCHEMA).sort("date")


def _finite(value: Any) -> float | None:
    """读不出一个有限数时给 None：0 是一个数，`None` 才是「这一格没值」。"""
    if value is None:
        return None
    try:
        out = float(value)
    except (TypeError, ValueError):
        return None
    return out if isfinite(out) else None
```

### `src/sams/services/fx.py`（全文）

```python
"""汇率的同步用例：把降级链的答复搬进 `fx_rates`，除此之外不决定任何事。

不放 `services/sync.py`：那个类按 `(market, code)` 组织，塞进来会让下一个读者以为
汇率也走市场路由。它也不解释数据——「今天的 USD→CNY 该用哪一天」是
`PortfolioService` 换算时问 `FxRepo.latest` 的事（D73 锚在行情日），不是这里。

一条红线：**链没答上不许落库**。半次成功写成一次成功，之后持仓那一屏的
`fx_pnl` 就建立在这次沉默上——它会显示一个看起来正常的 0。
"""

from __future__ import annotations

import logging
from dataclasses import dataclass
from datetime import date

from sams.core.errors import StorageError
from sams.datasource.registry import AdapterRegistry, Attempt
from sams.storage.repos.fx import FxRate, FxRepo

logger = logging.getLogger("sams.services.fx")


@dataclass(frozen=True)
class FxSyncSummary:
    """汇率同步的结果。

    不复用 `SyncSummary`：它的必填字段是 `market`/`code`，而外汇没有市场维度——
    把 `market=""` 塞进去，每一行打印都会多出两个空格和一句读不通的话。
    """

    base: str
    quote: str
    source: str = ""
    fetched: int = 0
    written: int = 0
    degraded: bool = False
    #: 每个有能力的源都独立答了「这段没有报价」：可以说「这几家都没有」，不能说「它没有」
    empty: bool = False
    #: 链上没有 `fx_rate` 的源（D38）：一个都没有时这一栏就是那句要说出口的缺
    skipped: tuple[tuple[str, str], ...] = ()
    attempts: tuple[Attempt, ...] = ()
    confirmed_empty_by: tuple[str, ...] = ()
    #: 最后写进去的那一天：`sams health` 用它判断汇率新不新
    latest_date: date | None = None


class FxSyncService:
    def __init__(self, registry: AdapterRegistry, fx: FxRepo) -> None:
        self.registry = registry
        self.fx = fx

    def sync(self, base: str, quote: str, start: date, end: date) -> FxSyncSummary:
        """汇率落进 `fx_rates`：跨币种持仓的 `fx_pnl` 全靠这张表有数。

        区间一次取（不在服务层按日循环）：源的历史接口本来就给一段，
        按日循环会把一次限流风险变成三十次。落库逐行走 `FxRepo.put`，
        它自带 `(date, base, quote)` 的幂等闸门，重跑不多出也不改动行。
        """
        result = self.registry.fetch_fx(base, quote, start, end)
        if result.empty_everywhere:
            # 一行都不写：写一行 0 或空记录，下次读取就会把「我们没查到」当成「它没有」
            return FxSyncSummary(
                base=base,
                quote=quote,
                source=result.source,
                empty=True,
                skipped=result.skipped,
                attempts=result.attempts,
                confirmed_empty_by=result.confirmed_empty_by,
            )
        rows = result.frame.to_dicts()
        for row in rows:
            self.fx.put(row["date"], base, quote, row["rate"], source=row["source"])
        logger.info("%s→%s 入库 %d 行（源 %s）", base, quote, len(rows), result.source)
        return FxSyncSummary(
            base=base,
            quote=quote,
            source=result.source,
            fetched=result.frame.height,
            written=len(rows),
            degraded=result.degraded,
            skipped=result.skipped,
            attempts=result.attempts,
            latest_date=result.frame["date"].max(),
        )

    def set_manual(self, base: str, quote: str, day: date, rate: float) -> FxRate:
        """采集链取不到时的出口：手工填一行，并把「这是手工的」写进库里那一列。

        走的是与 `sync` 同一个 `FxRepo.put`，不另开一条写路径：正数、有限、同币种那三道
        闸门在仓库里，抄一份到服务层迟早不一致。落库后再读一次返回——命令行走的是
        **库里的样子**，不是调用方传进来的那串字符。
        """
        self.fx.put(day, base, quote, rate, source="manual")
        stored = self.fx.get(day, base, quote)
        # 写完读不回来是库在说「这一行没落下去」：把它报成成功，下一层就拿着一个
        # 不存在的汇率去拆换汇影响
        if stored is None:  # pragma: no cover - put 之后必有一行
            raise StorageError(f"{day} 的 {base}→{quote} 汇率写完读不回来")
        logger.info("手工汇率 %s→%s %s = %g", base, quote, day, rate)
        return stored
```

```python
class _SourceFailed(Exception):
    """内部信号：把「这个源这次实际被调了几次」带出 `_call`。

    `cause` 是原始异常而不只是它的文字：汇率那条链要按「全链是不是都只抖了一下」决定
    这次算 `TransientError` 还是算一句永久的失败，而类型在字符串里活不下来。
    """

    def __init__(self, source: str, tries: int, error: str, cause: Exception | None = None) -> None:
        super().__init__(error)
        self.source = source
        self.tries = tries
        self.error = error
        self.cause = cause
```

```python
@dataclass(frozen=True)
class FxResult:
    """汇率链的返回，与 `FrameResult` 同一侧：0 行要说清是「源答了没有」还是「源没答」。

    不复用 `FrameResult`：它的必填字段是 `market`/`code`，而外汇没有市场维度——
    把 `market=""` 塞进去，每一句拒绝文案与每一行日志都会多出两个读不通的空格。
    """

    base: str
    quote: str
    frame: pl.DataFrame
    source: str
    attempts: tuple[Attempt, ...] = ()
    #: 链上没有 `fx_rate` 的源（D38）：「谁能给汇率」与健康页共用一份判据
    skipped: tuple[tuple[str, str], ...] = ()
    #: 每个有能力的源都独立答了「这段没有报价」时才非空（D36）
    confirmed_empty_by: tuple[str, ...] = ()

    @property
    def degraded(self) -> bool:
        return len(self.attempts) > 1

    @property
    def empty_everywhere(self) -> bool:
        """全链都答了「没有」，且一个都没答上——这才是可以落库的「确实没有」。"""
        return bool(self.confirmed_empty_by) and self.frame.is_empty()
```

```python
    def fetch_fx(self, base: str, quote: str, start: date, end: date) -> FxResult:
        """汇率链：注册表里所有「提供汇率」的源，按注册顺序试，与 `market_sources` 无关。

        不走 `chain_for` / `_walk` 是因为外汇没有市场维度，那两个是按 `(market, code)`
        路由的；仍然记源健康、仍然在连续失败时发 SystemAlarm——「汇率取不到」的后果
        （跨币种持仓的换汇影响拆不出）与「日线取不到」同量级，不能只让服务层的 notes
        单独承担。

        一次问整段区间（问 `start` 那天，收下源给的全部行）而不是按日循环：按日循环会把
        一次限流风险变成三十次，而汇价是连续序列，逐日问的答复不会比一次问的多。

        **「源答了这段没有」是一句答复，「源挂了」不是**（D36/D43，与日线同侧）：
        只有每个有能力的源都说了没有，才攒成一次可落库的确认 `confirmed_empty_by`。
        """
        cap = capability("fx_rate")
        attempts: list[Attempt] = []
        skipped: list[tuple[str, str]] = []
        empties: list[str] = []
        failed: list[tuple[str, Exception]] = []
        for name in self.adapters:
            adapter = self.adapter(name)
            if not has_fx(adapter):
                logger.info("汇率 跳过 %s（链上没有这个能力）", name)
                skipped.append((name, cap.lacking))
                continue
            try:
                frame = self._call_fx(name, base, quote, start, end)
            except _SourceFailed as exc:
                attempts.append(Attempt(name, False, exc.tries, exc.error))
                failed.append((name, exc.cause))
                continue
            except EmptyResponse as exc:
                attempts.append(Attempt(name, True, 1, f"无数据: {exc}"))
                empties.append(name)
                logger.info("汇率 %s 说 %s→%s 这段没有数据，换源", name, base, quote)
                continue
            self._mark_ok(name)
            attempts.append(Attempt(name, True))
            return FxResult(base, quote, frame, name, tuple(attempts), tuple(skipped))

        if empties and not failed:
            return FxResult(
                base,
                quote,
                empty_fx(),
                empties[0],
                tuple(attempts),
                tuple(skipped),
                confirmed_empty_by=tuple(empties),
            )

        detail = "；".join(f"{name}: {cause}" for name, cause in failed) or "（无）"
        said_no = "、".join(empties)
        reason = (
            f"取不到 {base}→{quote} 的汇率。链 {list(self.adapters)} 里"
            f"被跳过的源 {skipped or []}；真调用过又失败的 {detail}"
            + (
                f"；说这段没有数据的 {said_no}（因为 {'、'.join(n for n, _ in failed)} "
                f"没答上来，这些「没有」还不能写成确认）"
                if empties and failed
                else ""
            )
        )
        # 「待会儿重跑就好」还是「链真的坏了」，看的是失败本身值不值得再试一次：
        # 全链都是限流或抖动就抛 `TransientError`，掺进一个改版（`FieldDrift`）就抛
        # `DatasourceError`——调度器据这个分界决定是下轮再试还是出声告警。
        if failed and all(isinstance(cause, RETRYABLE) for _, cause in failed):
            raise TransientError("registry", reason)
        raise DatasourceError("registry", reason)

    def _call_fx(self, name: str, base: str, quote: str, start: date, end: date) -> pl.DataFrame:
        """对单个源的一次汇率取数：限速、退避、归类，与 `_call` 同一套预算。

        不复用 `_call` 是因为它按 `(market, code)` 组织调用；这里刻意把失败归类、
        源健康标记与「0 行继续问」那套语义原样抄一遍形状，让两条链对同一个源的
        判定不会分叉。
        """
        adapter = self.adapter(name)
        if self.bucket is not None:
            self.bucket.acquire()
        calls = 0

        def invoke() -> pl.DataFrame:
            nonlocal calls
            calls += 1
            return adapter.fx_rate(base, quote, start, end)

        try:
            frame = call_with_backoff(
                invoke,
                attempts=self.attempts_per_source,
                retryable=RETRYABLE,
                sleep=self.sleep,
                label=f"{name}:fx {base}→{quote}",
            )
        except EmptyResponse:
            raise
        except DatasourceError as exc:
            self._mark_failure(name, exc)
            raise _SourceFailed(name, calls, str(exc), exc) from exc
        except Exception as exc:  # noqa: BLE001 — 适配器漏归类时兜底，路由不能被打断
            wrapped = DatasourceError(name, f"{type(exc).__name__}: {exc}")
            self._mark_failure(name, wrapped)
            raise _SourceFailed(name, calls, str(wrapped), wrapped) from exc
        out = frame if isinstance(frame, pl.DataFrame) else pl.DataFrame(frame)
        if out.is_empty():
            # 一个 0 行的帧与一次 `EmptyResponse` 是同一句话：这段没有。
            # 让它走进「拿到了数据」那条分支，降级链就会替一句该继续问的空答复背书
            raise EmptyResponse(name, f"{base}→{quote} 在 {start}~{end} 返回 0 行")
        # 源的窗口比我们问的宽（Yahoo 给 `start` 前后几天）：请求说到的区间就是承诺，
        # 多出来的日子落进库会让人以为 `--end` 那天之前真的都同步过
        trimmed = out.filter(pl.col("date").is_between(start, end))
        if trimmed.is_empty():
            raise EmptyResponse(name, f"{base}→{quote} 给的行都在 {start}~{end} 之外")
        return trimmed
```

```python
            raise

            raise classify_error(self.name, exc) from exc

            raise EmptyResponse("yfinance", f"{symbol} 在 {start}~{end} 没有数据")

            (
                column
                for column in pl_frame.columns
                if column.lower() in DATE_INDEX_NAMES and parses_as_date(pl_frame, column)
            ),
            None,
        )

            raise FieldDrift(
                "yfinance",
                f"{symbol} 的响应里读不出日期列或收盘价，实际列 {pl_frame.columns}",
            )
        return normalize_fx(
            pl.DataFrame(
                {
                    "date": pl_frame.select(to_date_expr(pl.col(date_column))).to_series(),
                    "base": [base] * pl_frame.height,
                    "quote": [quote] * pl_frame.height,
                    "rate": pl_frame["Close"],
                }
            ),

            quote=quote,
```

### `src/sams/cli/common.py`（加 `currency_arg`）

```python
def currency_arg(value: str) -> str:
    """币种码归一成 ISO 4217 的大写，形状不对就当场拒收。

    小写不会让任何一层报错，只会让那一行**永远查不到**：持仓换算拿配置里的
    `USD`/`CNY` 去问 `fx_rates`，库里躺着的是 `usd`，症状是「换汇影响永远是 0」。
    这与仓库层「`RMB`/`rmb` 一律拒绝、不做模糊匹配」那条不是同一件事：这里不改字母，
    只是把大小写收到唯一的那种写法上（同 `canonical_code` 收键形态的道理，D50）。
    """
    code = value.strip().upper()
    if not re.fullmatch(r"[A-Z]{3}", code):
        raise ValidationError(f"币种 {value!r} 看不懂：要三字母代码，例如 USD / CNY")
    return code
```

### `src/sams/cli/sync.py`（`sams sync fx`）

参数：`--base --quote --start --end(默认今天) --json`，不吃 `--market`（外汇没有市场维度）。

```python
def run_fx(container: Container, args: argparse.Namespace) -> int:
    """汇率一条命令：区间一次问，答上了才一行行落库。

    取不到要**退出 1**，而不是打一行「0 条」：持仓那一屏对「换不出」的处置是说清哪一笔
    没进去，采集这层要是替它把「没数」报成一次成功的同步，下一层就没人再提了。
    「每个有能力的源都说这段没有」才是 0 与一句确认（D36），那条走 0。
    """
    try:
        base = currency_arg(args.base)
        quote = currency_arg(args.quote)
        start = iso_date(args.start)
        end = iso_date(args.end) if args.end else container.clock.today()
        if start > end:
            # 区间反了是手滑：链会照问不误然后回一句「这段没有」，把参数错报成数据缺
            raise ValidationError(f"--start {start} 在 --end {end} 之后：区间反了")
        summary = container.fx_sync.sync(base, quote, start, end)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit(
            {
                "status": "ok",
                "base": summary.base,
                "quote": summary.quote,
                "source": summary.source,
                "fetched": summary.fetched,
                "written": summary.written,
                "degraded": summary.degraded,
                "empty": summary.empty,
                "latest_date": str(summary.latest_date) if summary.latest_date else None,
                "confirmed_empty_by": list(summary.confirmed_empty_by),
                "skipped": [[name, reason] for name, reason in summary.skipped],
            }
        )
        return 0
    line = f"{summary.base}→{summary.quote}  源={summary.source or '—'}"
    if summary.empty:
        said = "+".join(summary.confirmed_empty_by)
        line += f" 入库=0  链上每个有能力的源都说这段没有报价（{said}）"
    else:
        line += f" 取回={summary.fetched} 入库={summary.written} 最新={summary.latest_date}"
    if summary.degraded:
        line += f" [降级: {_trail(summary.attempts)}]"
    print(line)
    # 跳过的是「这个源不给汇率」，不是「这个源没数据」：两者在报告上必须分开
    for name, reason in summary.skipped:
        print(f"           跳过 {name}：{reason}")
    return 0
```

`_trail(attempts)` 是这个模块已有的那一个（`sync` 的日线/除权路径都在用），没新写。

### `src/sams/cli/ledger.py`（`sams ledger fx set`）

```python
def _fx_parser(actions) -> None:
    """汇率的手工出口，唯一一条不吃 `--user` 的 `ledger` 子命令。

    它写的不是某个人的账，是一条全局事实：`fx_rates` 那一行没有归属，硬加 `--user`
    会读成「这行汇率只对该用户生效」，而换算时谁都不会被问到是谁。
    """
    add = actions.add_parser(
        "fx",
        help="set 手工填一行汇率（`sams sync fx` 那条链取不到时的出口）",
    )
    add.add_argument("action", choices=("set",))
    add.add_argument("--day", required=True, help="哪一天：YYYY-MM-DD")
    add.add_argument("--base", required=True, help="1 单位哪种币种：USD / HKD …")
    add.add_argument("--quote", required=True, help="换成哪种币种：CNY …")
    add.add_argument("--rate", required=True, type=float, help="那一天的汇率（正数）")
    add.add_argument("--json", action="store_true", help="机器可读输出（含 exit）")
    add.set_defaults(handler=run_fx)


def run_fx(container, args) -> int:
    """手工填一行汇率：正数、有限、同币种那三道闸门都由 `FxRepo.put` 说，不在这里重问。

    打印的是**落库之后读回来的那一行**（含 `source`）：把手工填进去的字符原样回显，
    等于替库背书说「这一行在这儿」。
    """
    try:
        day = iso_date(args.day)
        base = currency_arg(args.base)
        quote = currency_arg(args.quote)
        rate = container.fx_sync.set_manual(base, quote, day, args.rate)
        payload = {"status": "ok", "rate": _fx_payload(rate)}
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit(payload)
        return 0
    print(
        f"手工汇率 {rate.base}→{rate.quote}  1 {rate.base} = {rate.rate:g} {rate.quote}"
        f"  （{rate.day}，库里 source={rate.source}）"
    )
    # 换算按行情日取「当天或之前最近的一行」（D73）：填错一天不会报错，只会静默用不上
    print("  生效条件：这一天要在 `positions` 那一行的行情日当天或之前")
    return 0


def _fx_payload(rate) -> dict:
    return {
        "day": str(rate.day),
        "base": rate.base,
        "quote": rate.quote,
        "rate": rate.rate,
        #: 「manual」这一列是这条命令存在的理由：下次 `sams sync fx` 覆盖时要知道覆盖掉的是什么
        "source": rate.source,
    }
```

`bootstrap.py`：`Container` 加 `fx_sync: FxSyncService`，组装处 `fx_sync=FxSyncService(registry, fx_store)`。
