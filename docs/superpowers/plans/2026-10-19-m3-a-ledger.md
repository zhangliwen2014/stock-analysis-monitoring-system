```python
"""`Database.transaction()`：仓库层的多表写必须有一个原子出口。"""

from __future__ import annotations

import pytest

from sams.storage.db import Database


@pytest.fixture
def widget(db: Database) -> Database:
    db.execute("CREATE TABLE widget (id INTEGER PRIMARY KEY)")
    return db


def test_a_committed_block_leaves_every_row(widget: Database) -> None:
    with widget.transaction():
        widget.execute("INSERT INTO widget VALUES (1)")
        widget.execute("INSERT INTO widget VALUES (2)")
    assert widget.row_count("widget") == 2


def test_an_exception_rolls_back_the_whole_block(widget: Database) -> None:
    """回滚必须是整段：留下第 1 行就等于留下一笔已经扣了现金却没建的仓。"""
    with pytest.raises(ValueError), widget.transaction():
        widget.execute("INSERT INTO widget VALUES (1)")
        raise ValueError("后半段炸了")
    assert widget.row_count("widget") == 0


def test_nested_blocks_join_the_outer_one(widget: Database) -> None:
    """内层不另开事务，也不自己提交。

    `record_fill` 包住一整笔成交，顺手动现金的 `adjust_cash` 自己也包一层。如果嵌套
    报错，这两个方法就只能有一个真的开事务，而「谁负责原子性」从代码掉进口头约定；
    DuckDB 对嵌套 BEGIN 的原始报错读起来又像驱动 bug。所以这里明确支持重入。
    """
    with widget.transaction():
        with widget.transaction():
            widget.execute("INSERT INTO widget VALUES (1)")
        assert widget.row_count("widget") == 1  # 外层还没提交，但同连接可见
    assert widget.row_count("widget") == 1


def test_an_inner_exception_rolls_back_the_outer_block(widget: Database) -> None:
    with pytest.raises(ValueError), widget.transaction():
        widget.execute("INSERT INTO widget VALUES (1)")
        with widget.transaction():
            raise ValueError("内层炸了")
    assert widget.row_count("widget") == 0


def test_the_transaction_flag_reflects_the_outermost_block(widget: Database) -> None:
    assert widget.in_transaction is False
    with widget.transaction():
        assert widget.in_transaction is True
        with widget.transaction():
            assert widget.in_transaction is True
        assert widget.in_transaction is True
    assert widget.in_transaction is False
```

在 `execute` 之前插入：

```python
    @property
    def in_transaction(self) -> bool:
        """是否已经在一个事务块里（重入计数大于 0）。仓库层用它决定要不要自己开。"""
        return self._tx_depth > 0

    @contextmanager
    def transaction(self) -> Iterator[None]:
        """一个原子写单元：块内任何异常都整段回滚，包括调用方没接住的。

        账本的一笔卖出同时写 `sim_orders`、`sim_positions`、`sim_cash_balances`、
        `sim_cash_moves`：中途失败留下的「现金已扣、持仓未减」不是待修的脏数据，
        是整本账失去意义。重入时内层只是加入外层那个事务（DuckDB 不支持嵌套 BEGIN），
        所以任何一层抛异常，最外层统一回滚——「谁负责原子性」由第一个开块的人定，
        不由谁碰巧写了 BEGIN 定。迁移有自己的 BEGIN/COMMIT（`migrate.py`），
        那是 DDL 的引导路径，不需要也不该走这里。
        """
        if self._tx_depth == 0:
            self._run("BEGIN TRANSACTION", None)
        self._tx_depth += 1
        try:
            yield
        except Exception:
            if self._tx_depth == 1:
                self._run("ROLLBACK", None)
            raise
        else:
            if self._tx_depth == 1:
                self._run("COMMIT", None)
        finally:
            self._tx_depth -= 1
```

`__init__` 末尾加一行（`self.path = path` 之后）：

## Task T-2: `core/funds.py` —— 币种与两段拆分的唯一权威

**这是整个 M3-A 最重要的一块。** spec §5.4 的四条硬规则里，「录入与存储永远用标的币种」和「price_pnl + fx_pnl 两段拆分」都是数学，不是策略；数学放对地方，后面每个读用例（T-8/T-9）就只是调用，放错地方就会长出三份略微不一样的换算——而三份不一样的换算在账本里叫对不上账。

恒等式（已用代数验算，实现必须逐字对上）：

```
price_pnl = qty * (last_price − avg_cost)                       # 标的币种
fx_pnl    = qty * last_price * (rate_today − rate_base)          # 报告币种
price_pnl * rate_base + fx_pnl == qty * last_price * rate_today − qty * avg_cost * rate_base
```

右边正是「今日市值（报告币种）− 成本（报告币种）」。所以 `rate_base` 必须是**下单那天**的汇率（T-1 的 `fx_rate_ref`），`rate_today` 是今天；两段合起来不多不少等于总市值变动，既不会把换汇算成股价涨，也不会漏掉它。

```python
"""钱相关的纯数学（spec §5.4）。

放在 `core` 且不许 import 上层，是因为这段算术会被 T-8/T-9/T-12 和 M3-C 的规则引擎
反复用到：一旦有两份略微不同的换算，账本就从「可核对」变成「两份对不上的数」。
这里没有 IO：汇率、价格、均价都由调用方传进来，函数只负责算对。
"""

from __future__ import annotations

from dataclasses import dataclass

from sams.core.codes import canonical_market, canonical_code, currency_override
from sams.core.config import MarketCfg
from sams.core.errors import ValidationError

#: 超过这个天数的汇率算「陈旧」：4 = 周五汇率撑到周一不算陈旧，跨长假算。
FX_STALE_DAYS = 4

#: 报告币种里常见的三种符号；其余显示 `1234.50 XXX`，不猜符号。
_SYMBOLS = {"CNY": "¥", "USD": "$", "HKD": "HK$"}


def quote_currency(market: str, code: str, cfg: dict[str, MarketCfg]) -> str:
    """这个标的用什么币种记账（§5.4 硬规则 1：录入与存储永远用标的币种）。

    `currency_override` 先于市场默认值：B 股（`200*`/`900*`）以 USD/HKD 交易，
    HK 双柜台以 RMB 交易（D31）。反过来用市场币种会把同一只票的两个柜台
    记成两种成本。
    """
    key = canonical_market(market)
    try:
        currency = cfg[key].currency
    except KeyError:
        raise ValidationError(
            f"市场 {key} 未配置，给不出标的币种；请在配置的 [markets.{key}] 里补 currency"
        ) from None
    return currency_override(key, canonical_code(key, code)) or currency


def blended_cost(old_qty: float, old_avg: float, qty: float, price: float, fee: float) -> float:
    """加仓后的加权均价；手续费计入成本（§5.4「含费」口径）。"""
    if old_qty <= 0:
        return (qty * price + fee) / qty
    return (old_qty * old_avg + qty * price + fee) / (old_qty + qty)


def realized_pnl(qty: float, price: float, avg_cost: float, fee: float) -> float:
    """卖出实现的盈亏，标的币种，已扣手续费。"""
    return qty * (price - avg_cost) - fee


def weighted_rate(entries: list[tuple[float, float, bool]]) -> float | None:
    """按金额加权的汇率；`entries` 是 (该笔金额的标的币种数额, 其汇率, is_stale)。

    先剔陈旧：一个过期汇率混进多币种汇总，等于让周五的汇率决定周一的总市值。
    全空或全是陈旧返回 `None` 而不是 0——0 会把市值抹平成 0，而「算不出来」和
    「值是 0」是两件事。取数归调用方（`core` 不查库），这里只做算术。
    """
    usable = [(amount, rate) for amount, rate, stale in entries if not stale and amount > 0]
    if not usable:
        return None
    total_amount = sum(amount for amount, _ in usable)
    if total_amount <= 0:
        return None
    return sum(amount * rate for amount, rate in usable) / total_amount


@dataclass(frozen=True)
class PositionPnl:
    """一个持仓的两段盈亏。字段单位写在注释里，因为混读它们是本模块唯一的危险。"""

    qty: float
    avg_cost: float
    last_price: float
    price_pnl: float  # 标的币种
    fx_pnl: float  # 报告币种
    market_report: float  # 报告币种：qty*last_price*rate_today
    cost_report: float  # 报告币种：qty*avg_cost*rate_base
    rate_base: float | None
    rate_today: float | None
    rate_stale: bool

    @property
    def total_report(self) -> float:
        """报告币种里的「市值 − 成本」。恒等式：`price_pnl*rate_base + fx_pnl` 与它相等。"""
        return self.market_report - self.cost_report


def position_pnl(
    *,
    qty: float,
    avg_cost: float,
    last_price: float,
    rate_base: float | None,
    rate_today: float | None,
    rate_today_stale: bool = False,
) -> PositionPnl:
    """把持仓盈亏拆成价格段与换汇段（§5.4 line 434）。

    `rate_base` 是**下单那天**的汇率（`sim_orders.fx_rate_ref` 加权），`rate_today` 是今天。
    两者相同（同币种，或汇率没动）时 `fx_pnl` 为 0。传 `None` 表示同币种，
    此时不做换算，`market_report` 直接按 1:1 计入——这是报告币种=标的币种的正常路径，
    不是缺数降级。
    """
    price_pnl = qty * (last_price - avg_cost)
    if rate_base is None or rate_today is None:
        fx_pnl = 0.0
        market_report = qty * last_price
        cost_report = qty * avg_cost
        stale = False
    else:
        fx_pnl = qty * last_price * (rate_today - rate_base)
        market_report = qty * last_price * rate_today
        cost_report = qty * avg_cost * rate_base
        stale = rate_today_stale
    return PositionPnl(
        qty=qty,
        avg_cost=avg_cost,
        last_price=last_price,
        price_pnl=price_pnl,
        fx_pnl=fx_pnl,
        market_report=market_report,
        cost_report=cost_report,
        rate_base=rate_base,
        rate_today=rate_today,
        rate_stale=stale,
    )


def money_note(amount: float, currency: str) -> str:
    """带币种的金额文本（§5.4 硬规则 3：金额阈值必须带币种）。"""
    symbol = _SYMBOLS.get(currency)
    if symbol is None:
        return f"{amount:.2f} {currency}"
    return f"{symbol}{amount:.2f}"
```

```python
"""模拟盘账本的仓库层（spec §5.4）。

四张表的写入口收在这一个模块的两个类里：`SimAccountRepo`（账户/现金/流水）与
`SimOrderRepo`（订单/持仓）。放在同一层是因为一次成交天然跨两张表，而分开放的话
「谁负责开事务」会变成口头约定。

非负现金、币种一致这类前置条件在仓库强制，理由同 D81：M3 会引入绕过服务层的写入口，
一个忘了查的入口留下的不是一句报错，而是一本看起来正常的假账。
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import UTC, datetime

import polars as pl

from sams.core.errors import StorageError, ValidationError
from sams.storage.db import Database
from sams.storage.ids import new_id

#: 成交/调现金的时刻：带时区才收，裸 datetime 一律拒。
#: TIMESTAMPTZ 列会按会话时区解释无时区的值，那等于让 A 股的一笔记成 UTC，差 8 小时。
def _moment(ts: datetime | None) -> datetime:
    if ts is None:
        return datetime.now(UTC)
    if ts.tzinfo is None:
        raise ValidationError("时间必须带时区（例如 2024-10-25T15:00+08:00）：裸时刻会被当成 UTC")
    return ts.astimezone(UTC)


@dataclass(frozen=True)
class Account:
    id: str
    user_id: str
    name: str
    broker: str | None
    market_scope: tuple[str, ...]
    fee_model: str
    margin_enabled: bool
    report_currency: str


def _row_to_account(row: dict) -> Account:
    import json

    return Account(
        id=row["id"],
        user_id=row["user_id"],
        name=row["name"],
        broker=row["broker"],
        market_scope=tuple(json.loads(row["market_scope"] or "[]")),
        fee_model=row["fee_model"],
        margin_enabled=bool(row["margin_enabled"]),
        report_currency=row["report_currency"],
    )


class SimAccountRepo:
    """账户、按币种的现金余额、以及每一分钱的来处。"""

    def __init__(self, db: Database) -> None:
        self.db = db

    def open(
        self,
        user_id: str,
        name: str,
        *,
        broker: str | None = None,
        market_scope: tuple[str, ...] = (),
        fee_model: str = "default",
        margin_enabled: bool = False,
        report_currency: str = "CNY",
    ) -> Account:
        import json

        if not name.strip():
            raise ValidationError("账户名不能为空")
        account_id = new_id()
        try:
            self.db.execute(
                "INSERT INTO sim_accounts (id, user_id, name, broker, market_scope, "
                "fee_model, margin_enabled, report_currency) VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
                [
                    account_id,
                    user_id,
                    name,
                    broker,
                    json.dumps(list(market_scope)),
                    fee_model,
                    margin_enabled,
                    report_currency,
                ],
            )
        except StorageError as exc:
            # 唯一键 (user_id, name) 冲突要读成「这个名字已经有了」，不是「SQL 失败」
            if "Constraint" in str(exc) or "unique" in str(exc).lower():
                raise StorageError(f"账户 {name!r} 已存在（每个用户的账户名唯一）") from exc
            raise
        account = self.get(user_id, account_id)
        assert account is not None  # 刚写完，读不到就是库出了问题
        return account

    def get(self, user_id: str, account_id: str) -> Account | None:
        rows = self.db.query(
            "SELECT * FROM sim_accounts WHERE id = ? AND user_id = ?", [account_id, user_id]
        )
        return None if rows.is_empty() else _row_to_account(rows.to_dicts()[0])

    def list_for(self, user_id: str) -> list[Account]:
        rows = self.db.query(
            "SELECT * FROM sim_accounts WHERE user_id = ? ORDER BY created_at", [user_id]
        )
        return [_row_to_account(r) for r in rows.to_dicts()]

    def cash_balances(self, account_id: str) -> dict[str, float]:
        rows = self.db.query(
            "SELECT currency, amount FROM sim_cash_balances WHERE account_id = ? "
            "ORDER BY currency",
            [account_id],
        )
        return {r["currency"]: float(r["amount"]) for r in rows.to_dicts()}

    def cash_history(self, account_id: str) -> pl.DataFrame:
        """审计行按时间正序：倒序的流水在核对「哪一步开始对不上」时更难读。"""
        return self.db.query(
            "SELECT currency, delta, reason, actor_id, ts FROM sim_cash_moves "
            "WHERE account_id = ? ORDER BY ts, rowid",
            [account_id],
        )

    def adjust_cash(
        self,
        account_id: str,
        currency: str,
        delta: float,
        *,
        reason: str,
        actor_id: str,
        ts: datetime | None = None,
    ) -> float:
        """动现金的唯一入口：余额与流水同事务，失败整段回滚。

        返回调整后的余额，因为调用方（成交记账）要拿它判断「这一步之后还有多少钱」。
        """
        if not reason or not reason.strip():
            raise ValidationError("调整现金必须给理由：审计行没有理由就不是审计")
        if delta == 0:
            raise ValidationError("delta 不能是 0：没动就不要留一条流水")
        if currency not in {"CNY", "USD", "HKD", "EUR", "JPY", "GBP", "SGD", "AUD", "CAD"}:
            raise ValidationError(f"不认识币种 {currency!r}：三字母代码，不做模糊匹配")
        account = self._require(account_id)
        moment = _moment(ts)
        current = float(
            self.db.scalar(
                "SELECT amount FROM sim_cash_balances WHERE account_id = ? AND currency = ?",
                [account_id, currency],
            )
            or 0.0
        )
        after = current + delta
        if after < 0 and not account.margin_enabled:
            raise StorageError(
                f"现金不足：{currency} 余额 {current:.2f}，本次要动 {delta:.2f}，"
                f"结果 {after:.2f}。无杠杆账户不透支（§5.4）；"
                f"要记融资请在开户时把 margin_enabled 打开，或换一笔够钱的金额"
            )
        with self.db.transaction():
            self.db.execute(
                "INSERT INTO sim_cash_moves (id, account_id, currency, delta, reason, actor_id, ts) "
                "VALUES (?, ?, ?, ?, ?, ?, ?)",
                [new_id(), account_id, currency, delta, reason, actor_id, moment],
            )
            self.db.execute(
                "INSERT OR REPLACE INTO sim_cash_balances (account_id, currency, amount, updated_at) "
                "VALUES (?, ?, ?, ?)",
                [account_id, currency, after, moment],
            )
        return after

    def _require(self, account_id: str) -> Account:
        """账户是否存在（不校验归属，那是读用例那一层带 user_id 的事）。"""
        rows = self.db.query(
            "SELECT * FROM sim_accounts WHERE id = ?", [account_id]
        )
        if rows.is_empty():
            raise StorageError(f"账户 {account_id} 不存在")
        return _row_to_account(rows.to_dicts()[0])
```

```python
BUY = "buy"
SELL = "sell"
SIDES = (BUY, SELL)

#: 一次成交动的四张表里，只有订单是「历史」；持仓和余额是「状态」。
#: 状态可以由历史重放出来，所以写状态必须在历史之后、同一个事务里。


@dataclass(frozen=True)
class Position:
    account_id: str
    market: str
    code: str
    currency: str
    qty: float
    avg_cost: float
    #: 第一次建仓那一刻（不是最后一次成交）：§9.3 的「拿了多久」全靠它
    open_ts: datetime | None
    last_price: float | None
    last_price_ts: datetime | None


@dataclass(frozen=True)
class Order:
    id: str
    account_id: str
    market: str
    code: str
    currency: str
    side: str
    qty: float
    price: float
    ts: datetime
    fee: float
    thesis_id: str | None
    fx_rate_ref: float | None


def _row_to_order(row: dict) -> Order:
    return Order(
        id=row["id"],
        account_id=row["account_id"],
        market=row["market"],
        code=row["code"],
        currency=row["currency"],
        side=row["side"],
        qty=float(row["qty"]),
        price=float(row["price"]),
        ts=row["ts"],
        fee=float(row["fee"] or 0.0),
        thesis_id=row["thesis_id"],
        fx_rate_ref=row.get("fx_rate_ref"),
    )


def _finite(value: float) -> bool:
    from math import isfinite

    return isfinite(value)


class SimOrderRepo:
    """订单与持仓。`record_fill` 是唯一的成交入口。"""

    def __init__(self, db: Database, accounts: SimAccountRepo | None = None) -> None:
        self.db = db
        #: 现金与流水归 SimAccountRepo 管；成交要顺手动它，所以持有同一个 db 上的一个实例
        self.accounts = accounts or SimAccountRepo(db)

    def record_fill(
        self,
        *,
        account_id: str,
        user_id: str,
        market: str,
        code: str,
        side: str,
        qty: float,
        price: float,
        currency: str,
        fee: float = 0.0,
        thesis_id: str | None = None,
        fx_rate_ref: float | None = None,
        ts: datetime | None = None,
    ) -> Order:
        from sams.core.codes import canonical_code, canonical_market
        from sams.core.funds import blended_cost, realized_pnl

        if side not in SIDES:
            raise ValidationError(
                f"side 只能是 {'/'.join(SIDES)}，收到 {side!r}"
            )
        for label, value in (("qty", qty), ("price", price), ("fee", fee)):
            if not _finite(value):
                raise ValidationError(f"{label} 必须是有限小数，收到 {value!r}")
        if qty <= 0:
            raise ValidationError("qty 必须大于 0：清仓请用 sell 而不是 buy 0")
        if fee < 0:
            raise ValidationError("手续费不能为负：负手续费等于把成本做低")
        market = canonical_market(market)
        code = canonical_code(market, code)
        moment = _moment(ts)
        existing = self.position(account_id, market, code, strict=False)
        if side == SELL:
            held = existing.qty if existing else 0.0
            if qty > held:
                raise StorageError(
                    f"超卖：{market} {code} 持有 {held:g}，本次要卖 {qty:g}"
                )
        gross = qty * price
        cash_delta = -gross - fee if side == BUY else gross - fee
        order_id = new_id()
        with self.db.transaction():
            self.db.execute(
                "INSERT INTO sim_orders (id, account_id, market, code, currency, side, qty, "
                "price, ts, fee, fee_currency, fx_rate_applied, thesis_id, fx_rate_ref) "
                "VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
                [
                    order_id, account_id, market, code, currency, side, qty, price,
                    moment, fee, currency, fx_rate_ref, thesis_id, fx_rate_ref,
                ],
            )
            if side == BUY:
                new_qty = (existing.qty if existing else 0.0) + qty
                new_avg = blended_cost(
                    existing.qty if existing else 0.0,
                    existing.avg_cost if existing else 0.0,
                    qty,
                    price,
                    fee,
                )
            else:
                held = existing.qty if existing else 0.0
                new_qty = held - qty
                # 清仓后均价归 0：留着最后一笔成本会被读成「还持有一些」
                new_avg = existing.avg_cost if new_qty > 0 else 0.0
            self.db.execute(
                "INSERT OR REPLACE INTO sim_positions (account_id, market, code, currency, qty, "
                "avg_cost, open_ts, last_price, last_price_ts) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
                [
                    account_id, market, code, currency, new_qty, new_avg,
                    # 建仓日是第一次那一刻，不是最后一次：覆盖它等于抹掉「拿了多久」
                    existing.open_ts if existing else moment,
                    price,
                    moment,
                ],
            )
            self.accounts.adjust_cash(
                account_id,
                currency,
                cash_delta,
                reason=f"{'买入' if side == BUY else '卖出'} {code}",
                actor_id=user_id,
                ts=moment,
            )
        order = self.order(account_id, order_id)
        assert order is not None
        return order

    def order(self, account_id: str, order_id: str) -> Order | None:
        rows = self.db.query(
            "SELECT * FROM sim_orders WHERE id = ? AND account_id = ?", [order_id, account_id]
        )
        return None if rows.is_empty() else _row_to_order(rows.to_dicts()[0])

    def orders(self, account_id: str) -> list[Order]:
        rows = self.db.query(
            "SELECT * FROM sim_orders WHERE account_id = ? ORDER BY ts, id", [account_id]
        )
        return [_row_to_order(r) for r in rows.to_dicts()]

    def order_rows(self, account_id: str) -> int:
        return int(
            self.db.scalar("SELECT count(*) FROM sim_orders WHERE account_id = ?", [account_id])
            or 0
        )

    def orders_for_thesis(self, thesis_id: str) -> list[Order]:
        """U26：某个立论到底有没有真成交。跨账户查，因为 thesis 属于用户而账户也属于用户。"""
        rows = self.db.query(
            "SELECT o.* FROM sim_orders o JOIN sim_accounts a ON a.id = o.account_id "
            "WHERE o.thesis_id = ? ORDER BY o.ts",
            [thesis_id],
        )
        return [_row_to_order(r) for r in rows.to_dicts()]

    def positions(self, account_id: str) -> list[Position]:
        rows = self.db.query(
            "SELECT * FROM sim_positions WHERE account_id = ? ORDER BY market, code", [account_id]
        )
        return [_row_to_position(r) for r in rows.to_dicts()]

    def position(
        self, account_id: str, market: str, code: str, *, strict: bool = True
    ) -> Position | None:
        rows = self.db.query(
            "SELECT * FROM sim_positions WHERE account_id = ? AND market = ? AND code = ?",
            [account_id, canonical_market(market), canonical_code(market, code)],
        )
        if rows.is_empty():
            if strict:
                raise StorageError(f"账户 {account_id} 没有 {market} {code} 的持仓")
            return None
        return _row_to_position(rows.to_dicts()[0])

    def realized_pnl(self, account_id: str, market: str, code: str) -> float:
        """已实现盈亏（标的币种，已扣费）：由订单历史算，不另存一列。

        存下来就得再维护「成本基础随加仓变化」这件事，而它已经在 `sim_orders` 里了。
        """
        rows = self.db.query(
            "SELECT side, qty, price, fee FROM sim_orders WHERE account_id = ? "
            "AND market = ? AND code = ? ORDER BY ts",
            [account_id, canonical_market(market), canonical_code(market, code)],
        )
        if rows.is_empty():
            return 0.0
        running = 0.0
        basis = 0.0
        total = 0.0
        for r in rows.to_dicts():
            qty, price, fee = float(r["qty"]), float(r["price"]), float(r["fee"] or 0.0)
            if r["side"] == BUY:
                basis += qty * price + fee
                running += qty
            else:
                avg = basis / running if running > 0 else 0.0
                total += realized_pnl(qty=qty, price=price, avg_cost=avg, fee=fee)
                basis -= qty * avg
                running -= qty
        return total


def _row_to_position(row: dict) -> Position:
    return Position(
        account_id=row["account_id"],
        market=row["market"],
        code=row["code"],
        currency=row["currency"],
        qty=float(row["qty"]),
        avg_cost=float(row["avg_cost"]),
        open_ts=row["open_ts"],
        last_price=row["last_price"],
        last_price_ts=row["last_price_ts"],
    )
```

```python
"""`fx_rates` 的读写（spec §5.4 规则 2/3 的数据侧）。

存的时候只存一个方向：`(base, quote)` 表示 1 单位 base = rate 单位 quote。
读的时候**不取倒数**——方向写反的症状是「大约对」，那是最难查的错。
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import date

from sams.core.errors import ValidationError
from sams.core.funds import FX_STALE_DAYS
from sams.storage.db import Database


@dataclass(frozen=True)
class FxRate:
    day: date
    base: str
    quote: str
    rate: float
    source: str | None
    is_stale: bool
    #: `latest` 才填：距今几天。当天为 0。
    age_days: int = 0


class FxRepo:
    def __init__(self, db: Database) -> None:
        self.db = db

    def put(
        self,
        day: date,
        base: str,
        quote: str,
        rate: float,
        *,
        source: str | None = None,
        is_stale: bool = False,
    ) -> None:
        from math import isfinite

        if not isfinite(rate) or rate <= 0:
            raise ValidationError(f"汇率必须是正有限小数，收到 {rate!r}")
        if base == quote:
            raise ValidationError(f"{base} 对 {quote} 的汇率不需要存：同币种恒为 1")
        self.db.execute(
            "INSERT OR REPLACE INTO fx_rates (date, base_ccy, quote_ccy, rate, source, is_stale) "
            "VALUES (?, ?, ?, ?, ?, ?)",
            [day, base, quote, rate, source, is_stale],
        )

    def get(self, day: date, base: str, quote: str) -> FxRate | None:
        if base == quote:
            return FxRate(day, base, quote, 1.0, "same", False, 0)
        rows = self.db.query(
            "SELECT * FROM fx_rates WHERE date = ? AND base_ccy = ? AND quote_ccy = ?",
            [day, base, quote],
        )
        return None if rows.is_empty() else _row(rows.to_dicts()[0], 0)

    def latest(
        self, day: date, base: str, quote: str, *, max_age_days: int | None = None
    ) -> FxRate | None:
        """`day` 或之前最近的一条；超过 `max_age_days` 就不给（宁可没有，不要过期）。

        默认 `FX_STALE_DAYS`：周末用周五的汇率是合法的（标的不涨也不跌），
        长假第七天还在用同一个数是假数。
        """
        if base == quote:
            return FxRate(day, base, quote, 1.0, "same", False, 0)
        limit = FX_STALE_DAYS if max_age_days is None else max_age_days
        rows = self.db.query(
            "SELECT * FROM fx_rates WHERE base_ccy = ? AND quote_ccy = ? AND date <= ? "
            "ORDER BY date DESC LIMIT 1",
            [base, quote, day],
        )
        if rows.is_empty():
            return None
        row = rows.to_dicts()[0]
        age = (day - row["date"]).days
        if age > limit:
            return None
        return _row(row, age)


def _row(row: dict, age_days: int) -> FxRate:
    return FxRate(
        day=row["date"],
        base=row["base_ccy"],
        quote=row["quote_ccy"],
        rate=float(row["rate"]),
        source=row["source"],
        #: 陈旧是算出来的，不是存进去的：存进去的那一刻它不陈旧，第二天就陈旧了
        is_stale=bool(row["is_stale"]) or age_days > 0,
        age_days=age_days,
    )
```

```python
币种由配置推定、由仓库校验；thesis 链接由服务层校验，因为「这个立论现在能不能记成交」
问的是状态机，而状态机已经在 `repos/theses.py` 的 `TRANSITIONS` 里（D81）。
"""

from __future__ import annotations

from datetime import datetime

from sams.core.codes import canonical_code, canonical_market
from sams.core.config import Settings
from sams.core.errors import ValidationError
from sams.core.funds import quote_currency
from sams.storage.repos.ledger import Account, Order, Position, SimAccountRepo, SimOrderRepo
from sams.storage.repos.theses import (
    ARCHIVED,
    INVALIDATED,
    OBSERVING,
    OPENED,
    PROPOSED,
    ThesisRepo,
)

#: 能挂成交的立论状态：归档/失效的能挂（「事后按这个价位买了」是真实发生过的），
#: 还没建仓的不能挂（那会在时间线上留下一个从未发生的「我当时买了」）。
_FILLABLE_BUT_UNBORN = (OBSERVING, PROPOSED)


class LedgerService:
    def __init__(
        self,
        accounts: SimAccountRepo,
        orders: SimOrderRepo,
        theses: ThesisRepo,
        settings: Settings,
    ) -> None:
        self.accounts = accounts
        self.orders = orders
        self.theses = theses
        self.settings = settings

    def open_account(
        self,
        user_id: str,
        name: str,
        *,
        broker: str | None = None,
        market_scope: tuple[str, ...] = (),
        margin_enabled: bool = False,
    ) -> Account:
        return self.accounts.open(
            user_id,
            name,
            broker=broker,
            market_scope=market_scope,
            margin_enabled=margin_enabled,
            report_currency=self.settings.report_currency,
        )

    def deposit(self, account_id: str, currency: str, amount: float, *, reason: str, actor_id: str) -> float:
        return self.accounts.adjust_cash(account_id, currency, abs(amount), reason=reason, actor_id=actor_id)

    def withdraw(self, account_id: str, currency: str, amount: float, *, reason: str, actor_id: str) -> float:
        return self.accounts.adjust_cash(account_id, currency, -abs(amount), reason=reason, actor_id=actor_id)

    def buy(
        self,
        account_id: str,
        user_id: str,
        market: str,
        code: str,
        *,
        qty: float,
        price: float,
        fee: float = 0.0,
        thesis_id: str | None = None,
        fx_rate_ref: float | None = None,
        ts: datetime | None = None,
    ) -> Order:
        return self._fill(
            account_id, user_id, market, code, side="buy", qty=qty, price=price,
            fee=fee, thesis_id=thesis_id, fx_rate_ref=fx_rate_ref, ts=ts,
        )

    def sell(self, account_id: str, user_id: str, market: str, code: str, *, qty: float,
             price: float, fee: float = 0.0, thesis_id: str | None = None,
             fx_rate_ref: float | None = None, ts: datetime | None = None) -> Order:
        return self._fill(
            account_id, user_id, market, code, side="sell", qty=qty, price=price,
            fee=fee, thesis_id=thesis_id, fx_rate_ref=fx_rate_ref, ts=ts,
        )

    def _fill(self, account_id, user_id, market, code, *, side, qty, price, fee,
              thesis_id, fx_rate_ref, ts) -> Order:
        market = canonical_market(market)
        code = canonical_code(market, code)
        # 未配置的市场在这里就拒：它的币种推不出来（同 D84 的立论闸门）
        currency = quote_currency(market, code, self.settings.markets)
        if thesis_id is not None:
            self._check_thesis(user_id, thesis_id)
        return self.orders.record_fill(
            account_id=account_id, user_id=user_id, market=market, code=code, side=side,
            qty=qty, price=price, currency=currency, fee=fee, thesis_id=thesis_id,
            fx_rate_ref=fx_rate_ref, ts=ts,
        )

    def _check_thesis(self, user_id: str, thesis_id: str) -> None:
        thesis = self.theses.get(user_id, thesis_id)
        if thesis is None:
            raise ValidationError(f"thesis {thesis_id} 不存在或不属于用户 {user_id}")
        if thesis.user_id != user_id:
            raise ValidationError(f"thesis {thesis_id} 不属于用户 {user_id}")
        if thesis.status in _FILLABLE_BUT_UNBORN:
            raise ValidationError(
                f"立论 {thesis_id} 还在 {thesis.status} 阶段，不能记成交："
                f"先走到 opened（建仓）再记，否则时间线上会出现一个从未发生的「我当时买了」"
            )

    def accounts_of(self, user_id: str) -> list[Account]:
        return self.accounts.list_for(user_id)

    def position(self, account_id: str, market: str, code: str) -> Position | None:
        return self.orders.position(account_id, market, code, strict=False)

    def cash(self, account_id: str) -> dict[str, float]:
        return self.accounts.cash_balances(account_id)
```

## Task T-8: `PortfolioService.positions` —— 现价取未复权收盘（D73）

`avg_cost` 是成交那天的真价格，所以现价必须取**未复权 close**；取 hfq 绝对值会把
「持仓盈亏」算成一个从来没存在过的价格差（D73 立这条正是因为这两条链在 M2 已经分开了）。
没有行情时现价是 `None`，不是「用成本价顶一下」——顶一下会把「今天没数」显示成「今天持平」。

汇率那一天取：**与现价同日或之前最近的一条**（`FxRepo.latest(price_day, ...)`）。
这样 `PortfolioService` 不需要墙钟，也顺手避开了 `core/clock.py` 开头那句警告——
用 UTC 今日去推「哪个交易日」在美股上恒错一天，而把汇率锚在**已经有行情的那一天**
根本不需要问今天。

```python
"""持仓与账户的读用例（spec §5.4）。

写不在这个类里（`LedgerService`）：读用例不该有机会改账。
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import date

from sams.core.config import Settings
from sams.core.errors import ValidationError
from sams.core.funds import PositionPnl, position_pnl
from sams.storage.repos.bars import BarRepo
from sams.storage.repos.fx import FxRepo
from sams.storage.repos.ledger import SimAccountRepo, SimOrderRepo


@dataclass(frozen=True)
class PositionRow:
    market: str
    code: str
    currency: str
    qty: float
    avg_cost: float
    last_price: float | None
    last_price_date: date | None
    pnl: PositionPnl | None
    #: 给不出现价/换不出币时的一句话（显示与 `--json` 读同一句，别长两张嘴）
    unavailable_note: str | None = None
    #: False = 这一行不进汇总（缺现价或缺汇率），True = 数字可信
    convertible: bool = False

    @property
    def price_pnl(self) -> float | None:
        return None if self.pnl is None else self.pnl.price_pnl

    @property
    def fx_pnl(self) -> float:
        return 0.0 if self.pnl is None else self.pnl.fx_pnl

    @property
    def total_report(self) -> float:
        """报告币种里的市值−成本。没有现价的持仓贡献 0，但会在 `unavailable_note` 里点名。"""
        return 0.0 if self.pnl is None else self.pnl.total_report

    @property
    def rate_base(self) -> float | None:
        return None if self.pnl is None else self.pnl.rate_base

    @property
    def rate_today(self) -> float | None:
        return None if self.pnl is None else self.pnl.rate_today


class PortfolioService:
    def __init__(
        self,
        accounts: SimAccountRepo,
        orders: SimOrderRepo,
        bars: BarRepo,
        fx: FxRepo,
        settings: Settings,
    ) -> None:
        self.accounts = accounts
        self.orders = orders
        self.bars = bars
        self.fx = fx
        self.settings = settings

    def positions(self, user_id: str, account_id: str) -> list[PositionRow]:
        # 读也带 user_id：与 repos/theses.py 同形，不提供「查全量再由调用方过滤」的接口
        if self.accounts.get(user_id, account_id) is None:
            raise ValidationError(f"账户 {account_id} 不存在或不属于用户 {user_id}")
        rows: list[PositionRow] = []
        for pos in self.orders.positions(account_id):
            if pos.qty <= 0:
                continue
            price, price_day = self._last_price(pos.market, pos.code)
            if price is None:
                rows.append(
                    PositionRow(
                        market=pos.market,
                        code=pos.code,
                        currency=pos.currency,
                        qty=pos.qty,
                        avg_cost=pos.avg_cost,
                        last_price=None,
                        last_price_date=None,
                        pnl=None,
                        unavailable_note=(
                            f"{pos.market} {pos.code} 取不到现价（库里没有行情），"
                            f"本持仓不计入盈亏"
                        ),
                    )
                )
                continue
            rates = self._rates(account_id, pos, price_day)
            rows.append(
                PositionRow(
                    market=pos.market,
                    code=pos.code,
                    currency=pos.currency,
                    qty=pos.qty,
                    avg_cost=pos.avg_cost,
                    last_price=price,
                    last_price_date=price_day,
                    pnl=position_pnl(
                        qty=pos.qty,
                        avg_cost=pos.avg_cost,
                        last_price=price,
                        rate_base=rates[0],
                        rate_today=rates[1],
                        rate_today_stale=rates[2],
                    ),
                    convertible=rates[3],
                    unavailable_note=rates[4],
                )
            )
        return rows

    def _last_price(self, market: str, code: str) -> tuple[float | None, date | None]:
        """未复权收盘（D73）：`BarRepo.load` 给的 `close` 就是原始值，`adj_factor` 不参与。"""
        latest = self.bars.latest_date(market, code)
        if latest is None:
            return None, None
        frame = self.bars.load(market, code, latest, latest)
        if frame.is_empty():
            return None, None
        value = frame["close"][-1]
        return (None if value is None else float(value)), latest

    def _rates(self, account_id, pos, price_day) -> tuple[float | None, float | None, bool, bool, str | None]:
        """(基准汇率, 行情日汇率, 是否陈旧, 能否换算, 换算不出时的一句话)。

        汇率取**行情日或之前**最近的一条，不取「今天」：把换算锚在已经有行情的那一天，
        就不需要墙钟，也避开了用 UTC 日期推交易日在美股上恒错一天（`core/clock.py` 开头）。
        跨币种而换不出时 `convertible=False`：宁可不进汇总，也不要把一笔 USD 的数额
        当成 CNY 加进总市值——那是一个看起来完全正常的错数。
        """
        report = self.settings.report_currency
        if pos.currency == report:
            return None, None, False, True, None
        base = self._average_base_rate(account_id, pos)
        found = self.fx.latest(price_day, pos.currency, report)
        if found is None or base is None:
            missing = "汇率" if found is None else "买入当天的基准汇率"
            return base, None, True, False, (
                f"{pos.market} {pos.code} 以 {pos.currency} 计价，但取不到{missing}，"
                f"无法换算成 {report}：本持仓不计入汇总"
            )
        return base, found.rate, found.is_stale, True, None

    def _average_base_rate(self, account_id: str, pos) -> float | None:
        """这笔持仓的基准汇率 = 各笔买入 `fx_rate_ref` 按股数加权（§5.4 line 434）。"""
        total_qty = 0.0
        total = 0.0
        for order in self.orders.orders(account_id):
            if order.market != pos.market or order.code != pos.code:
                continue
            if order.side != "buy":
                continue
            if order.fx_rate_ref is None:
                return None  # 有一笔不知道基准汇率，整段拆分就不成立
            total_qty += order.qty
            total += order.qty * order.fx_rate_ref
        return None if total_qty <= 0 else total / total_qty
```

## Task T-9: `summary` —— 分币种小计、换汇单列、跨账户重复持仓

一个账户可能同时有 CNY/HKD/USD 三笔现金和两种币种的持仓。汇总有一种错法比「数字算错」
更隐蔽：把所有数额直接相加报成一个 CNY 数，看着完全正常。所以**先按币种分组给出本币小计，
再逐笔换算成报告币种**，`fx_pnl` 单独成行；换不出的那一笔不进合计但在 `notes` 里点名。

跨账户重复持仓（同一个用户在两个账户都买了 601088）单独提示：`sim_positions` 的主键是
`(account_id, market, code)`，库里不会冲突，但用户在两个地方看到同一只票两个成本时，
得有人告诉他那是两个账户而不是一只票坏了。

```python
@dataclass(frozen=True)
class AccountSummary:
    account_id: str
    name: str
    report_currency: str
    cash_by_currency: dict[str, float]
    position_value_by_currency: dict[str, float]
    #: 报告币种里的合计。缺现价/换不出的贡献 0，两者都在 notes 里点名。
    total_report: float
    price_pnl_report: float
    fx_pnl: float
    realized_report: float
    notes: tuple[str, ...]
    duplicate_holdings: tuple[str, ...]

    @property
    def total_pnl_report(self) -> float:
        return self.price_pnl_report + self.fx_pnl


class PortfolioService:
    def summary(self, user_id: str, account_id: str) -> AccountSummary:
        """账户汇总：本币小计分开、逐笔换算、换汇单列（§5.4 规则 2/3）。"""
        account = self.accounts.get(user_id, account_id)
        if account is None:
            raise ValidationError(f"账户 {account_id} 不存在或不属于用户 {user_id}")
        rows = self.positions(user_id, account_id)
        as_of = self._as_of(rows)
        notes: list[str] = []
        cash = self.accounts.cash_balances(account_id)
        value_by_ccy: dict[str, float] = {}
        price_report = 0.0
        fx_total = 0.0
        for row in rows:
            # 本币小计是「有几股、什么价」的事实，与能不能换算无关，所以先加再判换算
            value_by_ccy[row.currency] = value_by_ccy.get(row.currency, 0.0) + (
                row.qty * row.last_price if row.last_price is not None else row.qty * row.avg_cost
            )
            if row.pnl is None or not row.convertible:
                if row.unavailable_note:
                    notes.append(row.unavailable_note)
                continue
            price_report += (
                row.pnl.price_pnl
                if row.pnl.rate_base is None
                else row.pnl.price_pnl * row.pnl.rate_base
            )
            fx_total += row.pnl.fx_pnl
        cash_report = 0.0
        for ccy, amount in cash.items():
            if ccy == account.report_currency:
                cash_report += amount
                continue
            found = self.fx.latest(as_of, ccy, account.report_currency)
            if found is None:
                notes.append(
                    f"现金 {amount:.2f} {ccy} 换不出 {account.report_currency}"
                    f"（{as_of} 或之前取不到汇率），未计入合计"
                )
                continue
            cash_report += amount * found.rate
        positions_report = sum(
            row.pnl.market_report for row in rows if row.convertible and row.pnl is not None
        ) + cash_report
        return AccountSummary(
            account_id=account_id,
            name=account.name,
            report_currency=account.report_currency,
            cash_by_currency=cash,
            position_value_by_currency=value_by_ccy,
            total_report=positions_report,
            price_pnl_report=price_report,
            fx_pnl=fx_total,
            realized_report=self._realized_report(account, as_of),
            notes=tuple(notes),
            duplicate_holdings=self.duplicate_holdings(user_id, exclude_account=account_id),
        )

    def _as_of(self, rows: list[PositionRow]) -> date:
        """换算锚在哪一天：所有持仓行情日里最新的那个，没有持仓才退到日历今天。

        不读墙钟（见 T-8：用 UTC 日期推交易日，在美股上恒错一天）。
        """
        days = [r.last_price_date for r in rows if r.last_price_date is not None]
        return max(days) if days else date.today()

    def _realized_report(self, account, as_of: date) -> float:
        """已实现盈亏按持仓币种汇总后逐笔换算；换算口径与持仓一致。"""
        by_ccy: dict[str, float] = {}
        for pos in self.orders.positions(account.id):
            amount = self.orders.realized_pnl(account.id, pos.market, pos.code)
            if amount == 0.0:
                continue
            by_ccy[pos.currency] = by_ccy.get(pos.currency, 0.0) + amount
        out = 0.0
        for ccy, amount in by_ccy.items():
            if ccy == account.report_currency:
                out += amount
                continue
            found = self.fx.latest(as_of, ccy, account.report_currency)
            if found is not None:
                out += amount * found.rate
        return out

    def duplicate_holdings(
        self, user_id: str, *, exclude_account: str | None = None
    ) -> tuple[str, ...]:
        """同一用户在多个账户持有同一只票：那是两个账户，不是一只票的数据坏了。"""
        holders: dict[tuple[str, str], list[str]] = {}
        for account in self.accounts.list_for(user_id):
            if account.id == exclude_account:
                continue
            for pos in self.orders.positions(account.id):
                if pos.qty > 0:
                    holders.setdefault((pos.market, pos.code), []).append(account.name)
        return tuple(
            f"{market} {code} 同时在 {', '.join(sorted(names))} 里持有"
            for (market, code), names in sorted(holders.items())
            if len(names) > 1
        )
```

**两处签名与 T-8 的 `_rates` 要对上**：`convertible` 是 `PositionRow` 的字段（T-8 已加，
默认 `False`），跨币种而汇率缺失时 `_rates` 返回 `convertible=False` 并给出那句话；
`total_report` 在没有汇率时不参与汇总（`positions` 里那一行仍然出现，只是不进合计）。

```python
    )
```

```python
    add.add_argument("--qty", required=True, type=float)
    add.add_argument("--price", required=True, type=float)
    add.add_argument("--fee", type=float, default=0.0)
    add.add_argument("--thesis", help="挂到某个立论（必须已经走到 opened 之后）")

    add.add_argument("--fx-rate", type=float, help="标的币种→报告币种当日汇率")

def _moment(args: argparse.Namespace, container, market: str) -> datetime:
    """成交时刻：`--on` 那天市场本地 15:00（收盘附近）。

    时区的唯一出处是 `MarketCfg.zone()`（`core/config.py:28`），CLI 硬编码 UTC 会让
    A 股的一笔差 8 小时；而 `--on` 必填：记账要一个成交日，默认「今天」在跨时区
    补录时必错一天，且错的那一天屏幕上看不出来。
    """
    from sams.core.codes import canonical_market

    if args.on is None:
        raise ValidationError("--on 是必填的：记账要一个成交日（YYYY-MM-DD）")
    day = iso_date(args.on)
    zone = container.settings.market_cfg(canonical_market(market)).zone()
    return datetime(day.year, day.month, day.day, 15, 0, tzinfo=zone)
```

```python
"""汇率采集的契约与规整（spec §5.4 规则 2 的数据来源）。"""

from __future__ import annotations

from datetime import date

import polars as pl

from sams.core.errors import EmptyResponse

FX_COLUMNS = ("date", "base", "quote", "rate", "source")


def normalize_fx(raw: pl.DataFrame, *, base: str, quote: str, source: str = "unknown") -> pl.DataFrame:
    """校验方向、解析日期、补来源列。缺列/空框/方向反了都抛，不返回空框。"""
    missing = [c for c in ("date", "rate") if c not in raw.columns]
    if missing or raw.is_empty():
        raise EmptyResponse(f"汇率响应缺列或为空：{missing or '空框'}")
    rows = []
    for row in raw.to_dicts():
        got_base = str(row.get("base") or base)
        got_quote = str(row.get("quote") or quote)
        if (got_base, got_quote) != (base, quote):
            raise EmptyResponse(
                f"汇率方向不对：要 {base}→{quote}，源给的是 {got_base}→{got_quote}"
            )
        rows.append(
            {
                "date": date.fromisoformat(str(row["date"])[:10]),
                "base": base,
                "quote": quote,
                "rate": float(row["rate"]),
                "source": source,
            }
        )
    return pl.DataFrame(rows, schema={
        "date": pl.Date, "base": pl.Utf8, "quote": pl.Utf8,
        "rate": pl.Float64, "source": pl.Utf8,
    })
```

```python
class FxResult:
    """汇率链的返回。与 `FrameResult` 同一侧：0 行要说清是「源答了没有」还是「源没答」。"""

    base: str
    quote: str
    frame: pl.DataFrame
    source: str
    attempts: tuple[Attempt, ...] = ()
    skipped: tuple[tuple[str, str], ...] = ()
    confirmed_empty_by: tuple[str, ...] = ()

    @property
    def degraded(self) -> bool:
        return len(self.attempts) > 1

    def empty_everywhere(self) -> bool:
        """全链都答了「没有」，且一个都没答上——这才是可以落库的「确实没有」。"""
        return bool(self.confirmed_empty_by) and self.frame.is_empty()
```

```python
@dataclass(frozen=True)
class FxSyncSummary:
    """汇率同步的结果。不复用 `SyncSummary`：它的必填字段是 `market`/`code`，
    而外汇没有市场维度——把 `market=""` 塞进去会让每一处打印都多出两个空格
    和一句读不通的话。"""

    base: str
    quote: str
    source: str = ""
    fetched: int = 0
    written: int = 0
    degraded: bool = False
    empty: bool = False
    #: 最后写进去的那一天：`sams health` 用它判断汇率新不新
    latest_date: date | None = None
    attempts: tuple[Attempt, ...] = ()
    error: str | None = None


class FxSyncService:
    def __init__(self, registry: AdapterRegistry, fx: FxRepo) -> None:
        self.registry = registry
        self.fx = fx

    def sync(self, base: str, quote: str, start: date, end: date) -> FxSyncSummary:
        """汇率落进 `fx_rates`：跨币种持仓的 `fx_pnl` 全靠这张表有数。

        区间一次取（不在服务层按日循环）：Yahoo 的历史接口本来就给一段，
        按日循环会把一次限流风险变成三十次。
        """
        result = self.registry.fetch_fx(base, quote, start, end)
        if result.empty_everywhere:
            return FxSyncSummary(base=base, quote=quote, empty=True, attempts=result.attempts)
        written = 0
        for row in result.frame.to_dicts():
            self.fx.put(row["date"], base, quote, row["rate"], source=row["source"])
            written += 1
        return FxSyncSummary(
            base=base,
            quote=quote,
            source=result.source,
            fetched=result.frame.height,
            written=written,
            degraded=result.degraded,
            latest_date=(
                None if result.frame.is_empty() else result.frame["date"][-1]
            ),
            attempts=result.attempts,
        )
```
