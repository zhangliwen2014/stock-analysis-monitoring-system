# M3-B 通知层 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让一条规则命中能变成一条真实送达的告警：通道（企微/钉钉/PushPlus/邮件/桌面/语音）加密落库、按「规则指定 → 级别默认 → 用户默认」三级路由、免打扰时段压制中低级、每次都写 `notify_log`，并由 `sams channel` 与 `sams health [通知]` 可观测。

**Architecture:** 新增 `monitor` 层承载「发送」这件事本身，它只允许认识 `core` + `storage`——路由必须住在 monitor 而不是 services，因为 `services` 不许 import `monitor`，把路由放 services 等于给守卫开一个后门。通道凭据整份 JSON 用 Fernet 加密进 `channels.config_encrypted`，读明文只有一个入口。发送侧的退避复用采集层那一份（上提到 `core/retry.py`，全项目只有一种「重试几次、间隔多久」）。

**Tech Stack:** Python 3.13、DuckDB（新迁移 `0015`）、Polars（仅搬运，本层基本不碰）、`cryptography` Fernet、`requests`、`smtplib`、`subprocess`（osascript / say）、pytest（注入 `post`/`mail`/`run`/`env` 四个探针，全程离线）。

```python
"""退避重试：sleep 可注入，因此「429 之后等多久」不需要真等被测。

这里刻意**不**定义 `Clock`：`datasource/ratelimit.py` 的那个 `Clock` 是单调秒表（给
TokenBucket 记账用），而 `core/clock.py::Clock` 是 `now_utc()`。两个同名不同义的合同
放进同一个包，读的人要在两个文件之间跳一次才知道手上那个 clock 是哪种。退避只需要 sleep。
"""

from __future__ import annotations

import logging
import time
from collections.abc import Callable

logger = logging.getLogger("sams.retry")

Sleeper = Callable[[float], None]


def backoff_delays(attempts: int, *, base: float = 1.0, factor: float = 4.0) -> list[float]:
    """重试之间的等待序列：3 次尝试 → [1, 4]（第 3 次失败后不再等）。"""
    if attempts < 1:
        raise ValueError("attempts 至少为 1")
    return [base * factor**i for i in range(attempts - 1)]


def call_with_backoff(
    func: Callable[[], object],
    *,
    attempts: int = 3,
    base: float = 1.0,
    factor: float = 4.0,
    sleep: Sleeper = time.sleep,
    retryable: tuple[type[BaseException], ...],
    label: str = "",
) -> object:
    """退避重试；最后一次仍失败则原样抛出，由调用方决定降级还是上抛。

    `retryable` 必填且没有默认值：默认值属于谁家的异常元组，这个模块就认识了谁家的上层。
    """
    delays = backoff_delays(attempts, base=base, factor=factor)
    last: BaseException | None = None
    for index in range(attempts):
        try:
            return func()
        except retryable as exc:
            last = exc
            if index < len(delays):
                logger.warning(
                    "%s 第 %d 次失败（%s），%.1fs 后重试",
                    label or "调用",
                    index + 1,
                    type(exc).__name__,
                    delays[index],
                )
                sleep(delays[index])
    assert last is not None  # 循环必然至少失败一次才会走到这里
    raise last
```

```python
"""通道凭据的加密与掩码：全项目唯一知道「密文长什么样」的地方。

为什么整份 JSON 一个 token：逐字段加密看起来更「精细」，代价是字段名单变成存储层的知识
——加一个 `attempts` 字段就要记得把它加进「哪些字段要加密」，漏掉的那一个会以明文躺在库里。
一个 token 覆盖整份 config，就不存在「这个字段忘了」这件事。

`cipher_for` 是自由函数而不是 `Container` 上的方法：bootstrap 只需要
`cipher_for(settings)`，不该为了拿一个加密器去 import `services` 的某个部件——
组装根认识实现没错，但「认识多少」决定了下一个人能在哪里顺手接线。
"""

from __future__ import annotations

import base64
import hashlib
import json
from collections.abc import Mapping
from typing import Any

from cryptography.fernet import Fernet, InvalidToken

from sams.core.errors import ValidationError

#: 掩码时最多露出的尾部字符数。露尾巴是为了认得出「是哪一条」（两个 webhook 往往只差
#: 末尾几位），但短值露出来就可能直接可用，所以下面还有一条最低长度。
MASK_TAIL = 4
#: 低于这个长度的值**一个字符都不露**：一个 5 位的 token 露 4 位就是几乎全露。
MASK_MIN_CHARS = 8
MASK_PREFIX = "***"


def key_from_secret(secret_key: str) -> bytes:
    """`secret_key` → Fernet 要的 32 字节 urlsafe-base64 密钥。

    不直接 `secret_key.encode()`：用户的 key 可能是 8 个字符（短）或含非 base64 字符，
    Fernet 会在构造时抛 `ValueError`，那句话读起来像「库坏了」。sha256 折一次，
    长度与字符集就都稳定，而「换 secret_key ⇒ 换通道密钥」这条因果关系是直接的。
    """
    if not secret_key:
        raise ValidationError(
            "缺少 secret_key：通道凭据要加密，没有密钥就不能读也不能写"
            "（请设置环境变量 SAMS_SECRET_KEY）"
        )
    return base64.urlsafe_b64encode(hashlib.sha256(secret_key.encode()).digest())


class Cipher:
    """一份配置 ↔ 一个 token。`encrypt` 收 Mapping、`decrypt` 回 dict：调用方不需要知道 JSON。"""

    def __init__(self, secret_key: str) -> None:
        self._fernet = Fernet(key_from_secret(secret_key))

    def encrypt(self, config: Mapping[str, Any]) -> str:
        try:
            payload = json.dumps(dict(config), ensure_ascii=False, sort_keys=True)
        except (TypeError, ValueError) as exc:
            raise ValidationError(f"通道配置不能序列化：{exc}") from exc
        return self._fernet.encrypt(payload.encode()).decode()

    def decrypt(self, token: str) -> dict[str, Any]:
        try:
            raw = self._fernet.decrypt(token.encode())
        except InvalidToken as exc:
            # 这一句要说「密钥不对」而不是「数据坏了」：唯一常见的原因是 secret_key 换过，
            # 而换 key 之后旧凭据在数学上就读不回来——这不是可以用重试救的错
            raise ValidationError(
                "通道凭据解密失败：读它用的密钥与写入时不是同一把"
                "（换过 secret_key 就要重新 `sams channel add` 一次）"
            ) from exc
        loaded = json.loads(raw.decode())
        if not isinstance(loaded, dict):
            raise ValidationError("通道凭据解出来不是一个对象：库里那一行不是本模块写的形状")
        return loaded


def cipher_for(settings) -> Cipher:
    """从 `Settings` 建加密器。只读 `settings.secret_key`，不认任何其它字段。"""
    return Cipher(settings.secret_key)


def mask_secret(value: object) -> str:
    """掩码：短值完全不露，长值最多露 `MASK_TAIL` 个字符。

    露尾巴是为了认得出「这是哪个 webhook」——两家的地址常常只差末尾几位；
    而一个 5 位的 token 露 4 位就等于把凭据抄在屏幕上。所以短值一律不露。
    """
    if value is None:
        return ""
    text = str(value)
    if not text:
        return ""
    if len(text) < MASK_MIN_CHARS:
        return MASK_PREFIX
    return MASK_PREFIX + text[-MASK_TAIL:]


def mask_config(config: Mapping[str, Any]) -> dict[str, str]:
    """每个值都掩掉，键名原样保留：屏幕上要能看出「这条通道配了 url 和 token」，
    但不能看出任何一个值（§4.2：admin 只见启用状态，不见明文）。"""
```

```python
            )


def route_for(severity: str) -> tuple[str, ...]:

    if severity not in DEFAULT_ROUTE:

def format_headline(message: AlertMessage) -> str:
    """§11 那条标题行。六个通道共用同一句：同一件事在企微和邮件里长得不一样，
    读的人就要多花一次「这两条是不是同一条」的力气。"""
    prefix = _SEVERITY_PREFIX[message.severity]
    account = message.account_name.strip() or UNKNOWN_ACCOUNT
    name = message.name.strip() or UNKNOWN_NAME
    when = format_utc_moment(message.triggered_at)
    return (
        f"{prefix}[{message.market}][{account}] {name} {message.code}"
        f" | {message.rule_name or '未命名规则'}"
        f" | {message.value_text or '?'}/{message.threshold_text or '?'}"
        f" | {when} | {message.stage or '触发'}"
    )


def trim(text: str, *, limit: int) -> str:
    """按 UTF-8 **字节**剪到 `limit` 以内，末尾补 `…`；剪在半个汉字上就丢掉那半个。

    通道限制的是请求体字节数。按 `len(text)` 剪的话，一条 4000 个汉字的内容在这里
    是 4000「通过」，到企微那边是一个 4000+ 字节的 41003 错误——而错误发生在网络上，
    比在这里少发两个字贵得多。
    """
    encoded = text.encode()
    if len(encoded) <= limit:
        return text
    room = max(0, limit - len("…".encode()))
    kept = encoded[:room].decode(errors="ignore")
    return kept + "…"
```

```python
def format_utc_moment(value: datetime) -> str:
    """把任意时刻折到 UTC 后写成 `2026-10-20 14:30:00 UTC`。

    `TIMESTAMPTZ` 的回读在本库不稳定：有的表给 UTC，有的表给会话本地时区（Asia/Shanghai）。
    两者是同一个瞬间，按原样 `strftime` 会在同一屏上印出「立论 15:00 / 时间线 23:00」。
    到秒：两步落在同一分钟里时（脚本、M3 的触发器），分分辨不出先后。
    """
    if value.tzinfo is not None:
        value = value.astimezone(UTC)
    return f"{value:%Y-%m-%d %H:%M:%S} UTC"
```

```python
import os
import shutil
from collections.abc import Mapping
from dataclasses import dataclass

#: 判定「这是远程会话」的那几个变量。SSH_TTY 在没有图形转发的登录里也在，
#: 而它恰恰意味着「这里没有图形界面」。
_REMOTE_KEYS = ("SSH_TTY", "SSH_CONNECTION", "STY", "TMUX_PANE")

_TOOL_NAMES = ("osascript", "say", "notify-send", "open", "gnome-terminal")

#: 两条收尾建议：原因必须说到「那我该怎么办」，否则屏幕上是一句只有运维看得懂的诊断。
_DESKTOP_FIX = "本机桌面会话里才能弹；远程终端请改用手机侧通道（PushPlus/企微）。"
_VOICE_FIX = "需要一台装得下 macOS 且接着扬声器的机器；否则请把 strong 级也配一个手机侧通道。"


@dataclass(frozen=True)
class DesktopEnv:
    platform: str
    is_remote: bool
    has_display: bool
    osascript_path: str | None
    say_path: str | None
    notify_send_path: str | None
    terminal_launcher: str | None

    @classmethod
    def from_environment(
        cls,
        env: Mapping[str, str] | None = None,
        *,
        which: Mapping[str, str] | None = None,
        platform: str | None = None,
    ) -> DesktopEnv:
        """`which`/`platform` 传了就整个用它（测试），没传才走真实机器。

        参数叫 `which` 而不 `import from shutil import which`：`shutil` 整包引进来，
        回退那一行写 `shutil.which(name)`，免得在参数表里遮住同一个名字。
        """
        real = os.environ if env is None else env
        found = (
            which
            if which is not None
            else {name: path for name in _TOOL_NAMES if (path := shutil.which(name))}
        )
        return cls(
            platform=(platform or os.uname().sysname).lower(),
            is_remote=any(key in real for key in _REMOTE_KEYS),
            has_display=bool(real.get("DISPLAY") or real.get("WAYLAND_DISPLAY")),
            osascript_path=found.get("osascript"),
            say_path=found.get("say"),
            notify_send_path=found.get("notify-send"),
            terminal_launcher=found.get("open") or found.get("gnome-terminal"),
        )

    @property
    def desktop_available(self) -> bool:
        if self.is_remote:
            return False
        if self.platform == "darwin":
            return self.osascript_path is not None
        return self.notify_send_path is not None and self.has_display

    @property
    def voice_available(self) -> bool:
        return not self.is_remote and self.say_path is not None

    def desktop_reason(self) -> str:
        if self.desktop_available:
            return ""
        if self.is_remote:
            return _remote("桌面弹窗") + _DESKTOP_FIX
        if self.platform == "darwin":
            gap = [] if self.osascript_path else ["osascript"]
        else:
            gap = ([] if self.notify_send_path else ["notify-send"]) + (
                [] if self.has_display else ["DISPLAY"]
            )
        return _missing(gap, what="桌面弹窗") + _DESKTOP_FIX

    def voice_reason(self) -> str:
        if self.voice_available:
            return ""
        if self.is_remote:
            return _remote("语音朗读") + _VOICE_FIX
        # 念不出来的机器上就是没有 `say`：非 mac 没有等价物，这里不编第二条路径
        return _missing(["say"], what="语音朗读") + _VOICE_FIX


def _remote(what: str) -> str:
    return f"当前是远程会话（SSH/tmux），{what}没有落点。"


def _missing(names: list[str], *, what: str) -> str:
    return f"本机缺少 {'、'.join(names)}，{what}不可用。"
```

  两个形状上的选择写在这儿，因为它们会影响读代码的人怎么改它：`desktop_reason()` 里 mac/Linux 两支各只列自己那一条路（一支的名单里不出现另一支的二进制），所以「为什么我的 Linux 说缺 osascript」这种问题不可能发生；`_missing`/`_remote` 只拼句子，不做判断。

```python
class PostSpy:
    def __init__(self, responses=None):
        self.calls: list[tuple[str, dict, dict]] = []
        self.responses = list(responses or [{"errcode": 0, "errmsg": "ok"}])

    def __call__(self, url, body, headers):
        self.calls.append((url, body, headers))
        return self.responses[min(len(self.calls) - 1, len(self.responses) - 1)]


def deps(post=None):
    return NotifierDeps(
        post=post or PostSpy(),
        mail=lambda *a, **k: None,
        run=lambda *a, **k: None,
        env=DesktopEnv.from_environment({"PATH": "/x"}, which={}),

                account_name="主账户", rule_name="PR 低", value_text="PR=0.31",
                threshold_text="PR≤0.4", severity=severity, stage="入池",
                triggered_at=datetime.now(UTC), body="")
    base.update(over)
    return AlertMessage(**base)
```

```python
"""本机通道：`run` 注入件收的是 argv 列表，不是 shell 字符串。

拼接 shell 命令字符串等于把标的名称（来自外部数据源）送进一次 shell 解释：
一个名字叫 `x; rm -rf ~` 的票就是一次注入。argv 列表没有这个口（spec §12 的边界）。

from __future__ import annotations

from datetime import UTC, datetime

from sams.core.notify import AlertMessage
from sams.monitor.env import DesktopEnv
from sams.monitor.notifiers import NotifierDeps
from sams.monitor.notifiers.desktop import DesktopNotifier

class RunSpy:
    def __init__(self, returncode=0):
        self.calls: list[list[str]] = []
        self.returncode = returncode

    def __call__(self, argv, **kw):
        self.calls.append(list(argv))
        import subprocess

        return subprocess.CompletedProcess(argv, self.returncode, stdout="", stderr="")


def msg(**over):
    base = dict(alert_id="a", market="A", code="601088", name="中国神华",
                account_name="主账户", rule_name="PR 低", value_text="PR=0.31",
                threshold_text="PR≤0.4", severity="strong", stage="入池",
                triggered_at=datetime.now(UTC), body="")
    base.update(over)
    return AlertMessage(**base)


def env_with(tools, *, platform="darwin"):

    return DesktopEnv.from_environment(
        {"PATH": "/usr/bin", **tools[1]}, which=tools[0], platform=platform
```

```python
@dataclass(frozen=True)
class DeliveryReport:
    kind: str
    channel_id: str | None
    outcome: str
    error: str
    attempts: int

    @property
    def ok(self) -> bool: ...

@dataclass(frozen=True)
class DispatchResult:
    alert_id: str
    deliveries: tuple[DeliveryReport, ...]

    @property
    def any_sent(self) -> bool: ...
    @property
    def failed(self) -> tuple[DeliveryReport, ...]: ...

class NotifyService:
    def __init__(self, channels, alerts, users, *, settings, bus, clock,
                 factory=None, sleep=time.sleep, attempts=3, env=None) -> None: ...
    def dispatch(self, message: AlertMessage, *, user_id: str,
                 channel_ids: Sequence[str] = (), rule_id: str | None = None,
                 account_id: str | None = None) -> DispatchResult: ...
    def probe(self, channel_id: str, *, user_id: str | None = None) -> DeliveryReport: ...
    def local_status(self) -> dict[str, str]: ...
    def add(self, user_id: str, kind: str, name: str, config: Mapping) -> Channel: ...
    def list(self, user_id: str) -> list[Channel]: ...
    def show(self, channel_id: str, *, user_id: str | None = None) -> Channel: ...
    def set_enabled(self, channel_id: str, enabled: bool, *, user_id: str | None = None) -> None: ...
    def set_default(self, channel_id: str, *, user_id: str | None = None) -> None: ...
    def remove(self, channel_id: str, *, user_id: str | None = None) -> None: ...
    def set_dnd(self, user_id: str, *, start, end, tz) -> None: ...
    def clear_dnd(self, user_id: str) -> None: ...
    def _in_dnd(self, user: User, moment: datetime) -> bool: ...
```

  七条实现约束（附录 H 逐条落在代码里，这里先说为什么）：
  1. **`dispatch` 只写 `notify_log`，绝不写 `alerts.status`**（上面那枚用例）。
  2. **路由顺序**：`channel_ids`（规则指定）→ `DEFAULT_ROUTE[severity]` 里**该用户启用的**那些 kind → 用户的 `is_default` 通道 → `NO_CHANNEL`。第三级是「级别默认一个都没启用」时的兜底，`NO_CHANNEL` 那行的 `error` 必须写三步都能做什么（没配 / 都停用 / kind 不认识），因为这一行是用户看到的第一句「为什么我没收到」。
  3. **DND 只挡 `mid`/`weak`**。判据：`user.dnd_start` 三者齐备 ⇒ 把 `moment` 折到 `dnd_tz` ⇒ 跨午夜窗口用 `start <= t or t <= end`，非跨午夜用 `start <= t <= end`。**不新增事件、不新增列**：压制也写 `notify_log`（`channel_id=None`）。
  4. **重试在通道内**：`backoff_delays(self.attempts)` 配一个自己的循环（**不用** `call_with_backoff`，因为 `Notifier.send` 从不抛异常——它把异常包成 `NotifyResult(ok=False)`，而 `call_with_backoff` 只能靠异常决定重试）。退避的**秒数序列**仍然复用 `core/retry`，这是 T-1 上提它的唯一理由。`NotifyResult.ok=False` 重试到 `attempts` 次；`unavailable=True` **立即停止且不记失败**（本机做不到这件事，重试三次不会改变结论）。这两句的区别必须写在代码里，因为它们的症状都是「没送到」，而一个是环境一个是通道。至于 `ok=False` 为什么敢重试三次：对端给的「不」分不清「此刻不」（网关 5xx）与「永远不」（凭据作废），两者都只回一个 errcode；间隔 1s、4s 再问一次对一条一天最多几发的告警不算骚扰，而漏掉一条真告警的代价是反向的。
  5. **不可用 ⇒ `SUPPRESSED`**（`NotifyResult.unavailable`），且**不调 `mark_error`**：一个没有桌面的机器上的 desktop 通道，不该在健康页上被算成失败率。
  6. **成功 ⇒ `mark_ok`；真失败 ⇒ `mark_error` + `bus.publish(NotifyFailed(...))`**。`SystemAlarm` 不在这里发——§12 那句「连续失败降系统告警」的计数属于调度器（M3-D），router 只保证每一次失败都有一个订阅者能看见。
  7. **`alerts.create` 由 `dispatch` 负责**：先有 alert 才有投递（`notify_log.alert_id` NOT NULL），所以 `dispatch` 是「记录 + 发送」这一个动作，而 `AlertRepo.create` 单独可用是为了 M3-C 想在发送前先落一条历史。

## 附录 A：`src/sams/storage/repos/channels.py`（全文）

```python
"""通道仓库：写侧加密、读侧掩码，明文只有一个出口。

`credentials()` 之所以是这里唯一能返回明文的函数，不是因为别的函数「忘了」解密——
是因为把它们都写成解密的话，屏幕上迟早会出现一个 token。§4.2 那句话（admin 只见启用状态）
要能被机器复核，就得让「取明文」这件事在代码里只有一个形状。
"""

from __future__ import annotations

import logging
from collections.abc import Mapping, Sequence
from dataclasses import dataclass
from datetime import datetime

from sams.core.clock import SYSTEM_CLOCK, Clock
from sams.core.errors import StorageError, ValidationError
from sams.storage.crypto import Cipher, mask_config
from sams.storage.db import Database
from sams.storage.ids import new_id

logger = logging.getLogger("sams.storage.channels")

#: 屏幕上那一列的宽度上限：`name` 没有长度上限会让 `channel list` 折行，
#: 折行的表格读不出「哪一行是默认」
NAME_MAX_CHARS = 40
_KIND_MAX_CHARS = 32
_ERROR_MAX_CHARS = 400

_COLUMNS = (
    "id, user_id, kind, name, enabled, is_default,"
    " last_ok_at, last_error, last_error_at, config_encrypted"
)


@dataclass(frozen=True)
class Channel:
    """一行通道。`config` 永远是**掩码视图**，明文在 `config_encrypted` 里、只有
    `ChannelRepo.credentials()` 能取。"""

    id: str
    user_id: str
    kind: str
    name: str
    enabled: bool
    is_default: bool
    last_ok_at: datetime | None
    last_error: str
    last_error_at: datetime | None
    config: dict[str, str]


def _require_name(name: str) -> str:
    trimmed = (name or "").strip()
    if not trimmed:
        raise ValidationError(
            "通道必须有名字：`sams channel list` 里两行同 kind 的通道是不可区分的，"
            "而规则上存的正是 id"
        )
    if len(trimmed) > NAME_MAX_CHARS:
        raise ValidationError(f"通道名字请控制在 {NAME_MAX_CHARS} 字以内（收到 {len(trimmed)}）")
    return trimmed


def _require_kind(kind: str) -> str:
    """仓库层**不**认识通道的名单（那是 `monitor.notifiers` 的事，storage 不许 import monitor）。

    这里只挡空与超长；「不认识这个 kind」由 `sams channel add` 在工厂那一层报错，
    而健康页会数出库里那些工厂不认识的 kind（T-17 的 `unknown_kinds`）。
    """
    trimmed = (kind or "").strip().lower()
    if not trimmed:
        raise ValidationError("通道类型不能为空：wecom / dingtalk / pushplus / email / desktop / voice")
    if len(trimmed) > _KIND_MAX_CHARS:
        raise ValidationError(f"通道类型过长（{len(trimmed)} 字）")
    return trimmed


class ChannelRepo:
    def __init__(self, db: Database, *, cipher: Cipher, clock: Clock = SYSTEM_CLOCK) -> None:
        self.db = db
        self.cipher = cipher
        #: 时间戳来自构造函数而不是每个方法的参数：漏传一次的那一处会写 `now()`，
        #: 而假时钟测不出「这个时间戳是谁打的」（与 `SessionRepo(db, clock)` 同一理由）
        self.clock = clock

    def create(
        self, user_id: str, kind: str, name: str, config: Mapping[str, object]
    ) -> Channel:
        if not user_id:
            raise ValidationError("通道必须属于一个用户：跨用户共用凭据没有审计口径")
        channel_kind = _require_kind(kind)
        label = _require_name(name)
        if not isinstance(config, Mapping) or not dict(config):
            raise ValidationError(
                f"通道 {label} 没有配置：至少要有它需要的字段"
                "（`--config url=https://…`，一个 k=v 一个参数）"
            )
        channel_id = new_id()
        self.db.execute(
            "INSERT INTO channels"
            " (id, user_id, kind, name, config_encrypted, enabled, is_default, created_at)"
            " VALUES (?, ?, ?, ?, ?, TRUE, ?, ?)",
            [
                channel_id,
                user_id,
                channel_kind,
                label,
                self.cipher.encrypt(config),
                # 这个用户的第一个通道自动成为默认：没有默认的那份库，「级别默认也没启用」
                # 这一条兜底路由永远走不到，而用户只会觉得「配了却不发」
                not self._any_for(user_id),
                self.clock.now_utc(),
            ],
        )
        created = self.show(channel_id)
        assert created is not None  # 同事务刚写完
        return created

    def show(self, channel_id: str, *, user_id: str | None = None) -> Channel | None:
        clauses, params = ["id = ?"], [channel_id]
        if user_id is not None:
            clauses.append("user_id = ?")
            params.append(user_id)
        rows = self.db.query(
            f"SELECT {_COLUMNS} FROM channels WHERE {' AND '.join(clauses)}", params
        )
        return self._to_channel(rows.to_dicts()[0]) if rows.height else None

    def list_for(
        self, user_id: str, *, kinds: Sequence[str] | None = None
    ) -> list[Channel]:
        sql = f"SELECT {_COLUMNS} FROM channels WHERE user_id = ?"
        params: list[object] = [user_id]
        if kinds:
            sql += f" AND kind IN ({', '.join('?' for _ in kinds)})"
            params.extend(list(kinds))
        # 默认的排前面：屏幕上第一眼要看的是「兜底会走哪条」
        sql += " ORDER BY is_default DESC, name"
        return [self._to_channel(row) for row in self.db.query(sql, params).to_dicts()]

    def by_kind(self, user_id: str, kind: str) -> list[Channel]:
        return [
            row
            for row in self.list_for(user_id, kinds=[_require_kind(kind)])
            if row.enabled
        ]

    def credentials(self, channel_id: str) -> dict[str, object]:
        """明文出口。只给 router 用：它要拿 url/token 去真的发一次。

        不带 `user_id` 参数不是漏洞：归属在路由那一步判（这条通道属不属于这次告警的
        接收人）。把它加在这里等于允许任何持有 repo 的人用「猜 id + 传别人的 user_id」
        去试探，而试探的结果与「不存在」在屏幕上是一样的。
        """
        token = self.db.scalar("SELECT config_encrypted FROM channels WHERE id = ?", [channel_id])
        if not token:
            raise StorageError(f"通道 {channel_id} 不存在（拿不到凭据也就不能发）")
        return self.cipher.decrypt(token)

    def set_enabled(self, channel_id: str, enabled: bool) -> None:
        node = self._require(channel_id)
        if not enabled and node.is_default:
            raise ValidationError(
                f"通道 {node.name} 是默认通道，不能停用：默认是「级别没配到通道」时的兜底，"
                "先把默认移给别的一条（`sams channel default`）再停用它"
            )
        # 重新启用即清掉上次失败的那句话：留着它，屏幕上会在一个已经恢复的通道旁边
        # 挂一条旧错误，而「这个通道现在能用吗」的答案就被那句噪音盖住。
        # 停用**不**清：`last_error` 是「为什么这条现在不发」的唯一答案。
        if enabled:
            self.db.execute(
                "UPDATE channels SET enabled = TRUE, last_error = NULL, last_error_at = NULL"
                " WHERE id = ?",
                [channel_id],
            )
        else:
            self.db.execute("UPDATE channels SET enabled = FALSE WHERE id = ?", [channel_id])

    def set_default(self, channel_id: str) -> None:
        node = self._require(channel_id)
        if not node.enabled:
            raise ValidationError(
                f"通道 {node.name} 处于停用状态：默认必须是一条能发出去的通道"
            )
        with self.db.transaction():
            # 两条 UPDATE 必须在同一个事务里：中间断电的那一瞬库里没有默认通道，
            # 而「没有默认」会让兜底路由静默走到 no_channel
            self.db.execute(
                "UPDATE channels SET is_default = FALSE WHERE user_id = ? AND is_default",
                [node.user_id],
            )
            self.db.execute(
                "UPDATE channels SET is_default = TRUE WHERE id = ?", [channel_id]
            )

    def remove(self, channel_id: str) -> None:
        node = self._require(channel_id)
        with self.db.transaction():
            self.db.execute("DELETE FROM channels WHERE id = ?", [channel_id])
            if node.is_default:
                successor = self.db.scalar(
                    "SELECT id FROM channels WHERE user_id = ? AND enabled"
                    " ORDER BY name LIMIT 1",
                    [node.user_id],
                )
                if successor:
                    self.db.execute(
                        "UPDATE channels SET is_default = TRUE WHERE id = ?", [successor]
                    )
                    logger.info("默认通道 %s 已删除，移给 %s", node.name, successor)

    def mark_ok(self, channel_id: str) -> None:
        self._require(channel_id)
        self.db.execute(
            "UPDATE channels SET last_ok_at = ?, last_error = NULL, last_error_at = NULL"
            " WHERE id = ?",
            [self.clock.now_utc(), channel_id],
        )

    def mark_error(self, channel_id: str, *, error: str) -> None:
        self._require(channel_id)
        note = (error or "").strip()[:_ERROR_MAX_CHARS]
        self.db.execute(
            "UPDATE channels SET last_error = ?, last_error_at = ? WHERE id = ?",
            [note or "（对端没有给错误信息）", self.clock.now_utc(), channel_id],
        )

    def _any_for(self, user_id: str) -> bool:
        return bool(
            self.db.scalar("SELECT count(*) FROM channels WHERE user_id = ?", [user_id])
        )

    def _require(self, channel_id: str) -> Channel:
        node = self.show(channel_id)
        if node is None:
            raise StorageError(f"通道 {channel_id} 不存在")
        return node

    def _to_channel(self, row: Mapping[str, object]) -> Channel:
        token = row["config_encrypted"]
        try:
            masked = mask_config(self.cipher.decrypt(token) if token else {})
        except ValidationError as exc:
            # 库里这一行读不回来（多半是 secret_key 换过）。这里**不**把名字读成空：
            # 屏幕上要能列出这条通道并让用户删掉它，而不是让它凭空消失
            logger.warning("通道 %s 的凭据读不回来：%s", row["id"], exc)
            masked = {"（凭据读不回来）": str(exc)}
        return Channel(
            id=str(row["id"]),
            user_id=str(row["user_id"]),
            kind=str(row["kind"]),
            name=str(row["name"] or ""),
            enabled=bool(row["enabled"]),
            is_default=bool(row["is_default"]),
            last_ok_at=row["last_ok_at"],
            last_error=str(row["last_error"] or ""),
            last_error_at=row["last_error_at"],
            config=masked,
        )
```

  两处与库的形状有关的选择，已经在 T-3 与上面的代码里落定，不是留给执行时的悬念：
  1. `channels.created_at` 由 0015 追加（`ALTER TABLE channels ADD COLUMN IF NOT EXISTS created_at TIMESTAMPTZ`），所以上面的 `INSERT` 带着它；理由写在 0015 的注释里——`sams channel list` 要能回答「这条通道是刚配的还是三个月没动过的」。`tests/storage/test_migration_0015.py` 的名单断言里也有这一列。
  2. `set_enabled` 重新启用时清掉 `last_error`，停用时**不**清：清空只对「曾经坏掉、现在好了」有意义，而停用的那条通道屏幕上正需要那句「它上次为什么失败」。两个分支各一条 SQL（上面 Step 3 的实现里就是这两条），不写成一条带 `if enabled else` 的语句——那个形状的两个分支参数列表一模一样，读的人要停下来想「这里是不是本来该不一样」。

```python
def _validate_dnd(
    start: time | None, end: time | None, tz: str | None
) -> tuple[time, time, str]:
    """三列同存同空；全空是「清除」，半空是错误。

    跨午夜（23:00→07:00）是合法且最常见的用法，所以这里**不**要求 `start < end`。
    """
    given = (start is not None, end is not None, bool(tz))
    if not any(given):
        raise ValidationError("免打扰要三个一起给：--start/--end/--tz（清除用 `dnd clear`）")
    if not all(given):
        raise ValidationError(
            "免打扰的起止时间与时区要三个一起给：只有开始没有结束意思是「从今天起永远不打扰」，"
            "而这不是任何人的本意"
        )
    try:
        ZoneInfo(str(tz))
    except (ZoneInfoNotFoundError, ValueError) as exc:
        raise ValidationError(f"免打扰时区非法: {tz}") from exc
    return start, end, str(tz).strip()  # type: ignore[return-value]


class UserRepo:

            return
        window_start, window_end, zone = _validate_dnd(start, end, tz)
        self.db.execute(
            "UPDATE users SET dnd_start = ?, dnd_end = ?, dnd_tz = ? WHERE id = ?",
            [window_start, window_end, zone, user_id],

        """三个列名写在一条语句里：漏掉一个的症状是用户在屏幕上看到「已清除」，
        而库里还剩一列——下一次路由读到那个残留的窗口。"""
        self.db.execute(
```

```python
    时区解释裸值，那等于把北京 22:00 的一条止损记成 UTC 22:00。"""

            payload = {}
        return Alert(
            id=str(row["id"]),
            user_id=str(row["user_id"]),
            rule_id=row["rule_id"],
            account_id=row["account_id"],
            market=str(row["market"] or ""),

            symbol_name=str(payload.get("symbol_name") or ""),
            #: `alerts.severity` 那一列是权威，payload 里的那份只是冗余副本：
            #: 两者不一致时以列为准，因为列是 0015 之后新写的、payload 是老数据里搬来的

        )
```

```python
def build_notifier_factory(
    *, deps: NotifierDeps | None = None
) -> Callable[[str, Mapping[str, object]], Notifier]:
    """`kind` → 通道实现。名单只有一个出处（`core.notify.KINDS`）。"""

    active = deps if deps is not None else default_deps()

    def build(kind: str, config: Mapping[str, object]) -> Notifier:
        name = (kind or "").strip().lower()
        if name not in KINDS:
            raise ValidationError(
                f"不认识通道类型 {kind!r}：本期只有 {list(KINDS)}；"
                "库里若已有别的 kind，那是手工写入或旧版本的遗留，请用 `sams channel remove` 清掉"
            )
        # 延迟 import：六个模块各自 import 一次代价很小，而在函数体里 import 才能让
        # 「desktop 不可用时不加载 osascript 相关代码」这件事成立
        from sams.monitor.notifiers import dingtalk, desktop, email, pushplus, voice, wecom

        table = {
            "wecom": wecom.WecomNotifier,
            "dingtalk": dingtalk.DingTalkNotifier,
            "pushplus": pushplus.PushPlusNotifier,
            "email": email.EmailNotifier,
            "desktop": desktop.DesktopNotifier,
            "voice": voice.VoiceNotifier,
        }
        return table[name](config, deps=active)

    return build
```

```python
签名的失败通常是时间戳漂移（服务器钟不准 → errcode=310000），那句话要写进 error 里，
因为「机器人被限流」与「这台机器钟不对」的修法完全不同。
"""

from __future__ import annotations

import base64
import hashlib
import hmac
import time
import urllib.parse
from collections.abc import Mapping
from typing import Any

from sams.core.errors import ValidationError
from sams.core.notify import LIMITS, AlertMessage, NotifyResult, format_headline, trim
from sams.monitor.notifiers import NotifierDeps


def signed_url(url: str, secret: str, *, now_s: float | None = None) -> str:
    timestamp = int((time.time() if now_s is None else now_s) * 1000)
    raw = f"{timestamp}\n{secret}".encode()
    digest = hmac.new(secret.encode(), raw, hashlib.sha256).digest()
    sign = urllib.parse.quote_plus(base64.b64encode(digest))
    joiner = "&" if "?" in url else "?"
    return f"{url}{joiner}timestamp={timestamp}&sign={sign}"


def build_body(message: AlertMessage) -> dict[str, Any]:
    headline = format_headline(message)
    body = f"{headline}\n\n{message.body}" if message.body else headline
    return {
        "msgtype": "markdown",
        "markdown": {"title": trim(headline, limit=60), "text": trim(body, limit=LIMITS["dingtalk"])},
    }

        self.secret = str(config.get("secret") or "").strip()

    def available(self) -> bool:
        return True

    def unavailable_reason(self) -> str:

        code = reply.get("errcode") if reply else None
```

## 附录 F：`src/sams/monitor/notifiers/email.py`

```python
"""SMTP 通道：`smtplib` 在全项目的唯一落点（T-2 的 GATED 就指着这个文件）。

授权码（QQ/163 的「SMTP 密码」）与登录失败的原因都不许出现在 `NotifyResult.error` 里：
那句话会被写进 `notify_log.error`、被 `sams alerts` 打出来、也可能被贴进截图。
"""

from __future__ import annotations

import smtplib
from collections.abc import Mapping
from email.message import EmailMessage
from typing import Any

from sams.core.errors import ValidationError
from sams.core.notify import LIMITS, AlertMessage, NotifyResult, format_headline, trim
from sams.monitor.notifiers import NotifierDeps


def _require(config: Mapping[str, Any]) -> tuple[str, int, str, str, str, list[str], bool]:
    host = str(config.get("host") or "").strip()
    if not host:
        raise ValidationError("邮件通道需要 --config host=smtp.example.com")
    try:
        port = int(config.get("port") or 465)
    except (TypeError, ValueError) as exc:
        raise ValidationError(f"邮件通道端口非法: {config.get('port')}") from exc
    user = str(config.get("user") or "").strip()
    password = str(config.get("password") or "").strip()
    if not user or not password:
        raise ValidationError(
            "邮件通道需要 --config user=… --config password=…："
            "这里的 password 是服务商给的**授权码**，不是登录密码"
        )
    sender = str(config.get("from") or user).strip()
    raw = config.get("to")
    recipients = (
        [str(item).strip() for item in raw if str(item).strip()]
        if isinstance(raw, (list, tuple))
        else [str(item).strip() for item in str(raw or "").split(",") if item.strip()]
    )
    if not recipients:
        raise ValidationError("邮件通道需要 --config to=收件人（逗号分隔或重复给）")
    use_ssl = bool(config.get("ssl", port == 465))
    return host, port, user, password, sender, recipients, use_ssl


class EmailNotifier:
    name = "email"

    def __init__(self, config: Mapping[str, Any], *, deps: NotifierDeps) -> None:
        self.config = dict(config)
        (
            self.host,
            self.port,
            self.user,
            self.password,
            self.sender,
            self.recipients,
            self.use_ssl,
        ) = _require(config)
        self._deps = deps

    def available(self) -> bool:
        return True

    def unavailable_reason(self) -> str:
        return ""

    def render(self, message: AlertMessage) -> tuple[str, str]:
        headline = format_headline(message)
        body = message.body or headline
        # 主题就是标题行：收件箱列表里只显示主题，正文要等点开
        return trim(headline, limit=120), trim(body, limit=LIMITS["email"])

    def send(self, message: AlertMessage) -> NotifyResult:
        subject, body = self.render(message)
        try:
            self._deps.mail(self.config, list(self.recipients), subject, body)
        except Exception as exc:  # noqa: BLE001
            return NotifyResult(ok=False, error=_describe(exc))
        return NotifyResult(ok=True)


def _describe(exc: BaseException) -> str:
    """把 SMTP 的异常说成人话，且**不带上密码**。

    `smtplib` 的 `SMTPAuthenticationError` 会把「用户名 密码」回显在 message 里，
    直接 `str(exc)` 就等于把授权码写进库里的一列文本。
    """
    if isinstance(exc, smtplib.SMTPAuthenticationError):
        return "SMTP 登录被拒绝：授权码或账号不对（服务商通常要求用**授权码**而不是登录密码）"
    if isinstance(exc, smtplib.SMTPRecipientsRefused):
        return f"收件人被拒: {exc.recipients}"
    # 其余 SMTP 异常与网络异常同一句话：类型名 + 截断原文。截断是因为
    # `OSError` 的原文会带上完整 URL 与对端响应体，那比诊断有用更占屏幕。
    return f"{type(exc).__name__}: {str(exc)[:200]}"


def send_mail(
    config: Mapping[str, Any], to: list[str], subject: str, body: str
) -> None:
    """真实的 `NotifierDeps.mail`：465 走 SMTP_SSL，587 走 STARTTLS。

    两条都必须存在：只写一条的话，另一家服务商的配置会在屏幕上变成一个
    「连接被重置」，而那句错指向网络、真正的原因是没有 STARTTLS。
    """
    host = str(config["host"])
    port = int(config.get("port") or 465)
    user = str(config["user"])
    password = str(config["password"])
    sender = str(config.get("from") or user)
    use_ssl = bool(config.get("ssl", port == 465))

    letter = EmailMessage()
    letter["From"] = sender
    letter["To"] = ", ".join(to)
    letter["Subject"] = subject
    letter.set_content(body)

    if use_ssl:
        with smtplib.SMTP_SSL(host, port, timeout=20) as server:
            server.login(user, password)
            server.send_message(letter)
        return
    with smtplib.SMTP(host, port, timeout=20) as server:
        server.ehlo()
        server.starttls()
        server.ehlo()
        server.login(user, password)
        server.send_message(letter)
```

  `_describe` 只靠**分支的措辞**防泄漏：`SMTPAuthenticationError` 那一支不回显原文（服务商有时把那一行请求吐进 message），所以授权码不可能落进 `notify_log.error`。不做「从原文里把密码挖出来替换」那种事——`smtplib` 的异常对象没有 `password` 属性，写一段 `getattr(exc, "password")` 是在防一个不存在的形状，只会给下一个人「这里处理过密码」的错觉。

```python
argv 是列表而不是字符串，因为标的名称来自外部数据源：一个叫 `x; rm -rf ~` 的票名

from typing import Any

def escape(text: str) -> str:
    """AppleScript 的字面量转义：反斜杠与双引号。

    转义是为了**保留**内容而不是过滤掉它：被剥掉分号的标题行会让用户读到一个
    与告警当时不同的话。
    """
    return text.replace("\\", "\\\\").replace('"', '\\"')


class DesktopNotifier:
    name = "desktop"

    def __init__(self, config: Mapping[str, Any], *, deps: NotifierDeps) -> None:
        self._deps = deps
        self._env = deps.env

    def available(self) -> bool:
        return bool(self._env and self._env.desktop_available)

    def unavailable_reason(self) -> str:
        return self._env.desktop_reason() if self._env else "这一层没有拿到本机环境"

    def send(self, message: AlertMessage) -> NotifyResult:
        if not self.available():
            return NotifyResult(ok=False, error=self.unavailable_reason(), unavailable=True)
        text = trim(format_headline(message), limit=LIMITS["desktop"])

        if env and env.platform == "darwin":

                env.osascript_path or "osascript",
                "-e",
                f'display notification "{escape(text)}" with title "sams 告警"',
            ]
        else:
            argv = [env.notify_send_path or "notify-send", "sams 告警", text]  # type: ignore[assignment]
        return _run_local(self._deps.run, argv)


def _run_local(run, argv: list[str]) -> NotifyResult:
    try:
        done = run(argv)
    except Exception as exc:  # noqa: BLE001 — 协议不许网络/进程异常越界
        return NotifyResult(ok=False, error=f"{type(exc).__name__}: {exc}", unavailable=True)
    if getattr(done, "returncode", 0) != 0:
        return NotifyResult(
            ok=False, error=f"{argv[0]} 退出码 {done.returncode}：{(done.stderr or '').strip()[:200]}"
        )
    return NotifyResult(ok=True)
```

## 附录 H：`src/sams/monitor/router.py`（全文）

> 这一层有三个形状容易被前文的简写带偏，在这里钉死（代码已经按这三条写，读到不一致时以本节为准）：
> 1. **`NotifyService` 不接 `cipher`**：加解密与掩码是 `ChannelRepo` 的事，router 拿到的一直是密文/掩码视图。所以 T-14 的构造点里没有 `cipher=`，别加回去——留一个用不上的注入件比少一个更糟，下一个人会以为 router 需要明文。
> 2. **`suppressed` 允许带 `channel_id`**：DND 压制发生在选路之前，那一行没有通道（`None`）；而「这条 voice 通道在这台机器上发不出去」是**关于那条通道**的事实，它的 `channel_id` 要留着，否则屏幕上看不出是哪个通道被压制。附录 C 的 `record` 只拦 `no_channel` 带 id 与 `sent`/`failed` 不带 id 这两种矛盾，`suppressed` 的两种形状都放行。健康页的失败率只按 `outcome='failed'` 算，所以带 id 的 suppressed 行不会污染它。
> 3. **服务层只认 `user_id`，不认用户名**：CLI 先用 `resolve_actor` 把用户名换成 id，两层都认一遍的结果是一个人传名字、一个人传 id。位置参数给谁是有规则的：`dispatch`/`set_dnd`/`clear_dnd` 的那一个动作只围着用户转，`user_id` 排第一；`show`/`set_enabled`/`remove` 这些围着通道转的，位置留给 `channel_id`，`user_id` 作关键字（`None` = 不按归属收窄，只有 `probe`/`show` 用到）。

> 关于 `core/retry`：router 只用 `backoff_delays` + 自己循环，**不用 `call_with_backoff`**。原因是 `Notifier.send` 按协议不抛异常（附录 E/F/G 里每家都 `except Exception` 收进 `NotifyResult`），要重试的对象是「结果」而不是「异常」。`call_with_backoff` 仍与 `backoff_delays` 同住 `core/retry.py`：那串 `[1, 4]` 的秒数是这两个函数共享的知识，把它们分在两个模块里，改一处就会让「按异常重试」与「按结果重试」走出两种节奏。

```python
"""发送路由：三级选路、免打扰这道闸、每一次投递都落 `notify_log`（spec §11）。

住在 `monitor` 而不是 `services`：`services` 不许 import `monitor`（分层守卫），
把路由放 services 就等于要么放宽守卫、要么让通道实现反向 import 服务层——两条路都会
把「谁认识谁」变成一个问题。

这一层的每一条拒绝都要留下一行 `notify_log`。一句只打在屏幕上的拒绝，在用户关掉终端
之后就消失了，而「我那天为什么没收到」正是几天后才要问的那句话（§4.3）。
"""

from __future__ import annotations

import logging
import time
from collections.abc import Mapping, Sequence
from dataclasses import dataclass, replace
from datetime import UTC, datetime, time as dtime
from zoneinfo import ZoneInfo

from sams.core.errors import SamsError, ValidationError
from sams.core.events import EventBus, NotifyFailed
from sams.core.notify import (
    FAILED,
    NO_CHANNEL,
    SENT,
    STRONG,
    SUPPRESSED,
    AlertMessage,
    NotifyResult,
    route_for,
)
from sams.core.retry import Sleeper, backoff_delays
from sams.monitor.env import DesktopEnv
from sams.monitor.notifiers import Notifier, NotifierDeps, build_notifier_factory, probe_message
from sams.storage.repos.alerts import AlertRepo
from sams.storage.repos.channels import Channel, ChannelRepo
from sams.storage.repos.users import User, UserRepo

logger = logging.getLogger("sams.monitor.router")

#: 免打扰的屏幕上那句措辞（要包含「免打扰」四个字：T-14 的用例与 §11 的说明都引用它）
DND_WORD = "免打扰"


@dataclass(frozen=True)
class DeliveryReport:
    """一次投递的人话结论。与 `AlertRepo.Delivery`（库里那一行）不同：这里带着 `kind`，
    而 kind 不在 `notify_log` 里——它在 `channels` 那一行，读的时候才并起来。"""

    kind: str
    channel_id: str | None
    outcome: str
    error: str
    attempts: int

    @property
    def ok(self) -> bool:
        return self.outcome == SENT


@dataclass(frozen=True)
class DispatchResult:
    alert_id: str
    deliveries: tuple[DeliveryReport, ...]

    @property
    def any_sent(self) -> bool:
        return any(item.outcome == SENT for item in self.deliveries)

    @property
    def failed(self) -> tuple[DeliveryReport, ...]:
        return tuple(item for item in self.deliveries if item.outcome == FAILED)


class NotifyService:
    """通道的维护入口 + 一条告警的发送入口。

    两件事在一个类里不是凑合：它们共用同一份通道名单与同一个工厂，而「能维护却发不出去」
    与「能发出去却改不动」都是同一种故障（第二个入口走的是另一套选路）。
    """

    def __init__(
        self,
        channels: ChannelRepo,
        alerts: AlertRepo,
        users: UserRepo,
        *,
        settings,
        bus: EventBus,
        clock,
        factory=None,
        sleep: Sleeper = time.sleep,
        attempts: int = 3,
        env: DesktopEnv | None = None,
    ) -> None:
        self.channels = channels
        self.alerts = alerts
        self.users = users
        self.settings = settings
        self.bus = bus
        self.clock = clock
        self._factory = factory if factory is not None else build_notifier_factory()
        self.sleep = sleep
        self.attempts = max(1, int(attempts))
        self.env = env

    def dispatch(
        self,
        message: AlertMessage,
        *,
        user_id: str,
        channel_ids: Sequence[str] = (),
        rule_id: str | None = None,
        account_id: str | None = None,
    ) -> DispatchResult:
        """记录这条告警，然后把它送出去。**只写 `notify_log`，不碰 `alerts.status`**。"""
        user = self.users.get(user_id)
        if user is None:
            raise ValidationError(f"没有用户 {user_id}：告警没有接收人就不能发")
        alert = self.alerts.create(
            user_id,
            rule_id=rule_id,
            rule_name=message.rule_name,
            account_id=account_id,
            account_name=message.account_name,
            market=message.market,
            code=message.code,
            symbol_name=message.name,
            severity=message.severity,
            stage=message.stage,
            value_text=message.value_text,
            threshold_text=message.threshold_text,
            action_note=message.action_note,
            triggered_at=message.triggered_at,
            channel_ids=list(channel_ids),
            extra={"body": message.body},
        )
        # 发出去的那一份带着真实 id：库里那一行才是这条告警的身份（附录 A 的字段注释）
        message = replace(message, alert_id=alert.id)
        note = self._dnd_note(user, message)
        if note:
            # 压制发生在选路之前：没有「本来要发给谁」这件事被决定，那一行也就没有通道
            self.alerts.record(alert.id, None, SUPPRESSED, error=note, attempts=0)
            return DispatchResult(alert.id, (DeliveryReport("", None, SUPPRESSED, note, 0),))
        targets, refusal = self._route(user, message.severity, list(channel_ids))
        if not targets:
            self.alerts.record(alert.id, None, NO_CHANNEL, error=refusal, attempts=0)
            return DispatchResult(alert.id, (DeliveryReport("", None, NO_CHANNEL, refusal, 0),))
        return DispatchResult(
            alert.id, tuple(self._deliver(alert.id, message, node) for node in targets)
        )

    def probe(self, channel_id: str, *, user_id: str | None = None) -> DeliveryReport:
        """探针 = 发一条 weak 的合成消息（附录 D 开头那句话）。

        它**不**写 `notify_log`（那里每行都要挂一条 alert，而探针不是告警），但会
        `mark_ok`/`mark_error`：一次真发过的尝试，值得改变那条通道在健康页上的状态。
        """
        node = self._require(channel_id, user_id=user_id)
        report = self._attempt(node, probe_message(node.name))
        return report

    def local_status(self) -> dict[str, str]:
        """`{desktop: …, voice: …}`，值是可用的空串或那一句原因。健康页读它。"""
        env = self.env if self.env is not None else DesktopEnv.from_environment()
        return {
            "desktop": "" if env.desktop_available else env.desktop_reason(),
            "voice": "" if env.voice_available else env.voice_reason(),
        }

    # —— 维护入口（CLI 唯一认识的那一半）——

    def add(self, user_id: str, kind: str, name: str, config: Mapping[str, object]) -> Channel:
        """写之前先过一遍工厂：`kind` 拼错要在**落库之前**被拒。

        库里躺一条工厂不认识的 kind 的后果不是报错，是那条通道路由时被静默跳过，
        而用户看到的是「我配了通道却没收到」（`local_status` 与 `unknown_kinds` 都在补这一句）。
        """
        self._build(kind, config)  # 未知 kind / 缺必填字段 → ValidationError，从这里抛出去
        return self.channels.create(user_id, kind, name, config)

    def list(self, user_id: str) -> list[Channel]:
        return self.channels.list_for(user_id)

    def show(self, channel_id: str, *, user_id: str | None = None) -> Channel:
        return self._require(channel_id, user_id=user_id)

    def set_enabled(self, channel_id: str, enabled: bool, *, user_id: str | None = None) -> None:
        node = self._require(channel_id, user_id=user_id)
        self.channels.set_enabled(node.id, enabled)

    def set_default(self, channel_id: str, *, user_id: str | None = None) -> None:
        node = self._require(channel_id, user_id=user_id)
        self.channels.set_default(node.id)

    def remove(self, channel_id: str, *, user_id: str | None = None) -> None:
        node = self._require(channel_id, user_id=user_id)
        self.channels.remove(node.id)

    def set_dnd(self, user_id: str, *, start: dtime | None, end: dtime | None, tz: str | None) -> None:
        if self.users.get(user_id) is None:
            raise ValidationError(f"没有用户 {user_id}")
        self.users.set_dnd(user_id, start=start, end=end, tz=tz)

    def clear_dnd(self, user_id: str) -> None:
        if self.users.get(user_id) is None:
            raise ValidationError(f"没有用户 {user_id}")
        self.users.clear_dnd(user_id)

    # —— 内部 ——

    def _require(self, channel_id: str, *, user_id: str | None) -> Channel:
        node = self.channels.show(channel_id, user_id=user_id)
        if node is None:
            who = "" if user_id is None else f"（在用户 {user_id} 名下）"
            raise ValidationError(f"通道 {channel_id} 不存在{who}")
        return node

    def _build(self, kind: str, config: Mapping[str, object]) -> Notifier:
        return self._factory(kind, dict(config))

    def _dnd_note(self, user: User, message: AlertMessage) -> str:
        """在免打扰里就该压制的那一句原因；不该压制时空串。

        `strong` 不受挡：止损会落在北京 22:00（美股时段），而那只正是唯一一条
        不能错过的告警（§4.1 与 §11 相交处的方向性决定，T-14 有一枚用例钉它）。
        """
        if message.severity == STRONG:
            return ""
        if not self._in_dnd(user, message.triggered_at):
            return ""
        window = f"{user.dnd_start:%H:%M}-{user.dnd_end:%H:%M} {user.dnd_tz}"
        return f"{window} {DND_WORD}（{message.severity} 级被压制，strong 不受挡）"

    def _in_dnd(self, user: User, moment: datetime) -> bool:
        """跨午夜窗口用 `start <= t or t <= end`，非跨午夜用 `start <= t <= end`。

        时刻折到**用户的那个时区**而不是 `settings` 里的市场时区：免打扰讲的是这个人
        在睡觉，而他住哪儿与他的票在哪个交易所挂牌是两件事。
        """
        if user.dnd_start is None or user.dnd_end is None or not user.dnd_tz:
            return False
        try:
            zone = ZoneInfo(user.dnd_tz)
        except Exception as exc:  # noqa: BLE001 — 库里那串时区名可能是旧的
            logger.warning("用户 %s 的免打扰时区 %r 读不出：%s（按不在窗口里处理）",
                           user.username, user.dnd_tz, exc)
            return False
        local = (moment if moment.tzinfo else moment.replace(tzinfo=UTC)).astimezone(zone)
        now = local.timetz().replace(tzinfo=None)
        start, end = user.dnd_start, user.dnd_end
        return now >= start or now <= end if start > end else start <= now <= end

    def _route(
        self, user: User, severity: str, channel_ids: list[str]
    ) -> tuple[list[Channel], str]:
        picked, missing = [], []
        for channel_id in channel_ids:
            node = self.channels.show(channel_id, user_id=user.id)
            if node is None:
                missing.append(f"{channel_id}（不存在）")
            elif not node.enabled:
                missing.append(f"{node.name}（已停用）")
            else:
                picked.append(node)
        if picked:
            return picked, ""
        if channel_ids:
            # 规则指定了通道却全都不可用：不再降级到默认。降级会把一条明确指向 A 群的
            # 告警发到 B 群去，而屏幕上只写着「已发送」
            return [], (
                f"规则指定的通道都不可用：{'、'.join(missing)}。"
                f"修法：`sams channel enable --user {user.username} --channel <id>`"
                " 或改规则的 channel_ids"
            )
        wanted = set(route_for(severity))
        enabled = [node for node in self.channels.list_for(user.id) if node.enabled]
        by_level = [node for node in enabled if node.kind in wanted]
        if by_level:
            return by_level, ""
        default = next((node for node in enabled if node.is_default), None)
        if default is not None:
            return [default], ""
        if not enabled:
            return [], (
                f"用户 {user.username} 名下没有启用的通道，这条告警发不出去。"
                f"三步：`sams channel add --user {user.username} --kind wecom --name 主群 "
                "--config url=…` → `sams channel probe` 试一次 → 在规则里选它"
            )
        return [], (
            f"没有 kind 属于 {severity} 级默认路由（{sorted(wanted)}）的启用通道，"
            f"也没有默认通道；已有的是 {sorted({node.kind for node in enabled})}"
        )

    def _deliver(self, alert_id: str, message: AlertMessage, node: Channel) -> DeliveryReport:
        report = self._attempt(node, message)
        self.alerts.record(
            alert_id,
            node.id,
            report.outcome,
            error=report.error,
            attempts=report.attempts,
        )
        if report.outcome == FAILED:
            # §12 的「连续失败降 SystemAlarm」那个计数属于调度器（M3-D），不属于这里：
            # 一个 CLI 命令一个进程，进程里没有「连续」可言
            self.bus.publish(
                NotifyFailed(
                    alert_id=alert_id,
                    channel_id=node.id,
                    attempt=report.attempts,
                    error=report.error,
                )
            )
        return report

    def _attempt(self, node: Channel, message: AlertMessage) -> DeliveryReport:
        """真正的发送 + 退避。探针与真实告警共用这一个循环，唯一的区别是**调用方**要不要
        往 `notify_log` 写一行：探针不写（它不是一条告警的投递），但它一样会
        `mark_ok`/`mark_error`——一次真发过的尝试值得改变这条通道在健康页上的状态。"""
        try:
            notifier = self._build(node.kind, self.channels.credentials(node.id))
        except ValidationError as exc:
            self.channels.mark_error(node.id, error=str(exc))
            return DeliveryReport(node.kind, node.id, FAILED, str(exc), 0)
        except SamsError as exc:  # 解密失败之类：也是这条通道的失败，不是整条路由的失败
            self.channels.mark_error(node.id, error=str(exc))
            return DeliveryReport(node.kind, node.id, FAILED, str(exc), 0)
        if not notifier.available():
            # 自动禁用不算失败：远程终端上的 desktop/voice 每次都「失败」，
            # 健康页就永远在报故障，真坏掉的那条通道藏在噪音里
            return DeliveryReport(
                node.kind, node.id, SUPPRESSED, notifier.unavailable_reason(), 0
            )
        delays = backoff_delays(self.attempts)
        error = ""
        for index in range(self.attempts):
            result: NotifyResult = notifier.send(message)
            if result.ok:
                self.channels.mark_ok(node.id)
                return DeliveryReport(node.kind, node.id, SENT, "", index + 1)
            if result.unavailable:
                # 发出去才发现本机做不到（`osascript` 在 `PATH` 上消失了）：
                # 与 `available()` 那一支同待——压制、不记失败，理由见上面那条注释
                return DeliveryReport(node.kind, node.id, SUPPRESSED, result.error, index + 1)
            error = result.error or "（通道没有给原因）"
            if index < len(delays):
                logger.warning("通道 %s 第 %d 次失败（%s），%.1fs 后重试", node.name, index + 1,
                               error, delays[index])
                self.sleep(delays[index])
        self.channels.mark_error(node.id, error=error)
        return DeliveryReport(node.kind, node.id, FAILED, error, self.attempts)
```

```python
def _config_pairs(pairs: list[str]) -> dict[str, str]:
    out: dict[str, str] = {}
    for item in pairs:
        key, sep, value = item.partition("=")
        if not sep or not key.strip():
            raise ValidationError(f"--config 要的是 K=V，收到 {item!r}")
        out[key.strip()] = value
    return out


def _time_arg(value: str | None, flag: str) -> time | None:
    if value is None:
        return None
    try:
        return time.fromisoformat(value.strip())
    except ValueError:
        raise ValidationError(f"{flag} 要的是 HH:MM（24 小时），收到 {value!r}") from None


def _channel_row(node) -> dict:
    """出屏/出 JSON 的那一份视图：只有掩码值，永远不含 `config_encrypted`。"""
    return {
        "id": node.id,
        "name": node.name,
        "kind": node.kind,
        "enabled": node.enabled,
        "is_default": node.is_default,
        "last_ok_at": node.last_ok_at,
        "last_error": node.last_error,
        "last_error_at": node.last_error_at,
        "config": node.config,
    }


def add(container: Container, args) -> int:
    try:
        user = resolve_actor(container.users, args.user)
        node = container.notify.add(
            user.id, args.kind, args.name, _config_pairs(args.config)
        )
        if args.no_default and node.is_default:
            # 第一个通道会自动成为默认（附录 A）；--no-default 是在说「不要」，
            # 而这里不能把默认取消到「一个都没有」——那是兜底路由失效的形状
            raise ValidationError(
                "这是名下第一条通道，取消默认会留下「没有任何兜底」的状态；"
                "先加第二条，再 `sams channel default` 指过去"
            )
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "channel": _channel_row(node)})
    else:
        print(f"已添加通道 {node.name}（{node.kind}，默认={node.is_default}）")
        print(f"  下一步：sams channel probe --user {args.user} --channel {node.id}")
    return 0


def list_channels(container: Container, args) -> int:
    try:
        user = resolve_actor(container.users, args.user)
        rows = container.notify.list(user.id)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "channels": [_channel_row(node) for node in rows]})
        return 0
    if not rows:
        print(f"用户 {user.username} 名下没有通道：sams channel add --user {user.username} …")
        return 0
    print(f"通道（{user.username}）：")
    for node in rows:
        flags = "默认" if node.is_default else ("启用" if node.enabled else "停用")
        ok = stamp(node.last_ok_at) if node.last_ok_at else "从没成功发过"
        line = f"  {node.name:<{_LABEL}} {node.kind:<10} {flags:<4} {ok}"
        print(line + (f"\n      上次失败：{node.last_error}" if node.last_error else ""))
    return 0


def show(container: Container, args) -> int:
    try:
        user = resolve_actor(container.users, args.user)
        node = container.notify.show(args.channel, user_id=user.id)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "channel": _channel_row(node)})
        return 0
    print(f"通道 {node.name}（{node.kind}）")
    print(f"  状态：{'启用' if node.enabled else '停用'}｜默认={node.is_default}")
    print(f"  上次成功：{stamp(node.last_ok_at) if node.last_ok_at else '–'}")
    print(f"  上次失败：{node.last_error or '–'}")
    print("  配置（值已掩码）：")
    for key in sorted(node.config):
        print(f"    {key:<12} {node.config[key]}")
    return 0


def enable(container: Container, args) -> int:
    return _flip(container, args, True)


def disable(container: Container, args) -> int:
    return _flip(container, args, False)


def _flip(container: Container, args, on: bool) -> int:
    try:
        user = resolve_actor(container.users, args.user)
        node = container.notify.show(args.channel, user_id=user.id)
        container.notify.set_enabled(node.id, on, user_id=user.id)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "channel": node.id, "enabled": on})
    else:
        print(f"{'已启用' if on else '已停用'}通道 {node.name}")
        if not on:
            print("  注意：停用的通道仍在规则里被引用（`rules.channel_ids`），"
                  "那条规则命中时会写成一条 no_channel 记录")
    return 0


def default(container: Container, args) -> int:
    try:
        user = resolve_actor(container.users, args.user)
        node = container.notify.show(args.channel, user_id=user.id)
        container.notify.set_default(node.id, user_id=user.id)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "channel": node.id, "is_default": True})
    else:
        print(f"默认通道已设为 {node.name}（原默认仍在启用，只是不再是兜底）")
    return 0


def remove(container: Container, args) -> int:
    try:
        user = resolve_actor(container.users, args.user)
        node = container.notify.show(args.channel, user_id=user.id)
        if not args.yes:
            raise ValidationError(
                f"删除通道 {node.name} 需要 --yes：凭据只有那一份密文，删掉就要重新录入"
            )
        container.notify.remove(node.id, user_id=user.id)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "removed": node.id})
    else:
        print(f"已删除通道 {node.name}")
    return 0


def probe(container: Container, args) -> int:
    """退出码恒 0，即使探针失败：探针是在问一个问题，不是在承诺一个答案。

    与 `sams source probe` 同一个约定——把它做成非零，脚本里「配完通道顺手试一次」
    就会在探针报「远程终端没有桌面」时整条中断，而那句话本来就是要显示的答复本身。
    """
    try:
        user = resolve_actor(container.users, args.user)
        node = container.notify.show(args.channel, user_id=user.id)
        report = container.notify.probe(node.id, user_id=user.id)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({
            "status": "ok", "exit": 0,
            "probe": {
                "channel": node.id, "name": node.name, "kind": report.kind,
                "outcome": report.outcome, "error": report.error,
                "attempts": report.attempts,
            },
        })
    else:
        head = f"探针 → {node.name}（{report.kind}）："
        if report.outcome == SENT:
            print(f"{head}已送达。规则里选这条通道就能收到真实告警")
        elif report.outcome == SUPPRESSED:
            print(f"{head}本机发不出去（不算失败）：{report.error}")
        else:
            print(f"{head}没送出去：{report.error}")
    return 0


def dnd_set(container: Container, args) -> int:
    try:
        user = resolve_actor(container.users, args.user)
        container.notify.set_dnd(
            user.id,
            start=_time_arg(args.start, "--start"),
            end=_time_arg(args.end, "--end"),
            tz=(args.tz or None),
        )
        node = container.users.get(user.id)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "dnd": _dnd(node)})
    else:
        print(f"免打扰窗口：{node.dnd_start:%H:%M}–{node.dnd_end:%H:%M}（{node.dnd_tz}）")
        print("  strong 级不受挡（止损在深夜也必须响）；mid/weak 会被记成一条压制记录")
        print("  跨午夜是合法的：23:00–07:00 就是这条功能最主要的那个窗口")
    return 0


def dnd_show(container: Container, args) -> int:
    try:
        user = resolve_actor(container.users, args.user)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "dnd": _dnd(user)})
    else:
        print(_dnd_text(user))
    return 0


def dnd_clear(container: Container, args) -> int:
    try:
        user = resolve_actor(container.users, args.user)
        container.notify.clear_dnd(user.id)
    except SamsError as exc:
        return refused(args, exc, exit_for(exc))
    if args.json:
        emit({"status": "ok", "exit": 0, "dnd": {"start": None, "end": None, "tz": None}})
    else:
        print(f"已清除 {user.username} 的免打扰窗口（三个字段一起清，不留残留）")
    return 0


def _dnd(user) -> dict:
    return {
        "start": None if user.dnd_start is None else f"{user.dnd_start:%H:%M}",
        "end": None if user.dnd_end is None else f"{user.dnd_end:%H:%M}",
        "tz": user.dnd_tz,
    }


def _dnd_text(user) -> str:
    if user.dnd_start is None:
        return f"{user.username} 没有免打扰窗口：任何时刻都会发"
    return (
        f"{user.username} 的窗口：{user.dnd_start:%H:%M}–{user.dnd_end:%H:%M}"
        f"（{user.dnd_tz}）——strong 不受挡"
    )
```

  两个已经落进上面代码的选择，说明一下为什么长这样：`probe()` 直接把 `report.outcome` 与 `SENT`/`SUPPRESSED` 比，不建「outcome → 文案」的字典——三种可能里只有两种需要特殊文案，第三种（`failed`）是 `else`，写一张表等于把「默认那一支」也变成要维护的数据。启停只有 `_flip` 一个入口（`enable`/`disable` 各自只是一行调用），所以没有第二个 `_toggle` 需要存在。
